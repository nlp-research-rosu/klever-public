verification.k
     1	requires "semantic.k"
     2	
     3	module VERIFICATION
     4	  imports SEMANTIC
     5	
     6	  // Pure reference model.  Starting at candidate F, remove every occurrence
     7	  // of a factor while F * F can still fit in N; otherwise advance F.  At
     8	  // termination N is the greatest prime factor of the original input.
     9	  syntax Int ::= lpfSpec(Int, Int) [function]
    10	  rule lpfSpec(N, F) => N
    11	    requires F *Int F >Int N
    12	  rule lpfSpec(N, F) => lpfSpec(N /Int F, F)
    13	    requires F *Int F <=Int N andBool N %Int F ==Int 0
    14	  rule lpfSpec(N, F) => lpfSpec(N, F +Int 1)
    15	    requires F *Int F <=Int N andBool N %Int F =/=Int 0
    16	
    17	  // Named AST fragments are macros only: after kompilation the claims contain
    18	  // exactly the constructors emitted into solution.mpy.
    19	  syntax Stmt ::= "factorLoop" [macro]
    20	  rule factorLoop
    21	    => While(
    22	         Compare(BinOp("*", Name("factor"), Name("factor")),
    23	                 CmpOp("<=", Name("n"))),
    24	         If(Compare(BinOp("%", Name("n"), Name("factor")),
    25	                    CmpOp("==", Int(0))),
    26	            Assign(Name("n"), BinOp("//", Name("n"), Name("factor"))),
    27	            Assign(Name("factor"), BinOp("+", Name("factor"), Int(1)))))
    28	
    29	  syntax PyModule ::= "solutionModule" [macro]
    30	  rule solutionModule
    31	    => Module(
    32	         FuncDef("largest_prime_factor", Params("n"),
    33	           Assign(Name("factor"), Int(2))
    34	           factorLoop
    35	           Return(Name("n"))))
    36	endmodule

solution.py
     1	def largest_prime_factor(n: int):
     2	    factor = 2
     3	    while factor * factor <= n:
     4	        if n % factor == 0:
     5	            n = n // factor
     6	        else:
     7	            factor = factor + 1
     8	    return n

prove.sh
     1	#!/usr/bin/env bash
     2	set -euo pipefail
     3	set -x
     4	
     5	proof_build_dir="$(mktemp -d "$PWD/.kproof.XXXXXX")"
     6	trap 'rm -rf -- "$proof_build_dir"' EXIT
     7	
     8	# Recreate the submitted constructor term and require it to be current.
     9	python3 py2mpy.py solution.py > "$proof_build_dir/solution.mpy"
    10	cmp solution.mpy "$proof_build_dir/solution.mpy"
    11	
    12	# verification.k imports semantic.k, so this compiles both the interpreter and
    13	# the pure reference model into the definition used by krun and kprove.
    14	kompile verification.k \
    15	  --backend haskell \
    16	  --main-module VERIFICATION \
    17	  --syntax-module MPY-SYNTAX \
    18	  --output-definition "$proof_build_dir/verification-kompiled"
    19	
    20	# Exercise the actual generated program through the custom semantics.
    21	krun solution.mpy --definition "$proof_build_dir/verification-kompiled" \
    22	  -cN=13195 --output pretty | tee "$proof_build_dir/run-13195.out"
    23	grep -F 'result ( 29 )' "$proof_build_dir/run-13195.out"
    24	
    25	krun solution.mpy --definition "$proof_build_dir/verification-kompiled" \
    26	  -cN=2048 --output pretty | tee "$proof_build_dir/run-2048.out"
    27	grep -F 'result ( 2 )' "$proof_build_dir/run-2048.out"
    28	
    29	# Prove the loop refinement, universal end-to-end result, and both examples.
    30	kprove spec.k \
    31	  --definition "$proof_build_dir/verification-kompiled" \
    32	  --spec-module SPEC \
    33	  --output pretty
