     1	def fizz_buzz(n: int):
     2	    count = 0
     3	    i = 0
     4	    while i < n:
     5	        if i % 11 == 0 or i % 13 == 0:
     6	            x = i
     7	            while x > 0:

     9	                    count = count + 1
    10	                x = x // 10
    11	        i = i + 1
    12	    return count
