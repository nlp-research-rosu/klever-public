     3	module SPEC
     4	  imports VERIFICATION
     5	
     6	  // Inner-loop invariant: starting with x = X adds exactly the number of
     7	  // decimal sevens in X and terminates with x = 0.
     8	  claim <k> INNER-LOOP ~> REST => REST </k>
     9	        <env>
    10	          "count" |-> (C => C +Int digitSevens(X))
    11	          "i" |-> _I
    12	          "n" |-> _N
    13	          "x" |-> (X => 0)
    14	        </env>
    15	    requires X >=Int 0
    16	
    17	  // Outer-loop invariant: [I,N) contributes fizzFrom(I,N); x is zero at each
    18	  // outer-loop boundary because the inner loop exhausts it.
    19	  claim <k> OUTER-LOOP ~> REST => REST </k>
    20	        <env>
    21	          "count" |-> (C => C +Int fizzFrom(I, N))
    22	          "i" |-> (I => N)
    23	          "n" |-> N
    24	          "x" |-> 0
    25	        </env>
    26	    requires I >=Int 0 andBool I <=Int N
    27	
    28	  // End-to-end theorem for every integer input.  For N <= 0, fizzFrom(0,N)
    29	  // and the program's empty iteration both yield zero.
    30	  claim <fizz>
    31	          <k>
    32	            Module(
    33	              FuncDef("fizz_buzz", Params("n"),
    34	                Assign(Name("count"), Int(0))
    35	                Assign(Name("i"), Int(0))
    36	                OUTER-LOOP
    37	                Return(Name("count"))))

    39	          </k>

    44	            "n" |-> N

$ nl -ba /reference/k-proof/solution.py
     1	def fizz_buzz(n: int):
     2	    count = 0
     3	    i = 0
     4	    while i < n:
     5	        if i % 11 == 0 or i % 13 == 0:
     6	            x = i
     7	            while x > 0:
     8	                if x % 10 == 7:
     9	                    count = count + 1
    10	                x = x // 10
    11	        i = i + 1
    12	    return count
