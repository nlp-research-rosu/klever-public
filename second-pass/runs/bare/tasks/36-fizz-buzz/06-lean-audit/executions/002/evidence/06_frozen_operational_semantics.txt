     8	  syntax Program ::= "Module" "(" Stmts ")"
     9	
    10	  syntax Stmts ::= List{Stmt, ""}
    11	  syntax Stmt ::= "FuncDef" "(" String "," "Params" "(" String ")" "," Stmts ")"
    12	                | "Assign" "(" Expr "," Expr ")"
    13	                | "While" "(" Expr "," Stmts ")"
    14	                | "If" "(" Expr "," Stmts "," Stmts ")"
    15	                | "Return" "(" Expr ")"

    17	  syntax Expr ::= "Name" "(" String ")"
    18	                | "Int" "(" Int ")"
    19	                | "Bool" "(" Bool ")"
    20	                | "BinOp" "(" String "," Expr "," Expr ")"
    21	                | "Compare" "(" Expr "," CmpOp ")"
    22	                | "BoolOp" "(" String "," Expr "," Expr ")"
    23	  syntax CmpOp ::= "CmpOp" "(" String "," Expr ")"
    24	endmodule

    26	module SEMANTIC
    27	  imports MPY-SYNTAX
    28	  imports INT
    29	  imports BOOL
    30	  imports STRING
    31	  imports MAP
    32	
    33	  configuration
    34	    <fizz>
    35	      <k> $PGM:Program </k>
    36	      <input> $N:Int </input>
    37	      <env> .Map </env>
    38	      <result> 0 </result>
    39	    </fizz>
    40	
    41	  syntax KItem ::= assignTo(String)
    42	                 | binLeft(String, Expr)
    43	                 | binRight(String, Int)
    44	                 | compareLeft(String, Expr)
    45	                 | compareRight(String, Int)
    46	                 | orElse(Expr)
    47	                 | ifThenElse(Stmts, Stmts)
    48	                 | whileBody(Expr, Stmts)
    49	                 | "returnValue"
    50	
    51	  // Loading the translated module invokes its required entry point with the
    52	  // integer supplied in <input>.  Function-body execution below is ordinary
    53	  // small-step execution over the translated AST.
    54	  rule <k> Module(FuncDef("fizz_buzz", Params("n"), BODY)) => BODY </k>
    55	       <input> N </input>
    56	       <env> .Map => "n" |-> N "count" |-> 0 "i" |-> 0 "x" |-> 0 </env>
    57	
    58	  rule <k> (S:Stmt SS:Stmts) => S ~> SS ... </k>
    59	  rule <k> .Stmts => .K ... </k>
    60	
    61	  rule <k> Name(X) => Int(I) ... </k>
    62	       <env> ... X |-> I ... </env>
    63	
    64	  rule <k> Assign(Name(X), E) => E ~> assignTo(X) ... </k>
    65	  rule <k> Int(I) ~> assignTo(X) => .K ... </k>
    66	       <env> ... X |-> (_ => I) ... </env>
    67	
    68	  rule <k> BinOp(OP, L, R) => L ~> binLeft(OP, R) ... </k>
    69	  rule <k> Int(A) ~> binLeft(OP, R) => R ~> binRight(OP, A) ... </k>
    70	  rule <k> Int(B) ~> binRight("+", A) => Int(A +Int B) ... </k>
    71	  rule <k> Int(B) ~> binRight("%", A) => Int(A %Int B) ... </k>
    72	    requires B =/=Int 0
    73	  rule <k> Int(B) ~> binRight("//", A) => Int(A /Int B) ... </k>
    74	    requires B =/=Int 0
    75	
    76	  rule <k> Compare(L, CmpOp(OP, R)) => L ~> compareLeft(OP, R) ... </k>
    77	  rule <k> Int(A) ~> compareLeft(OP, R) => R ~> compareRight(OP, A) ... </k>
    78	  rule <k> Int(B) ~> compareRight("<", A) => Bool(A <Int B) ... </k>
    79	  rule <k> Int(B) ~> compareRight(">", A) => Bool(A >Int B) ... </k>
    80	  rule <k> Int(B) ~> compareRight("==", A) => Bool(A ==Int B) ... </k>
    81	
    82	  rule <k> BoolOp("or", L, R) => L ~> orElse(R) ... </k>
    83	  rule <k> Bool(true) ~> orElse(_) => Bool(true) ... </k>
    84	  rule <k> Bool(false) ~> orElse(R) => R ... </k>
    85	
    86	  rule <k> If(E, THEN, ELSE) => E ~> ifThenElse(THEN, ELSE) ... </k>
    87	  rule <k> Bool(true) ~> ifThenElse(THEN, _) => THEN ... </k>
    88	  rule <k> Bool(false) ~> ifThenElse(_, ELSE) => ELSE ... </k>
    89	
    90	  rule <k> While(E, BODY) => E ~> whileBody(E, BODY) ... </k>
    91	  rule <k> Bool(true) ~> whileBody(E, BODY) => BODY ~> While(E, BODY) ... </k>
    92	  rule <k> Bool(false) ~> whileBody(_, _) => .K ... </k>
    93	
    94	  rule <k> Return(E) => E ~> returnValue ... </k>
    95	  rule <k> Int(I) ~> returnValue ~> _REST:K => .K </k>
    96	       <result> _ => I </result>
    97	endmodule
