def _parse_axioms(output: str) -> set[str]:
    matches = re.findall(r"depends on axioms:\s*\[([^\]]*)\]", output)
    if len(matches) != 1:
        raise KleanFinalGateError("#print axioms output is missing or ambiguous")
    return {
        token.strip()
        for token in matches[0].split(",")
        if token.strip()
    }

def _input_hashes(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    candidate: Path | None,
) -> dict[str, Any]:
    try:
        return {
            "k_workspace_sha256": pipeline_contract.sha256_tree(
                frozen_input
            ),
            "stage1_export_sha256": klean_export.tree_digest(
                frozen_input
            ),
            "discovery_manifest_sha256": (
                pipeline_contract.sha256_file(discovery_manifest)
            ),
            "klean_generation_sha256": (
                pipeline_contract.sha256_tree(generation)
            ),
            "lean_workspace_sha256": (
                pipeline_contract.sha256_tree(candidate)
                if candidate is not None
                else None
            ),
        }
    except (
        OSError,
        klean_export.KleanExportError,
        pipeline_contract.PipelineContractError,
    ) as error:
        raise KleanFinalGateError(
            "cannot hash Stage 6 mechanical inputs"
        ) from error


def _require_bound_inputs(
    audit_input: dict[str, Any],
    observed: dict[str, Any],
) -> None:
    expected = audit_input["resolution"]["hashes"]
    for key, digest in observed.items():
        if expected.get(key) != digest:
            raise KleanFinalGateError(
                f"Stage 6 {key} differs from audit input"
            )
