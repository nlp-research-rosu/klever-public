     1	def starts_one_ends(n):
     2	    """
     3	    Given a positive integer n, return the count of the numbers of n-digit
     4	    positive integers that start or end with 1.
     5	    """
     6	    if n == 1:
     7	        return 1
     8	    else:
     9	        return 18 * 10 ** (n - 2)

    91	  rule evalInt(Name(X), (X |-> I) M) => I
    92	    requires notBool (X in_keys(M))
    93	  rule evalInt(BinOp("+", A, B), ENV) => evalInt(A, ENV) +Int evalInt(B, ENV)
    94	  rule evalInt(BinOp("-", A, B), ENV) => evalInt(A, ENV) -Int evalInt(B, ENV)
    95	  rule evalInt(BinOp("*", A, B), ENV) => evalInt(A, ENV) *Int evalInt(B, ENV)
    96	  rule evalInt(BinOp("**", A, B), ENV) => evalInt(A, ENV) ^Int evalInt(B, ENV)
    97

    99	  rule evalBool(Compare(A, CmpOp("==", B)), ENV)

COMMAND: nl -ba /reference/k-proof/verification.k
     1	requires "semantic.k"
     2	
     3	module VERIFICATION
     4	  imports MPY
     5	
     6	  // Mathematical specification by inclusion-exclusion.  For N > 1, fixing
     7	  // the first digit leaves 10 * 10^(N-2) choices; fixing the last leaves
     8	  // 9 * 10^(N-2), since the first digit cannot be zero; fixing both leaves
     9	  // 10^(N-2).  The union is starts + ends - overlap.
    10	  syntax Int ::= decimalMiddles(Int) [function]
    11	               | startsWithOne(Int) [function]
    12	               | endsWithOne(Int) [function]
    13	               | startsAndEndsWithOne(Int) [function]
    14	               | qualifyingCount(Int) [function]
    15	
    16	  rule decimalMiddles(K) => 10 ^Int K requires K >=Int 0
    17	  rule startsWithOne(N) => 10 *Int decimalMiddles(N -Int 2)
    18	    requires N >Int 1
    19	  rule endsWithOne(N) => 9 *Int decimalMiddles(N -Int 2)
    20	    requires N >Int 1
    21	  rule startsAndEndsWithOne(N) => decimalMiddles(N -Int 2)
    22	    requires N >Int 1
    23	
    24	  rule qualifyingCount(1) => 1
    25	  rule qualifyingCount(N)
    26	    => startsWithOne(N) +Int endsWithOne(N) -Int startsAndEndsWithOne(N)
    27	    requires N >Int 1
    28	endmodule
