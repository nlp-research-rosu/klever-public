    35	/-- Convert an Elan toolchain name to an Elan toolchain directory name. -/
    36	@[inline] partial def toolchain2Dir (toolchain : String) : FilePath :=
    37	  go "" 0
    38	where
    39	  go (acc : String) (pos : String.Pos) : FilePath :=
    40	    if h : toolchain.atEnd pos then
    41	      FilePath.mk acc
    42	    else
    43	      let c := toolchain.get' pos h
    44	      let pos' := toolchain.next' pos h
    45	      if c = '/' then
    46	        go (acc ++ "--") pos'
    47	      else if c = ':'  then
    48	        go (acc ++ "---") pos'
    49	      else
    50	        go (acc.push c) pos'

   160	/-! ## Detection Functions -/
   161	
   162	/--
   163	Attempt to detect an Elan installation by checking the `ELAN` and `ELAN_HOME`
   164	environment variables. If `ELAN` is set but empty, Elan is considered disabled.
   165	-/
   166	def findElanInstall? : BaseIO (Option ElanInstall) := do
   167	  if let some home ← IO.getEnv "ELAN_HOME" then
   168	    let elan := (← IO.getEnv "ELAN").getD "elan"
   169	    if elan.trim.isEmpty then
   170	      return none
   171	    else
   172	      return some {elan, home}
   173	  return none
   174	
   175	/--
   176	Attempt to find the sysroot of the given `lean` command (if it exists)
   177	by calling `lean --print-prefix` and returning the path it prints.
   178	Defaults to trying the `lean` in `PATH`.
   179	-/
   180	def findLeanSysroot? (lean := "lean") : BaseIO (Option FilePath) := do
   181	  let act : IO _ := do
   182	    let out ← IO.Process.output {
   183	      cmd := lean,
   184	      args := #["--print-prefix"]
   185	    }
   186	    if out.exitCode == 0 then
   187	      pure <| some <| FilePath.mk <| out.stdout.trim
   188	    else
   189	      pure <| none
   190	  act.catchExceptions fun _ => pure none
   191	
   192	/--
   193	Construct the `LeanInstall` object for the given Lean sysroot.
   194	
   195	Does the following:
   196	1. Find `lean`'s githash.
   197	2. Finds the  `ar` and `cc` to use with Lean.
   198	3. Computes the sub-paths of the Lean install.
   199	
   200	For (1), If `lake` is not-collocated with `lean`, invoke `lean --githash`;
   201	otherwise, use Lake's `Lean.githash`. If the invocation fails, `githash` is
   202	set to the empty string.
   203	
   204	For (2), if `LEAN_AR` or `LEAN_CC` are defined, it uses those paths.
   205	Otherwise, if Lean is packaged with an `llvm-ar` and/or `clang`, use them.
   206	If not, use the `ar` and/or `cc` from the `AR` / `CC` environment variables
   207	or the system's `PATH`. This last step is needed because internal builds of
   208	Lean do not bundle these tools (unlike user-facing releases).
   209	
   210	We also track whether `LEAN_CC` was set to determine whether it should
   211	be set in the future for `lake env`. This is because if `LEAN_CC` was not set,
   212	it needs to remain not set for `leanc` to work.
   213	Even setting it to the bundled compiler will break `leanc` -- see
   214	[leanprover/lean4#1281](https://github.com/leanprover/lean4/issues/1281).
   215	
   216	For (3), it assumes that the Lean installation is organized the normal way.
   217	That is, with its binaries located in `<lean-sysroot>/bin`, its
   218	Lean libraries in `<lean-sysroot>/lib/lean`, and its system libraries in
   219	`<lean-sysroot>/lib`.
   220	-/
   221	def LeanInstall.get (sysroot : FilePath) (collocated : Bool := false) : BaseIO LeanInstall := do
   222	  let githash ← do
   223	    if collocated then
   224	      pure Lean.githash
   225	    else
   226	      -- Remark: This is expensive (at least on Windows), so try to avoid it.
   227	      getGithash
   228	  let ar ← findAr
   229	  setCc {sysroot, githash, ar}
   230	where
   231	  getGithash := do
   232	    EIO.catchExceptions (h := fun _ => pure "") do
   233	      let out ← IO.Process.output {
   234	        cmd := leanExe sysroot |>.toString,
   235	        args := #["--githash"]
   236	      }
   237	      return out.stdout.trim
   238	  findAr := do
   239	    if let some ar ← IO.getEnv "LEAN_AR" then
   240	      return FilePath.mk ar
   241	    else
   242	      let ar := leanArExe sysroot
   243	      if (← ar.pathExists) then
   244	        return ar
   245	      else if let some ar ← IO.getEnv "AR" then
   246	        return ar
   247	      else
   248	        return "ar"
   249	  setCc i := do
   250	    if let some cc ← IO.getEnv "LEAN_CC" then
   251	      return withCustomCc i cc
   252	    else
   253	      let cc := leanCcExe sysroot
   254	      if (← cc.pathExists) then
   255	        return withInternalCc i cc
   256	      else if let some cc ← IO.getEnv "CC" then
   257	        return withCustomCc i cc
   258	      else
   259	        return withCustomCc i "cc"
