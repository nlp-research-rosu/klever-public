     1	def greatest_common_divisor(a: int, b: int) -> int:
     2	    if a < 0:
     3	        a = -a
     4	    if b < 0:
     5	        b = -b
     6	    r = 0
     7	    while b != 0:
     8	        r = a % b
     9	        a = b
    10	        b = r
    11	    return a

     6	  // Keeping gcdSpec constructor-valued makes the equations logical lemmas;

    13	  // reasoning uses the Euclidean equality simplifications below.

    15	    requires A >=Int 0

    17	    requires (A >=Int 0) andBool (B >Int 0) [simplification, concrete]
    18

    20	    requires A <Int 0
