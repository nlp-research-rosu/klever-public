     1	module VERIFICATION
     2	  imports INT
     3	  imports BOOL
     4	
     5	  // Inductive character sequences make the string traversal explicit.
     6	  syntax Chars ::= "nil" [constructor]
     7	                 | "cons" "(" Int "," Chars ")" [constructor]
     8	
     9	  // The contract-level decoder is independent of the AST evaluator.
    10	  syntax Int ::= decodeCode(Int) [function, total]
    11	  rule decodeCode(C) => ((C -Int 5 -Int 97) modInt 26) +Int 97
    12	
    13	  syntax Chars ::= decodeSpec(Chars) [function, total]
    14	  rule decodeSpec(nil) => nil
    15	  rule decodeSpec(cons(C, CS)) => cons(decodeCode(C), decodeSpec(CS))
    16	
    17	  syntax Int ::= encodeCode(Int) [function, total]
    18	  rule encodeCode(C) => ((C +Int 5 -Int 97) modInt 26) +Int 97
    19	
    20	  syntax Chars ::= encodeSpec(Chars) [function, total]
    21	  rule encodeSpec(nil) => nil
    22	  rule encodeSpec(cons(C, CS)) => cons(encodeCode(C), encodeSpec(CS))
    23	
    24	  syntax Bool ::= isLowerCode(Int) [function, total]
    25	  rule isLowerCode(C) => 97 <=Int C andBool C <=Int 122
    26	
    27	  syntax Bool ::= allLower(Chars) [function, total]
    28	  rule allLower(nil) => true
    29	  rule allLower(cons(C, CS)) => isLowerCode(C) andBool allLower(CS)
    30	endmodule

    31	  imports INT
    32	  imports STRING
    33	  imports BOOL
    34	  imports VERIFICATION
    35	
    36	  syntax Value ::= VInt(Int) [constructor]
    37	                 | VBool(Bool) [constructor]
    38	                 | VText(String) [constructor]
    39	                 | VChar(Int) [constructor]
    40	                 | VChars(Chars) [constructor]
    41	                 | VList(Chars) [constructor]
    42	
    43	  syntax KItem ::= exec(Stmts)
    44	                 | eval(Expr)
    45	                 | "discard"
    46	                 | "finish"
    47	                 | binLeft(String, Expr)
    48	                 | binRight(String, Value)
    49	                 | "doOrd"
    50	                 | "doChr"
    51	                 | "doJoin"
    52	                 | comp(String, Expr, Chars)
    53	                 | compTail(String, Expr, Chars, Int)
    54	                 | prepend(Int)

    57	    <m>
    58	      <k> $PGM:PyModule </k>
    59	      <s> nil </s>
    60	      <ch> 0 </ch>
    61	      <input> $INPUT:Chars </input>
    62	      <result> .K </result>
    63	    </m>
    64	
    65	  // Loading the translated definition also invokes its required entry point.
    66	  rule <k> Module(FuncDef("decode_shift", Params("s"), BODY)) => exec(BODY) </k>
    67	       <s> _ => CS </s>
    68	       <input> CS </input>
    69	
    70	  rule <k> exec(.Stmts) => .K </k>
    71	  rule exec(Expr(E) REST) => eval(E) ~> discard ~> exec(REST)
    72	  rule exec(Return(E) _REST) => eval(E) ~> finish
    73

    76	       <result> .K => V </result>

    78	  rule eval(Int(I)) => VInt(I)
    79	  rule eval(Bool(B)) => VBool(B)
    80	  rule eval(Str("")) => VChars(nil)
    81	  rule eval(Str(S)) => VText(S) requires S =/=String ""
    82	  rule <k> eval(Name("s")) => VChars(CS) ...</k>
    83	       <s> CS </s>
    84	  rule <k> eval(Name("ch")) => VChar(C) ...</k>
    85	       <ch> C </ch>
    86	
    87	  rule eval(BinOp(OP, LEFT, RIGHT))

    89	  rule V:Value ~> binLeft(OP, RIGHT)
    90	    => eval(RIGHT) ~> binRight(OP, V)
    91	  rule VInt(J) ~> binRight("+", VInt(I)) => VInt(I +Int J)
    92	  rule VInt(J) ~> binRight("-", VInt(I)) => VInt(I -Int J)
    93	  rule VInt(J) ~> binRight("%", VInt(I)) => VInt(I modInt J)

    95	  rule eval(Call(Name("ord"), ARG)) => eval(ARG) ~> doOrd
    96	  rule VChar(C) ~> doOrd => VInt(C)
    97	  rule VText(S) ~> doOrd => VInt(ordChar(S))
    98	
    99	  rule eval(Call(Name("chr"), ARG)) => eval(ARG) ~> doChr
   100	  rule VInt(C) ~> doChr => VChar(C)
   101	
   102	  rule eval(Call(Attribute(Str(""), "join"), ARG))
   103	    => eval(ARG) ~> doJoin
   104	  rule VList(CS) ~> doJoin => VChars(CS)
   105	
   106	  // A single-generator, unfiltered comprehension over a character sequence.
   107	  // The translator supplies Bool(true) when the Python comprehension has no ifs.
   108	  rule <k> eval(ListComp(E, CompFor(Name(X), Name(Y), Bool(true))))
   109	          => comp(X, E, CS) ...</k>
   110	       <s> CS </s>
   111	    requires Y ==String "s"
   112	
   113	  rule comp(_X, _E, nil) => VList(nil)
   114	  rule <k> comp(X, E, cons(C, CS))
   115	          => eval(E) ~> compTail(X, E, CS, OLD) ...</k>
   116	       <ch> OLD => C </ch>
   117	    requires X ==String "ch"
   118	  rule <k> VChar(D) ~> compTail(X, E, CS, SAVED)
   119	          => comp(X, E, CS) ~> prepend(D) ...</k>
   120	       <ch> _CURRENT => SAVED </ch>
   121	  rule VList(DS) ~> prepend(D) => VList(cons(D, DS))

     3	module SPEC
     4	  imports SEMANTIC
     5	
     6	  // The arithmetic used by decode_shift is the inverse of the prompt's
     7	  // encode_shift arithmetic for every lowercase character code.
     8	  claim <k> decodeCode(encodeCode(C:Int)) => C ...</k>
     9	    requires isLowerCode(C)
    10	  [label(code-inverse)]
    11	
    12	  // Loop invariant for the translated comprehension. The environment is
    13	  // restored after each character, so the invariant composes inductively.
    14	  claim
    15	    <k>
    16	      comp("ch",
    17	        Call(
    18	          Name("chr"),
    19	          BinOp(
    20	            "+",
    21	            BinOp(
    22	              "%",
    23	              BinOp(
    24	                "-",
    25	                BinOp("-", Call(Name("ord"), Name("ch")), Int(5)),
    26	                Call(Name("ord"), Str("a"))),
    27	              Int(26)),
    28	            Call(Name("ord"), Str("a")))),
    29	        CS:Chars) ~> KONT:K
    30	      => VList(decodeSpec(CS)) ~> KONT
    31	    </k>
