     7	def sort_even(l: list):
     8	    """This function takes a list l and returns a list l' such that
     9	    l' is identical to l in the odd indicies, while its values at the even indicies are equal
    10	    to the values of the even indicies of l, but sorted.
    11	    >>> sort_even([1, 2, 3])
    12	    [1, 2, 3]
    13	    >>> sort_even([5, 6, 3, 4])
    14	    [3, 6, 5, 4]
    15	    """
    16	    evens = l[::2]
    17	    odds = l[1::2]
    18	    evens.sort()
    19	    ans = []
    20	    for e, o in zip(evens, odds):
    21	        ans.extend([e, o])
    22	    if len(evens) > len(odds):
    23	        ans.append(evens[-1])
    24	    return ans
