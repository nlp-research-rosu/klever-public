def run_gate(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    candidate: Path | None,
    *,
    toolchain_lock: Path,
    audit_input: Path,
    output: Path,
) -> dict[str, Any]:
    output = Path(output)
    if output.exists() or output.is_symlink():
        raise KleanFinalGateError(f"output already exists: {output}")
    try:
        document = check_final(
            frozen_input,
            discovery_manifest,
            generation,
            candidate,
            toolchain_lock=toolchain_lock,
            audit_input=audit_input,
        )
    except KleanFinalGateError as error:
        document = {
            "status": "FAIL",
            "error": str(error),
            **_result_binding(audit_input),
        }
    except BaseException as error:
        document = {
            "status": "AUDIT_ERROR",
            "error": f"{type(error).__name__}: {error}",
            **_result_binding(audit_input),
        }
    temporary = output.with_name(f".{output.name}.tmp")
    temporary.write_text(json.dumps(document, indent=2, sort_keys=True) + "\n")
    os.replace(temporary, output)
    return document

def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--frozen-k", required=True, type=Path)
    parser.add_argument(
        "--discovery-manifest", required=True, type=Path
    )
    parser.add_argument("--generation", required=True, type=Path)
    parser.add_argument("--candidate", type=Path)
    parser.add_argument("--toolchain-lock", required=True, type=Path)
    parser.add_argument("--audit-input", required=True, type=Path)
    parser.add_argument("--output", type=Path)
    arguments = parser.parse_args(argv)
    try:
        if arguments.output is None:
            result = evaluate_gate(
                arguments.frozen_k,
                arguments.discovery_manifest,
                arguments.generation,
                arguments.candidate,
                toolchain_lock=arguments.toolchain_lock,
                audit_input=arguments.audit_input,
            )
        else:
            result = run_gate(
                arguments.frozen_k,
                arguments.discovery_manifest,
                arguments.generation,
                arguments.candidate,
                toolchain_lock=arguments.toolchain_lock,
                audit_input=arguments.audit_input,
                output=arguments.output,
            )
    except KleanFinalGateError as error:
        print(f"Klean final gate failed: {error}", file=sys.stderr)
        return 2
    print(json.dumps(result, sort_keys=True))
    return 0 if result["status"] == "PASS" else 1
