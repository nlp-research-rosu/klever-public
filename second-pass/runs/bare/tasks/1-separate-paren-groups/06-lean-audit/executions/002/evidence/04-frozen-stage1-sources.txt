     6	  // A direct contract model. It consumes input characters into a mathematical
     7	  // scanner state: depth, unfinished group, completed groups, and last loop
     8	  // character. Spaces change only the last-character component.
     9	  syntax ScanState ::= "scanState" "(" PInt "," Chars "," Outputs "," Chars ")"
    10	  syntax ScanState ::= "runSpec" "(" Chars "," PInt "," Chars "," Outputs "," Chars ")" [function]
    11	  syntax PInt ::= "stateDepth" "(" ScanState ")" [function]
    12	  syntax Chars ::= "stateCurrent" "(" ScanState ")" [function]
    13	                 | "stateLast" "(" ScanState ")" [function]
    14	  syntax Outputs ::= "stateOutput" "(" ScanState ")" [function]
    15	                   | "separateSpec" "(" Chars ")" [function]
    16	
    17	  rule runSpec(.Chars, D, CURRENT, OUT, LAST) => scanState(D, CURRENT, OUT, LAST)
    18	  rule runSpec(SP CS, D, CURRENT, OUT, _LAST) => runSpec(CS, D, CURRENT, OUT, SP)
    19	  rule runSpec(LP CS, D, CURRENT, OUT, _LAST) => runSpec(CS, succ(D), #concat(CURRENT, LP), OUT, LP)
    20	  rule runSpec(RP CS, zero, CURRENT, OUT, _LAST) => runSpec(CS, zero, .Chars, #snocOut(OUT, #concat(CURRENT, RP)), RP)
    21	  rule runSpec(RP CS, succ(zero), CURRENT, OUT, _LAST) => runSpec(CS, zero, .Chars, #snocOut(OUT, #concat(CURRENT, RP)), RP)
    22	  rule runSpec(RP CS, succ(succ(D)), CURRENT, OUT, _LAST) => runSpec(CS, succ(D), #concat(CURRENT, RP), OUT, RP)
    23	
    24	  rule stateDepth(scanState(D, _, _, _)) => D
    25	  rule stateCurrent(scanState(_, CURRENT, _, _)) => CURRENT
    26	  rule stateOutput(scanState(_, _, OUT, _)) => OUT
    27	  rule stateLast(scanState(_, _, _, LAST)) => LAST
    28	  rule separateSpec(CS) => stateOutput(runSpec(CS, zero, .Chars, .Outputs, .Chars))
    29	
    30	endmodule

     1	from typing import List
     2	
     3	
     4	def separate_paren_groups(paren_string: str) -> List[str]:
     5	    result = []
     6	    current = ""
     7	    depth = 0
     8	    ch = ""
     9	    for ch in paren_string:
    10	        if ch != " ":
    11	            current += ch
    12	            if ch == "(":
    13	                depth += 1
    14	            else:
    15	                depth -= 1
    16	                if depth == 0:
    17	                    result.append(current)
    18	                    current = ""
    19	    return result
