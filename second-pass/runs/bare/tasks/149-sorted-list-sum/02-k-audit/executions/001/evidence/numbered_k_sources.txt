===== semantic.k =====
     1	module MPY-SYNTAX
     2	  imports BOOL-SYNTAX
     3	  imports INT-SYNTAX
     4	  imports STRING-SYNTAX
     5	  imports MAP

     7	  // The constructor grammar emitted by py2mpy.py.  This task uses a deliberately
     8	  // small, explicit subset; unsupported Python constructs cannot silently run.
     9	  syntax Program ::= "Module" "(" Stmts ")" [symbol]
    10	
    11	  syntax Stmts ::= List{Stmt, ""}
    12	  syntax Stmt ::= "FuncDef" "(" String "," Params "," Stmts ")" [symbol]
    13	                | "Return" "(" Expr ")"                         [symbol]
    14	                | "If" "(" Expr "," Stmts "," Stmts ")"       [symbol]
    15	
    16	  syntax Params ::= "Params" "(" Strings ")" [symbol]
    17	  syntax Strings ::= List{String, ","}
    18	
    19	  syntax Exprs ::= List{Expr, ","}
    20	  syntax Expr ::= "Name" "(" String ")"                         [symbol]
    21	                | "Int" "(" Int ")"                             [symbol]
    22	                | "Str" "(" String ")"                          [symbol]
    23	                | "ListExpr" "(" Exprs ")"                      [symbol]
    24	                | "BinOp" "(" String "," Expr "," Expr ")"    [symbol]
    25	                | "BoolOp" "(" String "," Exprs ")"            [symbol]
    26	                | "Compare" "(" Expr "," CmpOps ")"            [symbol]
    27	                | "Call" "(" Expr "," Exprs ")"                [symbol]
    28	                | "Subscript" "(" Expr "," Index ")"           [symbol]
    29	                | "ValueExpr" "(" Value ")"                     [symbol]
    30	                | "ListComp" "(" Expr "," CompFors ")"          [symbol]
    31	                | "Lambda" "(" Params "," CellVars "," FreeVars "," Expr ")" [symbol]
    32	                | "TupleExpr" "(" Exprs ")"                     [symbol]
    33	                | "KwArg" "(" String "," Expr ")"              [symbol]
    34	  syntax CmpOps ::= List{CmpOp, ","}
    35	  syntax CmpOp ::= "CmpOp" "(" String "," Expr ")"             [symbol]
    36	  syntax Index ::= Expr
    37	                 | "Slice" "(" Bound "," Bound "," Bound ")"  [symbol]
    38	  syntax Bound ::= Expr | "NoBound" [symbol]
    39	  syntax CompFors ::= List{CompFor, ""}
    40	  syntax CompFor ::= "CompFor" "(" Expr "," Expr "," Exprs ")" [symbol]
    41	  syntax CellVars ::= "CellVars" "(" Strings ")" [symbol]
    42	  syntax FreeVars ::= "FreeVars" "(" Strings ")" [symbol]
    43	
    44	  syntax Words ::= List{String, ","}
    45	  syntax Value ::= "VInt" "(" Int ")"       [symbol]
    46	                 | "VStr" "(" String ")"    [symbol]
    47	                 | "VBool" "(" Bool ")"     [symbol]
    48	                 | "VList" "(" Words ")"    [symbol]
    49	                 | "VNone"                    [symbol]
    50	  syntax Values ::= List{Value, ","}
    51	
    52	  syntax Run ::= "Run" "(" Program "," Expr ")" [symbol]
    53	  syntax Result ::= "Result" "(" Value ")" [symbol]
    54	endmodule
    55	
    56	module MPY-SEMANTIC
    57	  imports MPY-SYNTAX
    58	  imports BOOL
    59	  imports INT
    60	  imports STRING
    61	  imports MAP
    62	
    63	  configuration <k> $PGM:Run </k>
    64	
    65	  syntax Value ::= evalExpr ( Expr , Map , Program ) [function]
    66	  syntax Values ::= evalExprs ( Exprs , Map , Program ) [function]
    67	  syntax Value ::= evalBinOp ( String , Value , Value ) [function]
    68	                 | evalCompare ( String , Value , Value ) [function]
    69	                 | evalBoolOp ( String , Values ) [function]
    70	                 | evalSubscript ( Value , Index , Map , Program ) [function]
    71	                 | callFunction ( Function , Values , Program ) [function]
    72	                 | returnedValue ( ExecResult ) [function]
    73	                 | makeList ( Values ) [function]
    74	  syntax Int ::= valueLength ( Value ) [function]
    75	  syntax Words ::= valueWords ( Value ) [function]
    76	                 | filterEven ( Words ) [function]
    77	                 | filterEvenDecision ( Bool , String , Words ) [function]
    78	                 | sortByKey ( Words ) [function]
    79	                 | insertBySemanticKey ( String , Words ) [function]
    80	                 | insertBySemanticDecision ( Bool , String , String , Words ) [function]
    81	  syntax Bool ::= semanticKeyLess ( String , String ) [function]
    82	
    83	  syntax Function ::= "Function" "(" Params "," Stmts ")" [symbol]
    84	                    | findFunction ( String , Stmts ) [function]
    85	  syntax Stmts ::= programStmts ( Program ) [function]
    86	  syntax Map ::= bindParams ( Params , Values ) [function]
    87	
    88	  syntax ExecResult ::= "Normal" [symbol]
    89	                      | "Returned" "(" Value ")" [symbol]
    90	                      | execStmts ( Stmts , Map , Program ) [function]
    91	                      | continueWith ( ExecResult , Stmts , Map , Program ) [function]
    92	                      | chooseBranch ( Value , Stmts , Stmts , Stmts , Map , Program ) [function]
    93	
    94	  rule <k> Run(P, E) => Result(evalExpr(E, .Map, P)) </k>
    95	
    96	  rule programStmts(Module(SS)) => SS
    97	
    98	  rule findFunction(F, FuncDef(F, PS, BODY) REST)
    99	    => Function(PS, BODY)
   100	  rule findFunction(F, FuncDef(G, PS, BODY) REST)
   101	    => findFunction(F, REST)
   102	    requires F =/=String G
   103	
   104	  rule bindParams(Params(.Strings), .Values) => .Map
   105	  rule bindParams(Params(X, XS), V, VS)
   106	    => (X |-> V) bindParams(Params(XS), VS)
   107	
   108	  rule evalExpr(ValueExpr(V), ENV, P) => V
   109	  rule evalExpr(Int(I), ENV, P) => VInt(I)
   110	  rule evalExpr(Str(S), ENV, P) => VStr(S)
   111	  rule evalExpr(Name(X), ENV, P) => {ENV[X]}:>Value
   112	
   113	  rule evalExprs(.Exprs, ENV, P) => .Values
   114	  rule evalExprs(E, ENV, P) => evalExpr(E, ENV, P)
   115	  rule evalExprs(E, ES, ENV, P)
   116	    => evalExpr(E, ENV, P), evalExprs(ES, ENV, P)
   117	
   118	  rule evalExpr(ListExpr(ES), ENV, P)
   119	    => makeList(evalExprs(ES, ENV, P))
   120	  rule makeList(.Values) => VList(.Words)
   121	  rule makeList(VStr(S)) => VList(S)
   122	  rule makeList(VStr(S), VS)
   123	    => prependWord(S, makeList(VS))
   124	
   125	  syntax Value ::= prependWord ( String , Value ) [function]
   126	  rule prependWord(S, VList(WS)) => VList(S, WS)
   127	
   128	  rule evalExpr(BinOp(OP, L, R), ENV, P)
   129	    => evalBinOp(OP, evalExpr(L, ENV, P), evalExpr(R, ENV, P))
   130	  rule evalBinOp("%", VInt(I), VInt(J)) => VInt(I %Int J)
   131	  rule evalBinOp("+", VList(LS), VList(RS)) => VList(LS, RS)
   132	
   133	  rule evalExpr(Compare(L, CmpOp(OP, R)), ENV, P)
   134	    => evalCompare(OP, evalExpr(L, ENV, P), evalExpr(R, ENV, P))
   135	  rule evalCompare("==", VInt(I), VInt(J)) => VBool(I ==Int J)
   136	  rule evalCompare("<", VInt(I), VInt(J)) => VBool(I <Int J)
   137	  rule evalCompare("<", VStr(S), VStr(T)) => VBool(S <String T)
   138	
   139	  rule evalExpr(BoolOp(OP, ES), ENV, P)
   140	    => evalBoolOp(OP, evalExprs(ES, ENV, P))
   141	  rule evalBoolOp("and", VBool(B1), VBool(B2))
   142	    => VBool(B1 andBool B2)
   143	
   144	  rule evalExpr(Call(Name("len"), E), ENV, P)
   145	    => VInt(valueLength(evalExpr(E, ENV, P)))
   146	
   147	  // Denotation of the exact comprehension emitted for
   148	  //   [word for word in lst if len(word) % 2 == 0]
   149	  // The binder names are matched, not hard-coded.
   150	  rule evalExpr(
   151	         ListComp(Name(X),
   152	           CompFor(Name(X), Name(L),
   153	             Compare(
   154	               BinOp("%", Call(Name("len"), Name(X)), Int(2)),
   155	               CmpOp("==", Int(0))))),
   156	         ENV,
   157	         P)
   158	    => VList(filterEven(valueWords({ENV[L]}:>Value)))
   159	
   160	  // Denotation of sorted(..., key=lambda word: (len(word), word)).
   161	  rule evalExpr(
   162	         Call(Name("sorted"), E,
   163	           KwArg("key",
   164	             Lambda(Params(X), CellVars(.Strings), FreeVars(.Strings),
   165	               TupleExpr(Call(Name("len"), Name(X)), Name(X))))),
   166	         ENV,
   167	         P)
   168	    => VList(sortByKey(valueWords(evalExpr(E, ENV, P))))
   169	
   170	  rule evalExpr(Call(Name(F), ES), ENV, P)
   171	    => callFunction(
   172	         findFunction(F, programStmts(P)),
   173	         evalExprs(ES, ENV, P),
   174	         P)
   175	    requires F =/=String "len" andBool F =/=String "sorted"
   176	  rule callFunction(Function(PS, BODY), VS, P)
   177	    => returnedValue(execStmts(BODY, bindParams(PS, VS), P))
   178	  rule returnedValue(Returned(V)) => V
   179	
   180	  rule valueLength(VStr(S)) => lengthString(S)
   181	  rule valueLength(VList(.Words)) => 0
   182	  rule valueLength(VList(S, SS)) => 1 +Int valueLength(VList(SS))
   183	
   184	  rule valueWords(VList(WS)) => WS
   185	
   186	  rule filterEven(.Words) => .Words
   187	  rule filterEven(S, SS)
   188	    => filterEvenDecision(lengthString(S) %Int 2 ==Int 0, S, filterEven(SS))
   189	  rule filterEvenDecision(B, S, SS) => S, SS
   190	    requires B
   191	  rule filterEvenDecision(B, S, SS) => SS
   192	    requires notBool B
   193	
   194	  rule sortByKey(.Words) => .Words
   195	  rule sortByKey(S, SS) => insertBySemanticKey(S, sortByKey(SS))
   196	  rule insertBySemanticKey(S, .Words) => S
   197	  rule insertBySemanticKey(S, T, TS)
   198	    => insertBySemanticDecision(semanticKeyLess(S, T), S, T, TS)
   199	  rule semanticKeyLess(S, T)
   200	    => (lengthString(S) <Int lengthString(T))
   201	       orBool ((lengthString(S) ==Int lengthString(T)) andBool (S <String T))
   202	  rule insertBySemanticDecision(B, S, T, TS) => S, T, TS
   203	    requires B
   204	  rule insertBySemanticDecision(B, S, T, TS)
   205	    => T, insertBySemanticKey(S, TS)
   206	    requires notBool B
   207	
   208	  rule evalExpr(Subscript(E, IX), ENV, P)
   209	    => evalSubscript(evalExpr(E, ENV, P), IX, ENV, P)
   210	  rule evalSubscript(VList(S, SS), Int(0), ENV, P) => VStr(S)
   211	  rule evalSubscript(VList(S, SS), Slice(Int(1), NoBound, NoBound), ENV, P)
   212	    => VList(SS)
   213	
   214	  rule execStmts(.Stmts, ENV, P) => Normal
   215	  rule execStmts(Return(E) REST, ENV, P)
   216	    => Returned(evalExpr(E, ENV, P))
   217	  rule execStmts(If(COND, THEN, ELSE) REST, ENV, P)
   218	    => chooseBranch(evalExpr(COND, ENV, P), THEN, ELSE, REST, ENV, P)
   219	
   220	  rule chooseBranch(VBool(B), THEN, ELSE, REST, ENV, P)
   221	    => continueWith(execStmts(THEN, ENV, P), REST, ENV, P)
   222	    requires B
   223	  rule chooseBranch(VBool(B), THEN, ELSE, REST, ENV, P)
   224	    => continueWith(execStmts(ELSE, ENV, P), REST, ENV, P)
   225	    requires notBool B
   226	  rule continueWith(Returned(V), REST, ENV, P) => Returned(V)
   227	  rule continueWith(Normal, REST, ENV, P) => execStmts(REST, ENV, P)
   228	endmodule
===== solution-program.k =====
     1	requires "semantic.k"

     3	module SOLUTION-PROGRAM
     4	  imports MPY-SEMANTIC
     5	  syntax Program ::= "solutionProgram" [function]
     6	  rule solutionProgram =>
     7	    Module(
     8	      FuncDef("sorted_list_sum", Params("lst"),
     9	        Return(
    10	          Call(
    11	            Name("sorted"),
    12	            ListComp(Name("word"),
    13	              CompFor(
    14	                Name("word"),
    15	                Name("lst"),
    16	                Compare(
    17	                  BinOp("%", Call(Name("len"), Name("word")), Int(2)),
    18	                  CmpOp("==", Int(0))))),
    19	            KwArg(
    20	              "key",
    21	              Lambda(
    22	                Params("word"),
    23	                CellVars(.Strings),
    24	                FreeVars(.Strings),
    25	                TupleExpr(Call(Name("len"), Name("word")), Name("word"))))))))
    26	endmodule
===== verification.k =====
     1	requires "semantic.k"
     2	requires "solution-program.k"
     3	
     4	module SORTED-LIST-VERIFICATION
     5	  imports SOLUTION-PROGRAM
     6	  imports BOOL
     7	  imports INT
     8	  imports STRING
     9	
    10	  // Independent mathematical specification: remove odd-length strings and
    11	  // insert every remaining string by the (length, lexicographic) key.
    12	  syntax Words ::= evenSorted ( Words ) [function]
    13	                 | insertByKey ( String , Words ) [function]
    14	                 | insertAfterDecision ( Bool , String , String , Words ) [function]
    15	                 | keepAfterDecision ( Bool , String , Words ) [function]
    16	                 | sortedListSumSpec ( Words ) [function]
    17	  syntax Bool ::= keyLess ( String , String ) [function]
    18	
    19	  rule keyLess(S, T)
    20	    => (lengthString(S) <Int lengthString(T))
    21	       orBool ((lengthString(S) ==Int lengthString(T)) andBool (S <String T))
    22	
    23	  rule sortedListSumSpec(WS) => sortByKey(filterEven(WS))
    24	
    25	  rule evenSorted(.Words) => .Words
    26	  rule evenSorted(S, SS)
    27	    => keepAfterDecision(
    28	         lengthString(S) %Int 2 ==Int 0,
    29	         S,
    30	         evenSorted(SS))
    31	  rule keepAfterDecision(B, S, SS) => insertByKey(S, SS)
    32	    requires B
    33	  rule keepAfterDecision(B, S, SS) => SS
    34	    requires notBool B
    35	
    36	  rule insertByKey(S, .Words) => S
    37	  rule insertByKey(S, T, TS)
    38	    => insertAfterDecision(keyLess(S, T), S, T, TS)
    39	  rule insertAfterDecision(B, S, T, TS) => S, T, TS
    40	    requires B
    41	  rule insertAfterDecision(B, S, T, TS) => T, insertByKey(S, TS)
    42	    requires notBool B
    43	
    44	endmodule
===== spec.k =====
     1	requires "verification.k"
     2	
     3	module SPEC
     4	  imports MPY-SEMANTIC
     5	  imports SORTED-LIST-VERIFICATION
     6	
     7	  // Universal functional-correctness theorem for every finite Words value.
     8	  // The specification is filter-even followed by ascending (length, text)
     9	  // sorting, with duplicates retained.
    10	  claim <k>
    11	          Run(solutionProgram,
    12	            Call(Name("sorted_list_sum"), ValueExpr(VList(INPUT:Words))))
    13	        => Result(VList(sortedListSumSpec(INPUT)))
    14	        </k>
    15	    [label(universal-correctness)]
    16	
    17	  // The empty-list base case.
    18	  claim <k>
    19	          Run(solutionProgram,
    20	            Call(Name("sorted_list_sum"), ValueExpr(VList(.Words))))
    21	        => Result(VList(evenSorted(.Words)))
    22	        </k>
    23	    [label(base)]
    24	
    25	  // Symbolic strings exercise both parity filtering and both ordering keys.
    26	  claim <k>
    27	          Run(solutionProgram,
    28	            Call(Name("sorted_list_sum"),
    29	              ValueExpr(VList(A:String, B:String))))
    30	        => Result(VList(evenSorted(A, B)))
    31	        </k>
    32	    requires lengthString(A) ==Int 2
    33	     andBool lengthString(B) ==Int 2
    34	     andBool A <String B
    35	    [label(symbolic-two)]
    36	
    37	  claim <k>
    38	          Run(solutionProgram,
    39	            Call(Name("sorted_list_sum"),
    40	              ValueExpr(VList(A:String, B:String))))
    41	        => Result(VList(evenSorted(A, B)))
    42	        </k>
    43	    requires lengthString(A) ==Int 2
    44	     andBool lengthString(B) ==Int 2
    45	     andBool notBool (A <String B)
    46	    [label(symbolic-two-reverse)]
    47	
    48	  claim <k>
    49	          Run(solutionProgram,
    50	            Call(Name("sorted_list_sum"),
    51	              ValueExpr(VList(A:String, B:String, C:String))))
    52	        => Result(VList(evenSorted(A, B, C)))
    53	        </k>
    54	    requires lengthString(A) ==Int 4
    55	     andBool lengthString(B) ==Int 2
    56	     andBool lengthString(C) ==Int 3
    57	    [label(symbolic-three)]
    58	
    59	  // Both examples stated in prompt.py.
    60	  claim <k>
    61	          Run(solutionProgram,
    62	            Call(Name("sorted_list_sum"),
    63	              ListExpr(Str("aa"), Str("a"), Str("aaa"))))
    64	        => Result(VList("aa"))
    65	        </k>
    66	    [label(prompt-example-one)]
    67	
    68	  claim <k>
    69	          Run(solutionProgram,
    70	            Call(Name("sorted_list_sum"),
    71	              ListExpr(Str("ab"), Str("a"), Str("aaa"), Str("cd"))))
    72	        => Result(VList("ab", "cd"))
    73	        </k>
    74	    [label(prompt-example-two)]
    75	endmodule
