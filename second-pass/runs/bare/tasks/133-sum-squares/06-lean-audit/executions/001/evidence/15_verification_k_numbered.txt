     3	module VERIFICATION
     4	  imports MPY
     5	
     6	  syntax Int ::= squareCeil(NumValue) [function]
     7	               | sumSquares(PList) [function, total]
     8	               | sumSquaresFrom(Int, PList) [function, total]
     9	
    10	  rule squareCeil(V) => ceilInt(V) *Int ceilInt(V)
    11	
    12	  rule sumSquares(VS) => sumSquaresFrom(0, VS)
    13	  rule sumSquaresFrom(A, nil) => A
    14	  rule sumSquaresFrom(A, cons(V, VS))
    15	    => sumSquaresFrom(A +Int squareCeil(V), VS)
    16	endmodule
