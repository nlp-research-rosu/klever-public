

def _starts_outer_transition(line: str) -> bool:
    keyword, rest, _word_end = _line_word(line)
    separated = bool(rest) and rest[0] in " \t"
    if keyword == "module":
        return separated and _MODULE_NAME_RE.fullmatch(rest) is not None
    if keyword == "endmodule":
        return not rest.strip()
    if keyword == "rule":
        return separated
    return keyword in _OUTER_SENTENCE_KEYWORDS and separated


def _terminal_attribute_line(line: str) -> bool:
    match = _TRAILING_ATTRIBUTES_RE.search(line)
    if match is None:
        return False
    start = match.start()
    return start == 0 or line[start - 1].isspace()


def _scan_rule_delimiters(
    line: str,
    delimiters: list[str],
) -> None:
    closing = {")": "(", "]": "[", "}": "{"}
    index = 0
    while index < len(line):
        if line[index] == "<":
            cell = _CELL_TAG_RE.match(line, index)
            if cell is not None:
                marker = f"<{cell.group('name')}>"
                if cell.group("close"):
                    if delimiters and delimiters[-1] == marker:
                        delimiters.pop()
                elif not cell.group("self_close"):
                    delimiters.append(marker)
                index = cell.end()
                continue
        character = line[index]
        if character in "([{":
            delimiters.append(character)
        elif character in closing:
            if delimiters and delimiters[-1] == closing[character]:
                delimiters.pop()
        index += 1


def _rule_bubble_end(masked_text: str, start: int) -> int:
    """Find when an outer rule bubble returns control to outer parsing."""

    seen_rewrite = False
    clause_has_content = False
    candidate_end: int | None = None
    delimiters: list[str] = []
    ranges = _line_ranges(masked_text, start)
    for line_start, line_end, next_start in ranges:
        line = masked_text[line_start:line_end]
        if not line.strip():
            continue

        continuation = (
            _rule_continuation(line)
            if seen_rewrite and candidate_end is not None
            else None
        )
        if candidate_end is not None:
            if continuation is None and _starts_outer_transition(line):
                return candidate_end
            candidate_end = None

        content_start = 0
        if continuation is not None:
            kind, content_start = continuation
            if kind in {"requires", "ensures"}:
                clause_has_content = False

        if not seen_rewrite:
            rewrite = line.find("=>")
            if rewrite >= 0:
                seen_rewrite = True
                content_start = rewrite + 2
            else:
                content_start = len(line)

        _scan_rule_delimiters(line, delimiters)
        if seen_rewrite and line[content_start:].strip():
            clause_has_content = True

        if (
            seen_rewrite
            and clause_has_content
            and not delimiters
            and _terminal_attribute_line(line)
        ):
            return next_start
        if seen_rewrite and clause_has_content and not delimiters:
            candidate_end = next_start
    if candidate_end is not None:
        return candidate_end
    return len(masked_text) + 1


def _modules(masked_text: str) -> tuple[_Module, ...]:
    modules: list[_Module] = []
    open_module: tuple[
        str, int, list[str], list[_Sentence]
    ] | None = None
    open_rule_until = 0
    for line_start, line_end, keyword, rest in _line_heads(masked_text):
        if line_start < open_rule_until:
            continue
        open_rule_until = 0
        separated = bool(rest) and rest[0] in " \t"
        if keyword == "module":
            if not separated:
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            named = _MODULE_NAME_RE.fullmatch(rest)
            if named is None or open_module is not None:
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            open_module = (named.group("name"), line_end, [], [])
            continue
        if keyword == "endmodule":
            if rest.strip():
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            if open_module is None:
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            name, content_start, imports, sentences = open_module
            modules.append(
                _Module(
                    name=name,
                    content_start=content_start,
                    content_end=line_start,
                    imports=tuple(imports),
                    sentences=tuple(sentences),
                )
            )
            open_module = None
            continue
        if open_module is not None and keyword == "rule":
            open_module[3].append(
                _Sentence(keyword=keyword, start=line_start)
            )
            open_rule_until = _rule_bubble_end(masked_text, line_start)
            continue
        if open_module is not None and keyword == "imports" and separated:
            imported = rest.strip().split()
            if len(imported) != 1:
                raise KRuleInventoryError(
                    "verification.k has malformed imports"
                )
            open_module[2].append(imported[0])
        if (
            open_module is not None
            and keyword in _OUTER_SENTENCE_KEYWORDS
            and separated
        ):
            open_module[3].append(
                _Sentence(keyword=keyword, start=line_start)
            )
    if not modules or open_module is not None:
        raise KRuleInventoryError(
            "verification.k has malformed module boundaries"
        )
    return tuple(modules)


def _verification_module(
    modules: tuple[_Module, ...], selected_name: str | None = None
) -> _Module:
    if selected_name is not None:
        selected = [module for module in modules if module.name == selected_name]
        if len(selected) != 1:
            raise KRuleInventoryError(
                "prove.sh selects an unknown verification main module: "
                + selected_name
            )
        return selected[0]
    candidates = [
        module
        for module in modules
        if module.name.endswith("VERIFICATION")
        and not module.name.endswith("-SYNTAX")
    ]
    if len(candidates) != 1:
        raise KRuleInventoryError(
            "verification.k must contain exactly one verification module"
        )
    return candidates[0]


def _last_verification_main_module(workspace: Path) -> str | None:
    script = workspace / "prove.sh"
    if not script.exists() and not script.is_symlink():
        return None
    try:
        mode = script.stat(follow_symlinks=False).st_mode
    except OSError as error:
        raise KRuleInventoryError(f"cannot inspect prove.sh: {script}") from error
    if not stat.S_ISREG(mode):
        raise KRuleInventoryError(f"prove.sh must be a regular file: {script}")
    try:
        text = script.read_text()
    except (OSError, UnicodeError) as error:
        raise KRuleInventoryError(f"cannot read prove.sh: {script}") from error
    logical = re.sub(r"\\\r?\n", " ", text)
    selected: str | None = None
    for line in logical.splitlines():
        code = line.split("#", 1)[0]
        if not re.search(r"(?:^|[;&|]\s*)kompile\s+", code):
            continue
        if re.search(r"(?:^|\s)(?:\./)?verification\.k(?:\s|$)", code) is None:
            continue
        match = re.search(
            r"(?:^|\s)--main-module(?:=|\s+)([A-Za-z][A-Za-z0-9_'-]*)",
            code,
        )
        if match is not None:
            selected = match.group(1)
    return selected


def _verification_closure(
    modules: tuple[_Module, ...], verification: _Module
) -> tuple[_Module, ...]:
    by_name = {module.name: module for module in modules}
    reached: set[str] = set()
    pending = [verification.name]
    while pending:
        name = pending.pop()
        if name in reached:
            continue
        reached.add(name)
        module = by_name.get(name)
        if module is not None:
            pending.extend(module.imports)
    return tuple(module for module in modules if module.name in reached)


def _without_comments(text: str) -> str:
    """Remove K comments while preserving strings and their exact contents."""

    output: list[str] = []
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                output.append(character)
                state = "code"
            index += 1
            continue
        if state == "string":
            output.append(character)
            if character == "\\" and following:
                output.append(following)
                index += 2
                continue
            if character == '"':
                state = "code"
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character in "\r\n":
                output.append(character)
            index += 1
            continue
        if character == "/" and following == "/":
            output.append(" ")
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output.append(" ")
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        output.append(character)
