     1	module MPY-SYNTAX
     2	  imports INT-SYNTAX
     3	  imports STRING-SYNTAX
     4	
     5	  syntax PyModule ::= "Module" "(" PyStmts ")" [symbol(Module)]
     6	
     7	  syntax PyStmts ::= List{PyStmt, ""}
     8	
     9	  syntax PyStmt ::= "ImportFrom" "(" String "," String ")" [symbol(ImportFrom)]
    10	                  | "FuncDef" "(" String "," "Params" "(" String ")" "," PyStmts ")" [symbol(FuncDef)]
    11	                  | "Assign" "(" PyExpr "," PyExpr ")" [symbol(Assign)]
    12	                  | "AugAssign" "(" PyExpr "," String "," PyExpr ")" [symbol(AugAssign)]
    13	                  | "For" "(" PyExpr "," PyExpr "," PyStmts ")" [symbol(For)]
    14	                  | "Return" "(" PyExpr ")" [symbol(Return)]
    15	
    16	  syntax PyExpr ::= "Name" "(" String ")" [symbol(Name)]
    17	                  | "Int" "(" Int ")" [symbol(IntExpr)]
    18	                  | "Call" "(" PyExpr "," PyExpr ")" [symbol(Call)]
    19	                  | "BinOp" "(" String "," PyExpr "," PyExpr ")" [symbol(BinOp)]
    20	endmodule
    21	
    22	module MPY-DOMAIN
    23	  imports INT
    24	
    25	  // Exact rationals model numeric inputs without relying on backend-specific
    26	  // floating-point hooks.  PosNat makes every denominator positive.
    27	  syntax PosNat ::= "one" [symbol(one)]
    28	                  | "next" "(" PosNat ")" [symbol(next)]
    29	                  | "ten" [symbol(ten)]
    30	
    31	  syntax Int ::= posInt(PosNat) [function, total]
    32	
    33	  rule posInt(one) => 1
    34	  rule posInt(next(P)) => 1 +Int posInt(P)
    35	  rule posInt(ten) => 10
    36	
    37	  syntax NumValue ::= "intVal" "(" Int ")" [symbol(intVal)]
    38	                    | "ratVal" "(" Int "," PosNat ")" [symbol(ratVal)]
    39	
    40	  syntax PList ::= "nil" [symbol(nil)]
    41	                 | "cons" "(" NumValue "," PList ")" [symbol(cons)]
    42	
    43	  syntax PValue ::= NumValue
    44	                  | "listVal" "(" PList ")" [symbol(listVal)]
    45	
    46	  syntax Int ::= ceilInt(NumValue) [function]
    47	
    48	  rule ceilInt(intVal(I)) => I
    49	  rule ceilInt(ratVal(N, D))
    50	    => 0 -Int ((0 -Int N) divInt posInt(D))
    51	endmodule
    52	
    53	module MPY
    54	  imports MPY-SYNTAX
    55	  imports MPY-DOMAIN
    56	  imports MAP
    57	  imports STRING
    58	
    59	  syntax Function ::= "function" "(" String "," PyStmts ")" [symbol(function)]
    60	  syntax Env ::= ".Env" [symbol(emptyEnv)]
    61	               | "binding" "(" String "," PValue ")" Env [symbol(binding)]
    62	
    63	  syntax PValue ::= evalExpr(PyExpr, Env) [function]
    64	                  | lookupValue(Env, String) [function]
    65	                  | ceilValue(PValue) [function]
    66	                  | addValue(PValue, PValue) [function]
    67	                  | mulValue(PValue, PValue) [function]
    68	
    69	  rule evalExpr(Int(I), _) => intVal(I)
    70	  rule lookupValue(binding(X, V) _, X) => V
    71	  rule lookupValue(binding(Y, _) RHO, X) => lookupValue(RHO, X)
    72	    requires X =/=String Y
    73	  rule evalExpr(Name(X), RHO) => lookupValue(RHO, X)
    74	  rule evalExpr(Call(Name("ceil"), E), RHO) => ceilValue(evalExpr(E, RHO))
    75	  rule evalExpr(BinOp("*", E1, E2), RHO)
    76	    => mulValue(evalExpr(E1, RHO), evalExpr(E2, RHO))
    77	  rule evalExpr(BinOp("+", E1, E2), RHO)
    78	    => addValue(evalExpr(E1, RHO), evalExpr(E2, RHO))
    79	
    80	  rule ceilValue(V:NumValue) => intVal(ceilInt(V))
    81	  rule addValue(intVal(I1), intVal(I2)) => intVal(I1 +Int I2)
    82	  rule mulValue(intVal(I1), intVal(I2)) => intVal(I1 *Int I2)
    83	
    84	  syntax KItem ::= load(PyStmts) [symbol(load)]
    85	                 | loadStmt(PyStmt) [symbol(loadStmt)]
    86	                 | callEntry(PValue) [symbol(callEntry)]
    87	                 | exec(PyStmts) [symbol(exec)]
    88	                 | loop(String, PValue, PyStmts) [symbol(loop)]
    89	                 | bind(String, PValue) [symbol(bind)]
    90	
    91	  configuration
    92	    <python>
    93	      <k> $PGM:PyModule ~> callEntry($ARGS:PValue) </k>
    94	      <functions> .Map </functions>
    95	      <env> .Env </env>
    96	    </python>
    97	
    98	  rule <k> Module(SS) => load(SS) ... </k>
    99	
   100	  rule <k> load(.PyStmts) => .K ... </k>
   101	  rule <k> load(S:PyStmt SS:PyStmts) => loadStmt(S) ~> load(SS) ... </k>
   102	
   103	  rule <k> loadStmt(ImportFrom(_, _)) => .K ... </k>
   104	  rule <k> loadStmt(FuncDef(F, Params(P), BODY)) => .K ... </k>
   105	       <functions> FUNS => FUNS [ F <- function(P, BODY) ] </functions>
   106	
   107	  rule <k> callEntry(V) => exec(BODY) ... </k>
   108	       <functions> ... "sum_squares" |-> function(P, BODY) ... </functions>
   109	       <env> _ => binding(P, V) .Env </env>
   110	
   111	  rule <k> exec(.PyStmts) => .K ... </k>
   112	
   113	  rule <k> exec(Assign(Name(X), E) SS) => exec(SS) ... </k>
   114	       <env> RHO => binding(X, evalExpr(E, RHO)) RHO </env>
   115	
   116	  rule <k> exec(AugAssign(Name(X), "+", E) SS) => exec(SS) ... </k>
   117	       <env> RHO => binding(X, addValue(lookupValue(RHO, X), evalExpr(E, RHO))) RHO </env>
   118	
   119	  rule <k> exec(For(Name(X), E, BODY) SS) => loop(X, evalExpr(E, RHO), BODY) ~> exec(SS) ... </k>
   120	       <env> RHO </env>
   121	
   122	  rule <k> exec(Return(E) _SS) => evalExpr(E, RHO) ... </k>
   123	       <env> RHO </env>
   124	
   125	  rule <k> loop(_, listVal(nil), _) => .K ... </k>
   126	  rule <k> loop(X, listVal(cons(V, VS)), BODY)
   127	        => bind(X, V) ~> exec(BODY) ~> loop(X, listVal(VS), BODY) ... </k>
   128	
   129	  rule <k> bind(X, V) => .K ... </k>
   130	       <env> RHO => binding(X, V) RHO </env>
   131	endmodule
