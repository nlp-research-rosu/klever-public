def _optional_tree_hash(root: Path) -> str | None:
    try:
        mode = root.lstat().st_mode
    except OSError:
        return None
    if not stat.S_ISDIR(mode):
        return None
    try:
        return sha256_tree(root)
    except AuditContractError:
        return None
