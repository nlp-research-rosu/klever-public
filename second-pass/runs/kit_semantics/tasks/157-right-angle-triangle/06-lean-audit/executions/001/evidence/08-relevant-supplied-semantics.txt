     1	// reference/src/semantics/float.k — MPY-FLOAT
     2	//
     3	// Minimal, TRUSTED float support. K/Haskell (kprove) has no usable symbolic IEEE-754 theory and
     4	// no Int2Float hook, so a float value stays an OPAQUE proof-domain primitive: `intFloatDiv(I, F)`
     5	// (the Python true division I / F) is `[no-evaluators]` for kprove — it never reduces on symbolic
     6	// input, and the proof verifies only the STRUCTURE (the result is the exact term the program
     7	// builds). The `[concrete]` rule evaluates it for krun (llvm HAS the Int2Float / /Float hooks), so
     8	// smoke tests run the real float value.
     9	//
    10	// Only the pieces needed so far: Float as a Val, the float literal, and Int / float true division
    11	// (the `a * h / 2.0` idiom). More float ops (float +/-/*, comparison) are added as float problems
    12	// need them — kept minimal + sound.
    13	
    14	module MPY-FLOAT
    15	  imports MPY-OPERATORS
    16	  imports MPY-BUILTINS
    17	  imports FLOAT

    19	  // Float is a value; the float literal evaluates to the K Float.
    20	  syntax Val ::= Float
    21	  rule <k> Float(F:Float) => F ... </k>
    22	
    23	  // Int / float true division. OPAQUE for kprove (no-evaluators); concrete for krun.
    24	  syntax Float ::= intFloatDiv(Int, Float) [function, total, symbol(intFloatDiv), no-evaluators]
    25	  rule intFloatDiv(I:Int, F:Float) => Int2Float(I, 53, 11) /Float F [concrete]
    26	
    27	  rule applyBin("/", I:Int, F:Float) => intFloatDiv(I, F)
    28	
    29	  // Int / Int true division (Python: always a float). OPAQUE for kprove; concrete for krun.
    30	  syntax Float ::= divII(Int, Int) [function, total, symbol(divII), no-evaluators]
    31	  rule divII(I1:Int, I2:Int) => Int2Float(I1, 53, 11) /Float Int2Float(I2, 53, 11) [concrete]
    32	  rule applyBin("/", I1:Int, I2:Int) => divII(I1, I2)

    34	  // float % float (e.g. `number % 1.0` = the fractional part). OPAQUE for kprove, concrete for
    35	  // krun. Python's float `%` is floor-based: a % b = a - floor(a/b)*b (K's %Float is IEEE
    36	  // remainder, which differs — 3.5 %Float 1.0 = -0.5 vs Python 0.5 — so it is NOT used).
    37	  syntax Float ::= floatMod(Float, Float) [function, total, symbol(floatMod), no-evaluators]
    38	  rule floatMod(F1:Float, F2:Float) => F1 -Float (floorFloat(F1 /Float F2) *Float F2) [concrete]
    39	  rule applyBin("%", F1:Float, F2:Float) => floatMod(F1, F2)

    42	  // concrete floats. kprove proofs return floats structurally and do not compare them.
    43	  rule applyCmp("==", F1:Float, F2:Float) => F1 ==Float F2
    44	  rule applyCmp("!=", F1:Float, F2:Float) => notBool (F1 ==Float F2)

    46	  // float `<` and abs — OPAQUE for kprove ([no-evaluators], so a SYMBOLIC float compare stays an
    47	  // uninterpreted Bool a proof case-splits on — a SINGLE compare per branch, not a nonlinear cascade),
    48	  // [concrete] for krun. Additive, sort-disjoint from the Int rules. (has_close_elements: the pairwise

    52	  rule applyCmp("<", F1:Float, F2:Float) => floatLt(F1, F2)

    84	  rule <k> V1:Val ~> #mathPow1(E2:Expr) => E2 ~> #mathPow2(V1) ... </k>
    85	  rule <k> V2:Val ~> #mathPow2(V1:Val) => powF(toF(V1), toF(V2)) ... </k>

    88	  rule toF(I:Int)   => intToF(I) [concrete]

    95	  rule ceilF(F:Float) => Float2Int(ceilFloat(F)) [concrete]

   103	  syntax Float ::= subF(Float, Float) [function, total, symbol(subF), no-evaluators]
   104	  rule subF(F1:Float, F2:Float) => F1 -Float F2 [concrete]
   105	  rule applyBin("-", F1:Float, F2:Float) => subF(F1, F2)

   108	  rule divF(F1:Float, F2:Float) => F1 /Float F2 [concrete]
   109	  rule applyBin("/", F1:Float, F2:Float) => divF(F1, F2)

   111	  syntax Float ::= addF(Float, Float) [function, total, symbol(addF), no-evaluators]
   112	  rule addF(F1:Float, F2:Float) => F1 +Float F2 [concrete]
   113	  rule applyBin("+", F1:Float, F2:Float) => addF(F1, F2)
   114

   116	  rule mulF(F1:Float, F2:Float) => F1 *Float F2 [concrete]
   117	  rule applyBin("*", F1:Float, F2:Float) => mulF(F1, F2)

   137	  rule applyBin("+",  F:Float, I:Int) => addF(F, intToF(I))

   141	  // ---- exact mixed Int/Float comparison ------------------------------------
   142	  // The old rules compared via intToF(I), which ROUNDS ints >= 2^53 to a nearby
   143	  // double (so e.g. 9007199254740993 > 9007199254740992.0 came out false). Compare
   144	  // the REAL values instead: for a finite float F and an integer I,
   145	  //     F <  I   <=>   floor(F) <  I
   146	  //     F >  I   <=>   ceil(F)  >  I
   147	  //     F == I   <=>   F is integral  AND  floor(F) == I
   148	  // floorFloat/ceilFloat are exact and Float2Int yields the EXACT integer of an
   149	  // integral float (no rounding), so this is exact for every finite F. Non-finite
   150	  // F keeps the old intToF path (rounding is irrelevant to +/-inf and NaN, and
   151	  // inf/NaN have no integer floor). Opaque for kprove, [concrete] for krun.
   152	  syntax Bool ::= eqF(Float, Float) [function, total, symbol(eqF), no-evaluators]
   153	  rule eqF(F1:Float, F2:Float) => F1 ==Float F2 [concrete]
   154

   156	  rule floatFinite(F:Float) => notBool isNaN(F) andBool notBool isInfinite(F) [concrete]
   157	
   158	  syntax Bool ::= ltFI(Float, Int) [function, total, symbol(ltFI), no-evaluators]   // F <  I
   159	                | ltIF(Int, Float) [function, total, symbol(ltIF), no-evaluators]   // I <  F
   160	                | eqIF(Int, Float) [function, total, symbol(eqIF), no-evaluators]   // I == F
   161	  rule ltFI(F:Float, I:Int) => Float2Int(floorFloat(F)) <Int I  requires floatFinite(F)          [concrete]
   162	  rule ltFI(F:Float, I:Int) => floatLt(F, intToF(I))            requires notBool floatFinite(F)   [concrete]
   163	  rule ltIF(I:Int, F:Float) => Float2Int(ceilFloat(F))  >Int I  requires floatFinite(F)          [concrete]
   164	  rule ltIF(I:Int, F:Float) => floatLt(intToF(I), F)            requires notBool floatFinite(F)   [concrete]
   165	  rule eqIF(I:Int, F:Float)
   166	    => (floorFloat(F) ==Float ceilFloat(F)) andBool (Float2Int(floorFloat(F)) ==Int I)
   167	       requires floatFinite(F) [concrete]
   168	  rule eqIF(I:Int, F:Float) => eqF(intToF(I), F) requires notBool floatFinite(F) [concrete]
   169	
   170	  rule applyCmp("==", I:Int, F:Float) => eqIF(I, F)
   171	  rule applyCmp("==", F:Float, I:Int) => eqIF(I, F)
   172	  rule applyCmp("!=", I:Int, F:Float) => notBool eqIF(I, F)
   173	  rule applyCmp("!=", F:Float, I:Int) => notBool eqIF(I, F)
   174	  rule applyCmp("<",  I:Int, F:Float) => ltIF(I, F)
   175	  rule applyCmp("<",  F:Float, I:Int) => ltFI(F, I)
   176	  rule applyCmp(">",  I:Int, F:Float) => ltFI(F, I)          // I >  F  <=>  F <  I
   177	  rule applyCmp(">",  F:Float, I:Int) => ltIF(I, F)          // F >  I  <=>  I <  F
   178	  rule applyCmp("<=", I:Int, F:Float) => notBool ltFI(F, I)  // I <= F  <=>  not (F < I)
   179	  rule applyCmp("<=", F:Float, I:Int) => notBool ltIF(I, F)  // F <= I  <=>  not (I < F)
   180	  rule applyCmp(">=", I:Int, F:Float) => notBool ltIF(I, F)  // I >= F  <=>  not (I < F)
   181	  rule applyCmp(">=", F:Float, I:Int) => notBool ltFI(F, I)  // F >= I  <=>  not (F < I)

   186	  rule applyCmp(OP:String, B:Bool, F:Float) => applyCmp(OP, boolAsInt(B), F) requires isEqOrdOp(OP)
   187	  rule applyCmp(OP:String, F:Float, B:Bool) => applyCmp(OP, F, boolAsInt(B)) requires isEqOrdOp(OP)

     2	// Bool operator cases + the truthy boolean operations (not / and / or; and-or are
     3	// value-returning short-circuit over any Vals, per Python).
     4	
     5	module MPY-BOOL
     6	  imports MPY-CORE
     7	
     8	  rule applyUn("not", V:Val) => notBool truthy(V)
     9	
    10	  rule applyCmp("==", B1:Bool, B2:Bool) => B1 ==Bool B2
    11	  rule applyCmp("!=", B1:Bool, B2:Bool) => B1 =/=Bool B2
    12	
    13	  // bool is a subtype of int: a Bool compared with an Int promotes to 0/1 and
    14	  // re-dispatches to the Int comparison cases (True==1, False==0, True<2, ...).
    15	  // The == / != Bool x Bool cases above are kept; only the four ordering ops are
    16	  // promoted for Bool x Bool (to avoid duplicating the two rules above).
    17	  rule applyCmp(OP:String, B:Bool, I:Int) => applyCmp(OP, boolAsInt(B), I)
    18	       requires isEqOrdOp(OP)
    19	  rule applyCmp(OP:String, I:Int, B:Bool) => applyCmp(OP, I, boolAsInt(B))
    20	       requires isEqOrdOp(OP)
    21	  rule applyCmp(OP:String, B1:Bool, B2:Bool) => applyCmp(OP, boolAsInt(B1), boolAsInt(B2))
    22	       requires isOrdOp(OP)
    23	
    24	  // ==== BoolOp: short-circuit, value-returning and / or =====================
    25	  // the node is its own accumulator: heat the HEAD element only, then either return it
    26	  // (short-circuit) or drop it and continue
    27	  context BoolOp(_, (HOLE:Expr, _:Exprs))
    28	  rule <k> BoolOp(_:String, (V:Val, .Exprs)) => V ... </k>
    29	  rule <k> BoolOp("and", (V:Val, A:Expr, REST:Exprs)) => BoolOp("and", (A, REST)) ... </k>
    30	       requires truthy(V)
    31	  rule <k> BoolOp("and", (V:Val, _:Expr, _:Exprs)) => V ... </k>
    32	       requires notBool truthy(V)
    33	  rule <k> BoolOp("or",  (V:Val, _:Expr, _:Exprs)) => V ... </k>
    34	       requires truthy(V)
    35	  rule <k> BoolOp("or",  (V:Val, A:Expr, REST:Exprs)) => BoolOp("or", (A, REST)) ... </k>
    36	       requires notBool truthy(V)
    37	
    38	  // ==== heap-object head: decide truthiness THROUGH the heap, never rewrite the
    39	  // operand — and/or return the OBJECT itself (Python identity), not its structure
    40	  rule <k> BoolOp(_:String, (ref(H:Int), .Exprs)) => ref(H) ... </k>
    41	       [priority(40)]
    42	  rule <k> BoolOp("and", (ref(H:Int), A:Expr, REST:Exprs)) => BoolOp("and", (A, REST)) ... </k>
    43	       <heap> ... H |-> V:Val ... </heap>
    44	       requires truthy(V)
    45	       [priority(40)]
    46	  rule <k> BoolOp("and", (ref(H:Int), _:Expr, _:Exprs)) => ref(H) ... </k>
    47	       <heap> ... H |-> V:Val ... </heap>
    48	       requires notBool truthy(V)
    49	       [priority(40)]
    50	  rule <k> BoolOp("or", (ref(H:Int), _:Expr, _:Exprs)) => ref(H) ... </k>
    51	       <heap> ... H |-> V:Val ... </heap>
    52	       requires truthy(V)
    53	       [priority(40)]
    54	  rule <k> BoolOp("or", (ref(H:Int), A:Expr, REST:Exprs)) => BoolOp("or", (A, REST)) ... </k>
    55	       <heap> ... H |-> V:Val ... </heap>
    56	       requires notBool truthy(V)
    57	       [priority(40)]
    58	endmodule
