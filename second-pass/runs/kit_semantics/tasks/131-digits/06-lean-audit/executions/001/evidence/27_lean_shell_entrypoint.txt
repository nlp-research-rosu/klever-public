   136	@[export lean_shell_main]
   137	private def shellMain
   138	    (args : List String)
   139	    (useStdin : Bool := false)
   140	    (onlyDeps : Bool := false)
   141	    (onlySrcDeps : Bool := false)
   142	    (depsJson : Bool := false)
   143	    (opts : Options := {})
   144	    (trustLevel : UInt32 := 0)
   145	    (rootDir? : Option System.FilePath := none)
   146	    (setupFileName? : Option System.FilePath := none)
   147	    (oleanFileName? : Option System.FilePath := none)
   148	    (ileanFileName? : Option System.FilePath := none)
   149	    (cFileName? : Option System.FilePath := none)
   150	    (bcFileName? : Option System.FilePath := none)
   151	    (jsonOutput : Bool := false)
   152	    (errorOnKinds : Array Name := #[])
   153	    (printStats : Bool := false)
   154	    (run : Bool := false)
   155	    : IO UInt32 := do
   156	  if onlyDeps && depsJson then
   157	    let fns ←
   158	      if useStdin then
   159	        (← IO.getStdin).lines
   160	      else
   161	        pure args.toArray
   162	    printImportsJson fns
   163	    return 0
   164	  let (fileName?, args) :=
   165	    match args with
   166	    | fileName :: args => (some fileName, args)
   167	    | [] => (none, args)
   168	  if !run && !args.isEmpty then
   169	    IO.eprintln "Expected exactly one file name"
   170	    displayHelp (useStderr := true)
   171	    return 1
   172	  let fileName ←
   173	    if let some fileName := fileName? then
   174	      pure fileName
   175	    else if useStdin then
   176	      pure "<stdin>"
   177	    else
   178	      IO.eprintln "Expected exactly one file name"
   179	      displayHelp (useStderr := true)
   180	      return 1
   181	  let contents ← decodeLossyUTF8 <$> do
   182	    if useStdin then
   183	      (← IO.getStdin).readBinToEnd
   184	    else
   185	      IO.FS.readBinFile fileName
   186	  if onlyDeps then
   187	    Elab.printImports contents fileName
   188	    return 0
   189	  if onlySrcDeps then
   190	    Elab.printImportSrcs contents fileName
   191	    return 0
   192	  -- Quick and dirty `#lang` support
   193	  ---TODO: make it extensible, and add `lean4md`
   194	  let contents ←
   195	    if contents.startsWith "#lang" then
   196	      let endLinePos := contents.posOf '\n'
   197	      let langId := contents.extract ⟨6⟩ endLinePos |>.trim
   198	      if langId == "lean4" then
   199	        pure () -- do nothing for now
   200	      else
   201	        IO.eprintln s!"unknown language '{langId}'\n";
   202	        return 1
   203	      -- Remove up to `\n`
   204	      pure <| contents.extract endLinePos contents.endPos
   205	    else
   206	      pure contents
   207	  let setup? ← setupFileName?.mapM ModuleSetup.load
   208	  let mainModuleName ←
   209	    if let some setup := setup? then
   210	      pure setup.name
   211	    else if let some fileName := fileName? then
   212	      try moduleNameOfFileName fileName rootDir? catch e =>
   213	        if oleanFileName?.isNone && ileanFileName?.isNone then
   214	          pure `_stdin
   215	        else
   216	          throw e
   217	    else
   218	      pure `_stdin
   219	  let env? ← Elab.runFrontend contents opts fileName mainModuleName trustLevel
   220	      oleanFileName? ileanFileName? jsonOutput errorOnKinds #[] printStats setup?
   221	  if let some env := env? then
   222	    if run then
   223	      return ← runMain env opts args
   224	    if let some c := cFileName? then
   225	      let .ok out ← IO.FS.Handle.mk c .write |>.toBaseIO
   226	        | IO.eprintln s!"failed to create '{c}'"
   227	          return 1
   228	      profileitIO "C code generation" opts do
   229	        let data ← IO.ofExcept <| IR.emitC env mainModuleName
   230	        out.write data.toUTF8

   234	        emitLLVM env mainModuleName bc

   240	    -- This is useful when profiling improvements to Lean startup time.
