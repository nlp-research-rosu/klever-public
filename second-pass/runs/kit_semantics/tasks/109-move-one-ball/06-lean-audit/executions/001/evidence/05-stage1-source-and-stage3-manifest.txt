     1	def move_one_ball(arr):
     2	    if not arr:
     3	        return True
     4	
     5	    first = arr[0]
     6	    previous = first
     7	    current = first
     8	    drops = 0
     9	
    10	    for current in arr:
    11	        drops = drops + (current < previous)
    12	        previous = current
    13	
    14	    drops = drops + (first < previous)
    15	
     16	    return drops <= 1
