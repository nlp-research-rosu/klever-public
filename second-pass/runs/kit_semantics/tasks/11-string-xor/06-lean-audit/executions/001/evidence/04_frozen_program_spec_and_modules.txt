FROZEN source solution.py
     1	def string_xor(a: str, b: str) -> str:
     2	    result = ""
     3	    x = ""
     4	    y = ""
     5	    for x, y in zip(a, b):
     6	        if x == y:
     7	            result += "0"
     8	        else:
     9	            result += "1"
    10	    return result
