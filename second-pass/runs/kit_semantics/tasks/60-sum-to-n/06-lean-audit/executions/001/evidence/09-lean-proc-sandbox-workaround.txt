#define _GNU_SOURCE

#include <errno.h>
#include <limits.h>
#include <sys/types.h>
#include <unistd.h>

/*
 * The audit command sandbox exposes /proc/self but not /proc/<namespace-pid>.
 * Lean 4.22 resolves its executable through /proc/<getpid()>/exe.  Return the
 * proc-visible numeric target of /proc/self so that lookup reaches the same
 * executable.  This shim is loaded only for the independent preflight build.
 */
pid_t getpid(void) {
    char target[64];
    ssize_t count = readlink("/proc/self", target, sizeof(target) - 1);
    long value = 0;

    if (count <= 0 || count >= (ssize_t)sizeof(target)) {
        return (pid_t)1;
    }
    target[count] = '\0';
    for (ssize_t index = 0; index < count; ++index) {
        if (target[index] < '0' || target[index] > '9') {
            errno = EINVAL;
            return (pid_t)1;
        }
        value = value * 10 + (target[index] - '0');
        if (value > INT_MAX) {
            errno = ERANGE;
            return (pid_t)1;
        }
    }
    return (pid_t)value;
}
