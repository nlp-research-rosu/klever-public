def _mask_non_code(text: str) -> str:
    """Mask comments and string bodies without changing source offsets."""

    output = list(text)
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                state = "code"
            else:
                output[index] = " "
            index += 1
            continue
        if state == "string":
            if character == "\\" and following:
                output[index] = " "
                output[index + 1] = " "
                index += 2
                continue
            if character == '"':
                state = "code"
            elif character not in "\r\n":
                output[index] = " "
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                output[index] = output[index + 1] = " "
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                output[index] = output[index + 1] = " "
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character not in "\r\n":
                output[index] = " "
            index += 1
            continue
        if character == "/" and following == "/":
            output[index] = output[index + 1] = " "
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output[index] = output[index + 1] = " "
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        if character == '"':
            state = "string"
        index += 1
    return "".join(output)


def _line_ranges(text: str, start: int = 0) -> list[tuple[int, int, int]]:
    ranges: list[tuple[int, int, int]] = []
    line_start = start
    while line_start < len(text):
        newline = text.find("\n", line_start)
        next_start = len(text) if newline < 0 else newline + 1
        line_end = len(text) if newline < 0 else newline
        content_end = (
            line_end - 1
            if line_end > line_start and text[line_end - 1] == "\r"
            else line_end
        )
        ranges.append((line_start, content_end, next_start))
        if newline < 0:
            break
        line_start = next_start
    return ranges


def _line_heads(text: str) -> list[tuple[int, int, str, str]]:
    """Return physical-line word heads without assigning K grammar meaning."""

    heads: list[tuple[int, int, str, str]] = []
    for line_start, content_end, _next_start in _line_ranges(text):
        index = line_start
        while index < content_end and text[index] in " \t":
            index += 1
        word_end = index
        while word_end < content_end and (
            text[word_end].isalnum() or text[word_end] in "_'"
        ):
            word_end += 1
        if word_end > index:
            heads.append(
                (
                    line_start,
                    content_end,
                    text[index:word_end],
                    text[word_end:content_end],
                )
            )
    return heads


def _line_word(line: str) -> tuple[str, str, int]:
    stripped = line.lstrip()
    indent = len(line) - len(stripped)
    word_end = 0
    while word_end < len(stripped) and (
        stripped[word_end].isalnum() or stripped[word_end] in "_'"
    ):
        word_end += 1
    return stripped[:word_end], stripped[word_end:], indent + word_end


def _rule_continuation(line: str) -> tuple[str, int] | None:
    stripped = line.lstrip()
    if stripped.startswith("["):
        return "attributes", len(line) - len(stripped)
    word, rest, word_end = _line_word(line)
    for keyword in ("requires", "ensures"):
        if word == keyword and (
            not rest
            or not (
                rest[0].isalnum()
                or rest[0] in "_'"
            )
        ):
            return keyword, word_end
    return None


def _starts_outer_transition(line: str) -> bool:
    keyword, rest, _word_end = _line_word(line)
    separated = bool(rest) and rest[0] in " \t"
    if keyword == "module":
        return separated and _MODULE_NAME_RE.fullmatch(rest) is not None
    if keyword == "endmodule":
        return not rest.strip()
    if keyword == "rule":
        return not rest or separated
    return keyword in _OUTER_SENTENCE_KEYWORDS and separated


def _terminal_attribute_line(line: str) -> bool:
    match = _TRAILING_ATTRIBUTES_RE.search(line)
    if match is None:
        return False
    start = match.start()
    return start == 0 or line[start - 1].isspace()


def _scan_rule_delimiters(
    line: str,
    delimiters: list[str],
) -> None:
    closing = {")": "(", "]": "[", "}": "{"}
    index = 0
    while index < len(line):
        if line[index] == "<":
            cell = _CELL_TAG_RE.match(line, index)
            if cell is not None:
                marker = f"<{cell.group('name')}>"
                if cell.group("close"):
                    if delimiters and delimiters[-1] == marker:
                        delimiters.pop()
                elif not cell.group("self_close"):
                    delimiters.append(marker)
                index = cell.end()
                continue
        character = line[index]
        if character in "([{":
            delimiters.append(character)
        elif character in closing:
            if delimiters and delimiters[-1] == closing[character]:
                delimiters.pop()
        index += 1


def _rule_bubble_end(masked_text: str, start: int) -> int:
    """Find when an outer rule bubble returns control to outer parsing."""

    seen_rewrite = False
    clause_has_content = False
    candidate_end: int | None = None
    delimiters: list[str] = []

    for line_start, line_end, next_start in ranges:
        line = masked_text[line_start:line_end]

            continue

            _rule_continuation(line)

            else None
        )

            if continuation is None and _starts_outer_transition(line):
                return candidate_end
            candidate_end = None

        if continuation is not None:
            kind, content_start = continuation
            if kind in {"requires", "ensures"}:
                clause_has_content = False

        if not seen_rewrite:
            rewrite = line.find("=>")
            if rewrite >= 0:
                seen_rewrite = True
                content_start = rewrite + 2
            else:
                content_start = len(line)

        if seen_rewrite and clause_has_content and not delimiters:

        return candidate_end
    return len(masked_text) + 1

def _preflight_snapshot(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
) -> PreflightSnapshot:
    generated = generation / "generated"
    try:
        sidecars = tuple(
            (
                name,
                hashlib.sha256(
                    pipeline_path.read_bytes()
                ).hexdigest(),
            )
            for name in _SIDECAR_NAMES

            and not pipeline_path.is_symlink()
        )
        if len(sidecars) != len(_SIDECAR_NAMES):
            raise KleanPreflightError(

            )
        return PreflightSnapshot(
            stage1_workspace_sha256=klean_export.tree_digest(
                frozen_input
            ),
            stage3_discovery_manifest_sha256=hashlib.sha256(
                discovery_manifest.read_bytes()
            ).hexdigest(),
            generated_tree_sha256=klean_export.tree_digest(generated),
            sidecar_sha256=sidecars,
        )
    except (OSError, klean_export.KleanExportError) as error:
        raise KleanPreflightError(
            "cannot snapshot immutable preflight inputs"
        ) from error

def _reject_proposition_trust(
    generated: Path,
    sources: list[Path],
    declarations: dict[str, tuple[str, str]],
) -> None:
    """Apply a policy that is independent of the generator's allowlist.

    Generated executable constants may remain opaque at the Klean boundary,
    but generated propositions/proofs may never be assumed.  In particular,
    the immutable target module must contain only a proposition definition,
    never an axiom or opaque proof.
    """

    for source in sources:
        relative = source.relative_to(generated).as_posix()
        if source.name == "Lemmas.lean" and re.search(
            r"(?m)^\s*(?:axiom|opaque)\s+", source.read_text()
        ):
            raise KleanPreflightError(
                f"proposition trust is forbidden in {relative}"
            )
    for name, (_kind, lean_type) in declarations.items():
        normalized = " ".join(lean_type.split())
        proposition_like = (
            normalized in {"Prop", "True", "False"}
            or " Prop" in f" {normalized}"
            or re.search(r":\s*Prop(?:\s|$)", normalized) is not None
            or normalized.startswith(("∀ ", "∃ "))
            or re.search(r"(^|[ (])(?:True|False)(?:$|[ )])", normalized)
            is not None
            or re.search(r"(?:^| )[^ ]+\s*=\s*[^ ]+", normalized) is not None
        )
        if proposition_like:
            raise KleanPreflightError(
                f"proposition trust is forbidden for {name}: {lean_type}"
            )

def validate_trust_parameter_links(
    expected_ids: list[str], parameters: list[object]
) -> set[str]:
    """Validate every opaque parameter without requiring one per theorem."""

    bound_rule_ids: set[str] = set()
    for parameter in parameters:
        if not isinstance(parameter, dict):
            raise KleanPreflightError("generated trust parameter is malformed")
        source_rule_ids = parameter.get("source_rule_ids")
        if (
            not isinstance(source_rule_ids, list)
            or not source_rule_ids
            or not all(
                isinstance(source_rule_id, str)
                and source_rule_id in expected_ids
                for source_rule_id in source_rule_ids
            )
        ):
            raise KleanPreflightError(
                "trust parameter is not bound to source obligations"
            )
        bound_rule_ids.update(source_rule_ids)
    return bound_rule_ids

class LemmaDiscoveryContractError(RuntimeError):
    """Raised when a Stage 3 manifest crosses its trust boundary."""


def _regular_json(path: Path) -> dict[str, Any]:
    path = Path(path)
    try:
        mode = path.stat(follow_symlinks=False).st_mode
    except OSError as error:
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest must be a regular file: {path}"
        ) from error
    if not stat.S_ISREG(mode):
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest must be a regular file: {path}"
        )
    try:
        document = json.loads(path.read_bytes())
    except (OSError, UnicodeError, ValueError) as error:
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest is malformed: {path}"
        ) from error
    if not isinstance(document, dict):
        raise LemmaDiscoveryContractError(
            "lemma discovery manifest must be a JSON object"
        )
    return document


def _exact_keys(
    document: dict[str, Any], expected: set[str], label: str
) -> None:
    unexpected = sorted(set(document) - expected)
    if unexpected:
        raise LemmaDiscoveryContractError(
            f"{label} has unexpected key: {unexpected[0]}"
        )
    missing = sorted(expected - set(document))
    if missing:
        raise LemmaDiscoveryContractError(
            f"{label} is missing key: {missing[0]}"
        )
