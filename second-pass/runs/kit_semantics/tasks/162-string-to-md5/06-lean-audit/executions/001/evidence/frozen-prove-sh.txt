     1	#!/usr/bin/env bash
     2	set -euo pipefail
     3	
     4	python3 py2mpy.py solution.py > solution.mpy
     5	python3 py2mpy.py smoke-empty.py > smoke-empty.mpy
     6	python3 py2mpy.py smoke.py > smoke.mpy
     7	
     8	kompile --backend llvm reference-semantics/semantics.k \
     9	  --main-module MPY-KRUN \
    10	  --syntax-module MPY-SYNTAX \
    11	  --output-definition runtime-kompiled
    12	
    13	krun smoke-empty.mpy --definition runtime-kompiled
    14	
    15	if krun smoke.mpy --definition runtime-kompiled; then
    16	  echo "UNEXPECTED: opaque nonempty MD5 execution succeeded under LLVM"
    17	  exit 1
    18	else
    19	  llvm_probe_rc=$?
    20	  echo "EXPECTED_OPAQUE_LLVM_FAILURE exit=$llvm_probe_rc"
    21	fi
    22	
    23	kompile --backend haskell verification.k \
    24	  --main-module VERIFICATION \
    25	  --syntax-module MPY-SYNTAX \
    26	  --output-definition verification-kompiled
    27	
    28	krun smoke.mpy --definition verification-kompiled
    29	
    30	kprove spec.k \
    31	  --definition verification-kompiled \
    32	  --spec-module SPEC \
    33	  --claims SPEC.empty-input
    34	
    35	kprove spec.k \
    36	  --definition verification-kompiled \
    37	  --spec-module SPEC \
    38	  --claims SPEC.nonempty-input
    39	
    40	kprove spec.k \
    41	  --definition verification-kompiled \
    42	  --spec-module SPEC
    43	
    44	if kprove spec-vacuity.k \
    45	  --definition verification-kompiled \
    46	  --spec-module SPEC-VACUITY
    47	then
    48	  echo "UNEXPECTED: false postcondition was proved"
    49	  exit 1
    50	else
    51	  vacuity_probe_rc=$?
    52	  echo "EXPECTED_VACUITY_FAILURE exit=$vacuity_probe_rc"
    53	fi
    54	
    55	if kprove spec-body-mutation.k \
    56	  --definition verification-kompiled \
    57	  --spec-module SPEC-BODY-MUTATION
    58	then
    59	  echo "UNEXPECTED: mutated function body satisfied the original result"
    60	  exit 1
    61	else
    62	  body_probe_rc=$?
    63	  echo "EXPECTED_BODY_MUTATION_FAILURE exit=$body_probe_rc"
    64	fi
    65	
    66	python3 test_solution.py
