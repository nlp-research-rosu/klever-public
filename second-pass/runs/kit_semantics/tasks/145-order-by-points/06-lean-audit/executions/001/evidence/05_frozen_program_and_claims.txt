SOLUTION_PY
     1	def digit_sum(number):
     2	    if number < 0:
     3	        sign = -1
     4	        number = -number
     5	    else:
     6	        sign = 1
     7	    total = 0
     8	    while number >= 10:
     9	        total += number % 10
    10	        number //= 10
    11	    return total + sign * number
    12	
    13	
    14	def order_by_points(nums):
    15	    return sorted(nums, key=digit_sum)
