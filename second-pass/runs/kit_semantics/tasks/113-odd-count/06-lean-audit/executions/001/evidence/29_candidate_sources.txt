   240	private def md5RoundChoice (index b c d : Nat) : Nat × Nat :=
   241	  if index < 16 then
   242	    (Nat.lor (Nat.land b c) (Nat.land (md5Not b) d), index)
   243	  else if index < 32 then
   244	    (Nat.lor (Nat.land d b) (Nat.land (md5Not d) c),
   245	      (5 * index + 1) % 16)
   246	  else if index < 48 then
   247	    (Nat.xor (Nat.xor b c) d, (3 * index + 5) % 16)
   248	  else
   249	    (Nat.xor c (Nat.lor b (md5Not d)), (7 * index) % 16)

   251	private def runMd5Rounds :
   252	    Nat → Nat → List Nat → Nat → Nat → Nat → Nat →
   253	      Nat × Nat × Nat × Nat
   254	  | 0, _, _, a, b, c, d => (a, b, c, d)
   255	  | fuel + 1, index, block, a, b, c, d =>
   256	      let choice := md5RoundChoice index b c d
   257	      let rotated := md5RotateLeft
   258	        (md5Word
   259	          (a + choice.1 + md5Constants.getD index 0 +
   260	            md5BlockWord block choice.2))
   261	        (md5Shifts.getD index 0)
   262	      runMd5Rounds fuel (index + 1) block d
   263	        (md5Word (b + rotated)) b c

   343	        makeStringValue (lowerMappedSequence codes)

   345	  else if method = "upper" then
