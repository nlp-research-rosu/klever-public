#define _GNU_SOURCE

#include <dlfcn.h>
#include <errno.h>
#include <stddef.h>
#include <string.h>
#include <unistd.h>

/*
 * The Codex audit sandbox uses a private PID namespace with the host /proc
 * mount. Lean 4.22 asks readlink("/proc/<inner-pid>/exe"), which therefore
 * fails with ENOENT. Preserve normal readlink behavior and retry only that
 * failed executable-path lookup through the kernel's /proc/self alias.
 */
ssize_t readlink(const char *path, char *buffer, size_t size) {
    static ssize_t (*real_readlink)(const char *, char *, size_t) = NULL;
    if (real_readlink == NULL) {
        real_readlink = dlsym(RTLD_NEXT, "readlink");
        if (real_readlink == NULL) {
            errno = ENOSYS;
            return -1;
        }
    }

    ssize_t result = real_readlink(path, buffer, size);
    if (result >= 0 || errno != ENOENT || path == NULL) {
        return result;
    }

    const char prefix[] = "/proc/";
    const char suffix[] = "/exe";
    size_t length = strlen(path);
    if (length > sizeof(prefix) - 1 + sizeof(suffix) - 1 &&
        strncmp(path, prefix, sizeof(prefix) - 1) == 0 &&
        strcmp(path + length - (sizeof(suffix) - 1), suffix) == 0) {
        return real_readlink("/proc/self/exe", buffer, size);
    }
    return result;
}
