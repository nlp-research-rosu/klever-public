import Proof

namespace BridgeAudit

open Proof

def intValues : List Int → SortValSeq
  | [] => SortValSeq.«.ValSeq_MPY-CORE_ValSeq»
  | value :: rest =>
      SortValSeq.«vCons(_,_)_MPY-CORE_ValSeq_Val_ValSeq»
        (SortVal.inj_SortInt value) (intValues rest)

def mixedValues : SortValSeq :=
  SortValSeq.«vCons(_,_)_MPY-CORE_ValSeq_Val_ValSeq»
    (SortVal.inj_SortBool true) (intValues [1, 2])

-- Empty and singleton collection constructors have their K meanings.
example : Proof.«.List».coll = [] := rfl
example : Proof.«.Map».coll = [] := rfl
example (key value : SortKItem) : (Proof.«_|->_» key value).coll = [(key, value)] := rfl
example (item : SortKItem) : (Proof.ListItem item).coll = [item] := rfl

-- Arithmetic and Boolean hooks reject identity/constant counterfactuals.
example : Proof.«_-Int_» 7 3 = 4 := by decide
example : Proof.«_-Int_» 7 3 ≠ 7 := by decide
example : Proof.«_+Int_» (-4) 9 = 5 := by decide
example : Proof._andBool_ true false = false := rfl
example : Proof.«_<Int_» 4 5 = true := rfl
example : Proof.«_<=Int_» 5 4 = false := rfl

-- Sequence recognizer, length, indexing, and half-length edge cases.
example : Proof.«allInts(_)_VERIFICATION-BASE_Bool_ValSeq»
    (intValues [-3, 0, 8]) = true := rfl
example : Proof.«allInts(_)_VERIFICATION-BASE_Bool_ValSeq» mixedValues = false := rfl
example : Proof.«vsLen(_)_MPY-CORE_Int_ValSeq» (intValues [4, 5, 6]) = 3 := rfl
example : Proof.«valSeqAt(_,_)_MPY-SUBSCRIPT_Val_ValSeq_Int»
    (intValues [4, 5, 6]) 1 = SortVal.inj_SortInt 5 := rfl
example : Proof.«valSeqAt(_,_)_MPY-SUBSCRIPT_Val_ValSeq_Int»
    (intValues [4, 5, 6]) 0 ≠ SortVal.inj_SortInt 0 := by
  simp [Proof.«valSeqAt(_,_)_MPY-SUBSCRIPT_Val_ValSeq_Int»,
    Proof.valSeqAtModel, intValues]
example : Proof.«halfLen(_)_VERIFICATION-BASE_Int_ValSeq» (intValues []) = 0 := by decide
example : Proof.«halfLen(_)_VERIFICATION-BASE_Int_ValSeq» (intValues [1]) = 0 := by decide
example : Proof.«halfLen(_)_VERIFICATION-BASE_Int_ValSeq» (intValues [1, 2]) = 1 := by decide
example : Proof.«halfLen(_)_VERIFICATION-BASE_Int_ValSeq» (intValues [1, 2, 3, 4, 5]) = 2 := by decide
example : Proof.«halfLen(_)_VERIFICATION-BASE_Int_ValSeq» (intValues [1, 2, 3, 4, 5]) ≠ 5 := by decide

-- The comparison bridge is defined for integer operands and preserves !=.
example : Proof.«applyCmp(_,_,_)_MPY-CORE_Bool_String_Val_Val?» "!="
    (SortVal.inj_SortInt 2) (SortVal.inj_SortInt 9) = some true := rfl
example : Proof.«applyCmp(_,_,_)_MPY-CORE_Bool_String_Val_Val?» "!="
    (SortVal.inj_SortInt 2) (SortVal.inj_SortInt 2) = some false := rfl

-- The map bridge handles a disjoint singleton union and is not constant-empty.
example (key value : SortKItem) :
    Proof._Map_ (Proof.«_|->_» key value) Proof.«.Map» =
      Proof.«_|->_» key value := by
  simp [Proof._Map_, Proof.«_|->_», Proof.«.Map»,
    Proof.mapDisjointModel, Proof.mapInsertModel]

-- Concrete mismatch counts reject the tempting constant-zero implementation.
example : Proof.«mismatchCount(_,_,_)_VERIFICATION-BASE_Int_ValSeq_Int_Int»
    (intValues [1, 2, 3, 1]) 0 2 = 1 := by
  rfl
example : Proof.«mismatchCount(_,_,_)_VERIFICATION-BASE_Int_ValSeq_Int_Int»
    (intValues [1, 2, 3, 4]) 0 2 = 2 := by
  rfl

end BridgeAudit
