     6	private noncomputable def kleanMapLookupModel
     7	    (entries : List (SortKItem × SortKItem))
     8	    (key : SortKItem) : Option SortKItem :=
     9	  match entries with
    10	  | [] => none
    11	  | (candidate, value) :: rest =>
    12	      if candidate = key then some value
    13	      else kleanMapLookupModel rest key
    14	
    15	private noncomputable def kleanMapContainsModel
    16	    (entries : List (SortKItem × SortKItem))
    17	    (key : SortKItem) : Bool :=
    18	  match entries with
    19	  | [] => false
    20	  | (candidate, _) :: rest =>
    21	      if candidate = key then true
    22	      else kleanMapContainsModel rest key
    23	
    24	private noncomputable def kleanMapDisjointModel
    25	    (left right : List (SortKItem × SortKItem)) : Bool :=
    26	  match right with
    27	  | [] => true
    28	  | (key, _) :: rest =>
    29	      if kleanMapContainsModel left key then false
    30	      else kleanMapDisjointModel left rest
    31	
    32	private noncomputable def kleanMapDeleteModel
    33	    (entries : List (SortKItem × SortKItem))
    34	    (key : SortKItem) : List (SortKItem × SortKItem) :=
    35	  match entries with
    36	  | [] => []
    37	  | (candidate, value) :: rest =>
    38	      if candidate = key then kleanMapDeleteModel rest key
    39	      else (candidate, value) :: kleanMapDeleteModel rest key

    41	private noncomputable def kleanKeyOrderModel : SortKItem → SortKItem → Bool
    42	  | SortKItem.inj_SortInt a,    SortKItem.inj_SortInt b    => decide (a < b)
    43	  | SortKItem.inj_SortInt _,    _                          => true
    44	  | _,                          SortKItem.inj_SortInt _    => false
    45	  | SortKItem.inj_SortString a, SortKItem.inj_SortString b => decide (a < b)
    46	  | SortKItem.inj_SortString _, _                          => true
    47	  | _,                          SortKItem.inj_SortString _ => false
    48	  | _, _ => false

    50	private noncomputable def kleanMapInsertModel
    51	    (key value : SortKItem) :
    52	    List (SortKItem × SortKItem) → List (SortKItem × SortKItem)
    53	  | [] => [(key, value)]
    54	  | (candidate, oldValue) :: rest =>
    55	      if kleanKeyOrderModel candidate key then
    56	        (candidate, oldValue) :: kleanMapInsertModel key value rest
    57	      else (key, value) :: (candidate, oldValue) :: rest

    60	    (entries : List (SortKItem × SortKItem))
    61	    (key value : SortKItem) : List (SortKItem × SortKItem) :=
    62	  kleanMapInsertModel key value (kleanMapDeleteModel entries key)
