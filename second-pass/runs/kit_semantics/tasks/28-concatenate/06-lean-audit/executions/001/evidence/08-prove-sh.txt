     1	#!/usr/bin/env bash
     2	set -eu
     3	
     4	python3 py2mpy.py solution.py > solution.mpy
     5	python3 py2mpy.py concrete_tests.py > concrete_tests.mpy
     6	
     7	kompile --backend llvm reference-semantics/semantics.k \
     8	  --main-module MPY-KRUN \
     9	  --syntax-module MPY-SYNTAX \
    10	  --output-definition runtime-kompiled
    11	
    12	concatenate_krun_output="$(krun concrete_tests.mpy --definition runtime-kompiled)"
    13	printf '%s\n' "$concatenate_krun_output"
    14	printf '%s\n' "$concatenate_krun_output" | grep -F '"empty" |-> str ( .IntSeq )'
    15	printf '%s\n' "$concatenate_krun_output" | grep -F '"example" |-> str ( iCons ( 97 , iCons ( 98 , iCons ( 99 , .IntSeq ) ) ) )'
    16	printf '%s\n' "$concatenate_krun_output" | grep -F '"mixed" |-> str ( iCons ( 120 , iCons ( 121 , iCons ( 122 , .IntSeq ) ) ) )'
    17	
    18	kompile --backend haskell verification.k \
    19	  --main-module VERIFICATION \
    20	  --syntax-module VERIFICATION \
    21	  --output-definition verification-kompiled
    22	
    23	kprove spec.k \
    24	  --definition verification-kompiled \
    25	  --spec-module SPEC
    26	
    27	python3 differential_tests.py
    28	
    29	set +e
    30	kprove spec-vacuity.k \
    31	  --definition verification-kompiled \
    32	  --spec-module SPEC-VACUITY
    33	concatenate_vacuity_status=$?
    34	kprove spec-body-mutation.k \
    35	  --definition verification-kompiled \
    36	  --spec-module SPEC-BODY-MUTATION
    37	concatenate_body_status=$?
    38	set -e
    39	
    40	printf 'VACUITY_EXIT=%s\n' "$concatenate_vacuity_status"
    41	printf 'BODY_MUTATION_EXIT=%s\n' "$concatenate_body_status"
    42	
    43	if [ "$concatenate_vacuity_status" -eq 0 ]; then
    44	  printf '%s\n' 'error: false-result mutation unexpectedly proved' >&2
    45	  exit 1
    46	fi
    47	if [ "$concatenate_body_status" -eq 0 ]; then
    48	  printf '%s\n' 'error: changed-body mutation unexpectedly proved' >&2
    49	  exit 1
    50	fi
