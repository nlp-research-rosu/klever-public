SOLUTION.PY
     1	def can_arrange(arr):
     2	    index = -1
     3	    i = 0
     4	    previous = 0
     5	    value = 0
     6	    for value in arr:
     7	        if i > 0:
     8	            if not value >= previous:
     9	                index = i
    10	        previous = value
    11	        i = i + 1
    12	    return index
