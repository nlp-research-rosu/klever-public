    47	/- Exact binary64 decomposition for MPY-FLOAT's unbounded Int/Float bridge. -/
    48	def operationalFiniteFloatParts
    49	    (value : SortFloat) : SortBool × Nat × SortInt :=
    50	  let bits := value.toBits.toNat
    51	  let negative := decide (bits / (2 ^ 63) = 1)
    52	  let storedExponent := (bits / (2 ^ 52)) % (2 ^ 11)
    53	  let fraction := bits % (2 ^ 52)
    54	  if storedExponent = 0 then
    55	    (negative, fraction, -1074)
    56	  else
    57	    (negative, (2 ^ 52) + fraction, Int.ofNat storedExponent - 1075)

    59	def operationalSplitFloatMagnitude
    60	    (significand : Nat) (binaryExponent : SortInt) : Nat × Nat :=
    61	  if binaryExponent ≥ 0 then
    62	    (significand * (2 ^ binaryExponent.toNat), 0)
    63	  else
    64	    let denominator := 2 ^ (-binaryExponent).toNat
    65	    (significand / denominator, significand % denominator)

    67	def operationalFloorFiniteFloat (value : SortFloat) : SortInt :=
    68	  let (negative, significand, binaryExponent) := operationalFiniteFloatParts value
    69	  let (quotient, remainder) :=
    70	    operationalSplitFloatMagnitude significand binaryExponent
    71	  if negative then
    72	    -Int.ofNat (quotient + if remainder = 0 then 0 else 1)
    73	  else
    74	    Int.ofNat quotient

    82	  else
