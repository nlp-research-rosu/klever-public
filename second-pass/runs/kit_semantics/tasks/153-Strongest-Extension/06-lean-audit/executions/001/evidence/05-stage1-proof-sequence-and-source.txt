#!/usr/bin/env bash
set -eu

expect_failure() {
  probe_name=$1
  shift
  if "$@"; then
    echo "UNEXPECTED SUCCESS: ${probe_name}" >&2
    exit 1
  else
    probe_status=$?
    echo "EXPECTED FAILURE: ${probe_name} (exit ${probe_status})"
  fi
}

  --syntax-module MPY-SYNTAX \
  --output-definition runtime-kompiled
krun concrete_tests.mpy --definition runtime-kompiled

# Bridge-free projection, yield, and inner-loop connection proofs.

  --main-module VERIFICATION-BASE \
  --syntax-module MPY-SYNTAX \
  --output-definition connection-kompiled
kprove connection-spec.k \
  --definition connection-kompiled \
  --spec-module CONNECTION-SPEC

# Outer-loop connection proof, using only the already-connected inner bridge.
kompile --backend haskell verification.k \
  --main-module VERIFICATION \
  --syntax-module MPY-SYNTAX \
  --output-definition outer-connection-kompiled
kprove outer-connection-spec.k \

requires "verification.k"

module CONNECTION-SPEC
  imports VERIFICATION-BASE

  // Datatype value-fidelity theorem for the guarded total projection.
  claim [projection-identity]:
    <k> V:Val => projectStrTotal(V) </k>
    requires isStringVal(V)

  // Universal fixed-semantics connection for the static-sort yield twin.
  claim [yield-connection]:
    <k>
      #iterYield(V:Val, list(VS))
      ~> #loopStep(Name("extension"), OUTER-BODY)
      => #bindTgt(Name("extension"), projectStrTotal(V))
         ~> OUTER-BODY
         ~> #loopLbl(
              #loop(
                list(VS),
                Name("extension"),
                OUTER-BODY
              )
            )
      ...
    </k>
    requires isStringVal(V)

  // Universal fixed-semantics connection for the complete inner-loop bridge.
  claim [inner-loop]:
    <k>
      #loop(STRING:Str, Name("character"), INNER-BODY)
      => .K
      ...
    </k>
    <env> L </env>
    <scopes>
      ...
      L |-> scope(
        "class_name" |-> str(CLASS)
        "extensions" |-> list(ALLEXTS)
        "strongest" |-> str(BEST)
        "best_strength" |-> BESTVAL:Val
        "extension" |-> str(EXT)
        "strength" |-> (
          S:Int => S +Int extensionStrength(codesOf(STRING))
        )
        "character" |-> (
          str(OLDCHAR)
          => str(lastCharacter(codesOf(STRING), OLDCHAR))
        ),
        parent(0)
      )
      ...
    </scopes>
endmodule
requires "verification.k"

def Strongest_Extension(class_name, extensions):
    strongest = ""
    best_strength = None
    extension = ""
    character = ""

    for extension in extensions:
        strength = 0

        for character in extension:
            if character.isupper():
                strength += 1
            elif character.islower():
                strength -= 1

        if best_strength is None or strength > best_strength:
            strongest = extension
            best_strength = strength

    return class_name + "." + strongest
Module(
  FuncDef("Strongest_Extension", Params("class_name", "extensions"),
    Assign(Name("strongest"), Str(""))
    Assign(Name("best_strength"), NoneVal)
    Assign(Name("extension"), Str(""))
    Assign(Name("character"), Str(""))
    For(Name("extension"), Name("extensions"),
      Assign(Name("strength"), Int(0))
      For(Name("character"), Name("extension"),
        If(Call(Attribute(Name("character"), "isupper"), ),
          AugAssign(Name("strength"), "+", Int(1)),
          If(Call(Attribute(Name("character"), "islower"), ),
            AugAssign(Name("strength"), "-", Int(1)),
            )))
      If(BoolOp(
        "or",
        Compare(Name("best_strength"), CmpOp("is", NoneVal)),
        Compare(Name("strength"), CmpOp(">", Name("best_strength")))),
        Assign(Name("strongest"), Name("extension"))
        Assign(Name("best_strength"), Name("strength")),
        ))
    Return(
      BinOp(
        "+",
        BinOp("+", Name("class_name"), Str(".")),
        Name("strongest")))))
