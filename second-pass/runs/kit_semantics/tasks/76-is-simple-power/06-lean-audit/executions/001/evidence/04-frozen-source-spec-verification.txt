     1	def is_simple_power(x, n):
     2	    if x == 1:
     3	        return True
     4	    if n == 0:
     5	        return x == 0
     6	    if n == 1:
     7	        return False
     8	    if n == -1:
     9	        return x == -1
    10	    if x == 0:
    11	        return False
    12	    while x % n == 0:
    13	        x = x // n
    14	    return x == 1
