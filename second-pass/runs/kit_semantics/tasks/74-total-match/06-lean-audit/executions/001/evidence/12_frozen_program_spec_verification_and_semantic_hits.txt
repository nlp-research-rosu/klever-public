     3	    '''
     4	    Write a function that accepts two lists of strings and returns the list that has 
     5	    total number of chars in the all strings of the list less than the other list.
     6	
     7	    if the two lists have the same number of chars, return the first list.
     8	
     9	    Examples
    10	    total_match([], []) ➞ []
    11	    total_match(['hi', 'admin'], ['hI', 'Hi']) ➞ ['hI', 'Hi']
    12	    total_match(['hi', 'admin'], ['hi', 'hi', 'admin', 'project']) ➞ ['hi', 'admin']
    13	    total_match(['hi', 'admin'], ['hI', 'hi', 'hi']) ➞ ['hI', 'hi', 'hi']
    14	    total_match(['4'], ['1', '2', '3', '4', '5']) ➞ ['4']
    15	    '''
FILE: /reference/k-proof/solution.py
     1	def _total_length(strings):
     2	    total = 0
     3	    string = None
     4	    for string in strings:
     5	        total += len(string)
     6	    return total
     7	
     8	
     9	def total_match(lst1, lst2):
    10	    if _total_length(lst1) <= _total_length(lst2):
    11	        return lst1
    12	    return lst2

    14	  // definition exhaustive but cannot affect a claim in the formal domain.
    15	  syntax IntSeq ::= stringCodes(Val) [function, total]
    16	  rule stringCodes(str(CS:IntSeq)) => CS
    17	  rule stringCodes(_V:Val) => .IntSeq [owise]
    18

    20	  // the exact fixed-semantics string recognizer.
    21	  rule seqLen(V:Val) => isLen(stringCodes(V))
    22	    requires isStrV(V)

    24	
    25	  // A left-fold summary that exactly follows the program's accumulator update.
    26	  syntax Int ::= totalLen(ValSeq) [function, total]
