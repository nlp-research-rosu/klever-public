     3	    '''
     4	    Write a function that accepts two lists of strings and returns the list that has 
     5	    total number of chars in the all strings of the list less than the other list.
     6	
     7	    if the two lists have the same number of chars, return the first list.
     8	
     9	    Examples
    10	    total_match([], []) ➞ []
    11	    total_match(['hi', 'admin'], ['hI', 'Hi']) ➞ ['hI', 'Hi']
    12	    total_match(['hi', 'admin'], ['hi', 'hi', 'admin', 'project']) ➞ ['hi', 'admin']
    13	    total_match(['hi', 'admin'], ['hI', 'hi', 'hi']) ➞ ['hI', 'hi', 'hi']
    14	    total_match(['4'], ['1', '2', '3', '4', '5']) ➞ ['4']
    15	    '''
FILE: /reference/k-proof/solution.py
     1	def _total_length(strings):
     2	    total = 0
     3	    string = None
     4	    for string in strings:
     5	        total += len(string)
     6	    return total
     7	
     8	
     9	def total_match(lst1, lst2):
    10	    if _total_length(lst1) <= _total_length(lst2):
    11	        return lst1
    12	    return lst2

     1	Module(
     2	  FuncDef("_total_length", Params("strings"),
     3	    Assign(Name("total"), Int(0))
     4	    Assign(Name("string"), NoneVal)
     5	    For(Name("string"), Name("strings"),
     6	      AugAssign(Name("total"), "+", Call(Name("len"), Name("string"))))
     7	    Return(Name("total")))
     8	  FuncDef("total_match", Params("lst1", "lst2"),
     9	    If(Compare(
    10	      Call(Name("_total_length"), Name("lst1")),
    11	      CmpOp("<=", Call(Name("_total_length"), Name("lst2")))),
    12	      Return(Name("lst1")),
    13	      )
    14	    Return(Name("lst2"))))

     6	  // The prompt's list-of-strings domain, unfolded one ValSeq constructor at a
     7	  // time.  isStrV is supplied by the fixed semantics.
     8	  syntax Bool ::= onlyStrings(ValSeq) [function, total]
     9	  rule onlyStrings(.ValSeq) => true
    10	  rule onlyStrings(vCons(V:Val, REST:ValSeq))
    11	    => isStrV(V) andBool onlyStrings(REST)
    12	
    13	  // A total projection used only under isStrV.  Its non-string case makes the
    14	  // definition exhaustive but cannot affect a claim in the formal domain.
    15	  syntax IntSeq ::= stringCodes(Val) [function, total]
    16	  rule stringCodes(str(CS:IntSeq)) => CS
    17	  rule stringCodes(_V:Val) => .IntSeq [owise]
    18	
    19	  // This is the reference rule seqLen(str(CS)) => isLen(CS), generalized under
    20	  // the exact fixed-semantics string recognizer.
    21	  rule seqLen(V:Val) => isLen(stringCodes(V))
    22	    requires isStrV(V)
    23	    [simplification]
    24	
    25	  // A left-fold summary that exactly follows the program's accumulator update.
    26	  syntax Int ::= totalLen(ValSeq) [function, total]
    27	               | totalLenFrom(Int, ValSeq) [function, total]
    28	  rule totalLen(ITEMS:ValSeq) => totalLenFrom(0, ITEMS)
    29	  rule totalLenFrom(ACC:Int, .ValSeq) => ACC
    30	  rule totalLenFrom(ACC:Int, vCons(V:Val, REST:ValSeq))
    31	    => totalLenFrom(ACC +Int isLen(stringCodes(V)), REST)
    32	
    33	  // The exact final value left in a Python for-loop target.
    34	  syntax Val ::= lastLoopValue(ValSeq, Val) [function, total]
    35	  rule lastLoopValue(.ValSeq, OLD:Val) => OLD
    36	  rule lastLoopValue(vCons(V:Val, REST:ValSeq), _OLD:Val)
