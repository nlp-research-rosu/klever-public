     1	#define _GNU_SOURCE

     3	#include <dlfcn.h>
     4	#include <errno.h>
     5	#include <stddef.h>
     6	#include <string.h>
     7	#include <unistd.h>
     8	
     9	extern char *program_invocation_short_name;
    10	
    11	typedef ssize_t (*readlink_fn)(const char *, char *, size_t);
    12	
    13	ssize_t readlink(const char *path, char *buffer, size_t size) {
    14	    static readlink_fn real_readlink = NULL;
    15	    static const char toolchain_bin[] =
    16	        "/opt/elan/toolchains/leanprover--lean4---v4.22.0/bin/";
    17	
    18	    if (real_readlink == NULL) {
    19	        real_readlink = (readlink_fn)dlsym(RTLD_NEXT, "readlink");
    20	    }
    21	
    22	    if (path != NULL && strcmp(path, "/proc/self/exe") == 0 &&
    23	        program_invocation_short_name != NULL &&
    24	        (strcmp(program_invocation_short_name, "lean") == 0 ||
    25	         strcmp(program_invocation_short_name, "lake") == 0 ||
    26	         strcmp(program_invocation_short_name, "leanc") == 0)) {
    27	        char resolved[512];
    28	        size_t prefix_len = strlen(toolchain_bin);
    29	        size_t name_len = strlen(program_invocation_short_name);
    30	        size_t total = prefix_len + name_len;
    31	
    32	        if (total > sizeof(resolved) || size < total) {
    33	            errno = ENAMETOOLONG;
    34	            return -1;
    35	        }
    36	        memcpy(resolved, toolchain_bin, prefix_len);
    37	        memcpy(resolved + prefix_len, program_invocation_short_name, name_len);
    38	        memcpy(buffer, resolved, total);
    39	        return (ssize_t)total;
    40	    }
    41	
    42	    if (real_readlink == NULL) {
    43	        errno = ENOSYS;
    44	        return -1;
    45	    }
    46	    return real_readlink(path, buffer, size);
    47	}
