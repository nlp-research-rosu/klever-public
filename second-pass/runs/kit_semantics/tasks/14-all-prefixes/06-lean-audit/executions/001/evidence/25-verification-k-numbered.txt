     1	requires "reference-semantics/semantics.k"
     2	
     3	module VERIFICATION
     4	  imports MPY
     5	
     6	  // Starting with a prefix P and an already-built output ACC, consume the
     7	  // remaining character codes and append each successively longer prefix.
     8	  syntax ValSeq ::= prefixesAcc(IntSeq, IntSeq, ValSeq) [function, total]
     9	  rule prefixesAcc(_P:IntSeq, .IntSeq, ACC:ValSeq) => ACC
    10	  rule prefixesAcc(P:IntSeq, iCons(C:Int, R:IntSeq), ACC:ValSeq)
    11	    => prefixesAcc(
    12	         seqConcat(P, iCons(C, .IntSeq)),
    13	         R,
    14	         valSeqConcat(
    15	           ACC,
    16	           vCons(str(seqConcat(P, iCons(C, .IntSeq))), .ValSeq)))
    17	
    18	  // These two summaries describe the function-local variables left by a
    19	  // completed for-loop. They do not replace execution.
    20	  syntax IntSeq ::= finishPrefix(IntSeq, IntSeq) [function, total]
    21	  rule finishPrefix(P:IntSeq, .IntSeq) => P
    22	  rule finishPrefix(P:IntSeq, iCons(C:Int, R:IntSeq))
    23	    => finishPrefix(seqConcat(P, iCons(C, .IntSeq)), R)
    24	
    25	  syntax Val ::= finishChar(Val, IntSeq) [function, total]
    26	  rule finishChar(CH:Val, .IntSeq) => CH
    27	  rule finishChar(_CH:Val, iCons(C:Int, R:IntSeq))
    28	    => finishChar(str(iCons(C, .IntSeq)), R)
    29	endmodule
