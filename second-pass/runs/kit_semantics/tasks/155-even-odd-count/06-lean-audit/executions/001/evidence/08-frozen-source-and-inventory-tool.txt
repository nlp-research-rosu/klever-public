def canonical_json_sha256(document: Any) -> str:
    encoded = json.dumps(
        document,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
    ).encode()
    return hashlib.sha256(encoded).hexdigest()

def inventory_verification(workspace: Path) -> dict[str, Any]:
    workspace = Path(workspace)
    try:
        workspace_mode = workspace.stat(follow_symlinks=False).st_mode
    except OSError as error:
        raise KRuleInventoryError(
            f"workspace must be a real directory: {workspace}"
        ) from error
    if not stat.S_ISDIR(workspace_mode):
        raise KRuleInventoryError(
            f"workspace must be a real directory: {workspace}"
        )
    verification = workspace / "verification.k"
    try:
        verification_mode = verification.stat(
            follow_symlinks=False
        ).st_mode
    except OSError as error:
        raise KRuleInventoryError(
            f"workspace has no regular verification.k: {verification}"
        ) from error
    if not stat.S_ISREG(verification_mode):
        raise KRuleInventoryError(
            f"workspace has no regular verification.k: {verification}"
        )
    try:
        verification_bytes = verification.read_bytes()
        text = verification_bytes.decode()
    except (OSError, UnicodeError) as error:
        raise KRuleInventoryError(
            f"cannot read verification.k: {verification}"
        ) from error
    masked_text = _mask_non_code(text)
    modules = _modules(masked_text)
    module = _verification_module(
        modules, _last_verification_main_module(workspace)
    )
    closure = _verification_closure(modules, module)
    rule_documents = _rule_documents(text, closure)
    return {
        "schema_version": 2,
        "verification_file": "verification.k",
        "verification_sha256": hashlib.sha256(verification_bytes).hexdigest(),
        "verification_module": module.name,
        "verification_modules": [module.name for module in closure],
        "rules": rule_documents,
        "inventory_sha256": canonical_json_sha256(rule_documents),
    }

def even_odd_count(num):
    """Return the counts of even and odd decimal digits in num."""
    num = abs(num)
    even = 0
    odd = 0

    if num == 0:
        return (1, 0)

    while num > 0:
        even = even + 1 - num % 2
        odd = odd + num % 2
        num = num // 10

    return (even, odd)
