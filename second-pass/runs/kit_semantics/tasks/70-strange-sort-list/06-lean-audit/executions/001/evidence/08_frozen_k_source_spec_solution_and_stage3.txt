     1	def strange_sort_list(lst):
     2	    ordered = sorted(lst)
     3	    result = []
     4	    left = 0
     5	    right = len(ordered) - 1
     6	    while left < right:
     7	        result.append(ordered[left])
     8	        result.append(ordered[right])
