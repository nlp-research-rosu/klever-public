    35	/-- Convert an Elan toolchain name to an Elan toolchain directory name. -/
    36	@[inline] partial def toolchain2Dir (toolchain : String) : FilePath :=
    37	  go "" 0
    38	where
    39	  go (acc : String) (pos : String.Pos) : FilePath :=
    40	    if h : toolchain.atEnd pos then
    41	      FilePath.mk acc
    42	    else
    43	      let c := toolchain.get' pos h
    44	      let pos' := toolchain.next' pos h
    45	      if c = '/' then
    46	        go (acc ++ "--") pos'
