   112	/- Decode binary64 exactly for the mixed Int/Float comparisons used by the
   113	   frozen Python semantics.  Finite values are compared as signed mantissas
   114	   over powers of two, so large Int values are never rounded through Float. -/
   115	private structure FloatParts where
   116	  negative : Bool
   117	  exponent : Nat
   118	  fraction : Nat
   119	
   120	private def floatParts (f : SortFloat) : FloatParts :=
   121	  let bits := f.toBits.toNat
   122	  {
   123	    negative := ((bits / (2 ^ 63)) % 2) == 1
   124	    exponent := (bits / (2 ^ 52)) % 2048
   125	    fraction := bits % (2 ^ 52)
   126	  }
   127	
   128	private def pow2Int (n : Nat) : SortInt :=
   129	  Int.ofNat (2 ^ n)
   130	
   131	private def signedMantissa (p : FloatParts) : SortInt :=
   132	  let magnitude :=
   133	    if p.exponent == 0 then p.fraction else (2 ^ 52) + p.fraction
   134	  if p.negative then -(Int.ofNat magnitude) else Int.ofNat magnitude
   135	
   136	private def floatLtIntImpl (f : SortFloat) (i : SortInt) : SortBool :=
   137	  let p := floatParts f
   138	  if p.exponent == 2047 then
   139	    if p.fraction == 0 then p.negative else false
   140	  else if 1075 ≤ p.exponent then
   141	    decide (signedMantissa p * pow2Int (p.exponent - 1075) < i)
   142	  else
   143	    let shift := if p.exponent == 0 then 1074 else 1075 - p.exponent
   144	    decide (signedMantissa p < i * pow2Int shift)
   145	
   146	private def intLtFloatImpl (i : SortInt) (f : SortFloat) : SortBool :=
   147	  let p := floatParts f
   148	  if p.exponent == 2047 then
   149	    if p.fraction == 0 then !p.negative else false
   150	  else if 1075 ≤ p.exponent then
   151	    decide (i < signedMantissa p * pow2Int (p.exponent - 1075))
   152	  else
   153	    let shift := if p.exponent == 0 then 1074 else 1075 - p.exponent
   154	    decide (i * pow2Int shift < signedMantissa p)

   156	private def maxFloatImpl (a b : SortFloat) : SortFloat :=
   157	  if a.isNaN then b
   158	  else if b.isNaN then a
   159	  else if a < b then b
   160	  else if b < a then a
   161	  else if (a == (0.0 : Float)) && (b == (0.0 : Float)) then
   162	    if (floatParts a).negative then b else a
   163	  else a
