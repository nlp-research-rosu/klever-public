     1	def prime_fib(n: int):
     2	    a = 0
     3	    b = 1
     4	    count = 0
     5	    next_value = 0
     6	    divisor = 2
     7	    is_prime = False
     8	    while count < n:
     9	        next_value = a + b
    10	        a = b
    11	        b = next_value
    12	        divisor = 2
    13	        is_prime = a >= 2
    14	        while divisor * divisor <= a:
    15	            if a % divisor == 0:
    16	                is_prime = False
    17	            divisor = divisor + 1
    18	        count = count + is_prime
    19	    return a
