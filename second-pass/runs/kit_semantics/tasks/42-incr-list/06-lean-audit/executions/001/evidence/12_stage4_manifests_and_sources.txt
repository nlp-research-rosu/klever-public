   280	def «Int2Bytes(_,_,_)_BYTES-HOOKED_Bytes_Int_Int_Endianness» (x0 x1 : SortInt) (x2 : SortEndianness) : Option SortBytes :=
   281	  match x2 with
   282	  | .littleEndianBytes => some littleEndian.toByteArray
   283	  | .bigEndianBytes => some littleEndian.reverse.toByteArray
   284	  where
   285	  littleEndian : List UInt8 :=
   286	    let pad := ite (x1 < 0) 0xFF 0x00
   287	    let rec bytes (len : Nat) (n : Int) : List UInt8 :=
   288	      match len, n with
   289	      | 0, _ => []
   290	      | .succ l, 0 => pad :: bytes l 0
   291	      | .succ l, n => ⟨(n % 256).toNat⟩ :: bytes l (n / 256)
   292	    bytes x0.toNat x1

   294	-- Adapted from
   295	-- https://github.com/runtimeverification/haskell-backend/blob/362dab30d6435ec117862fea722be67373572034/kore/src/Kore/Builtin/InternalBytes.hs#L527-L543
   296	-- Note that we use `List.foldl` and not `ByteArray.foldl` for ease of reasoning
   297	def «Bytes2Int(_,_,_)_BYTES-HOOKED_Int_Bytes_Endianness_Signedness» (bytes : SortBytes) (endian : SortEndianness) (sign : SortSignedness) : Option SortInt :=
   298	  match sign with
   299	  | .unsignedBytes => Int.ofNat unsigned
   300	  | .signedBytes => if 2 * unsigned >= modulus then (Int.ofNat unsigned) - (Int.ofNat modulus)
   301	                    else Int.ofNat unsigned
   302	  where
   303	  modulus : Nat := res.1
   304	  unsigned : Nat := res.2
   305	  res : Nat×Nat :=
   306	    let littleEndian := match endian with
   307	                        | .littleEndianBytes => bytes.toList
   308	                            --match bytes with |⟨⟨l⟩⟩ => l
   309	                        | .bigEndianBytes => bytes.toList.reverse
   310	                            --match bytes with |⟨⟨l⟩⟩ => l.reverse
   311	    let go (res : Nat×Nat) (b : UInt8) : Nat×Nat :=
   312	      -- `place` is `res.1`
   313	      -- `acc` is `res.2`
   314	      let place := res.1 * 0x100
   315	      let acc := res.2 + res.1 * b.toNat
   316	      ⟨place, acc⟩
   317	    List.foldl go (1, 0) littleEndian

   340	def «replaceAtBytes(_,_,_)_BYTES-HOOKED_Bytes_Bytes_Int_Bytes» (dest : SortBytes) (index : SortInt) (src : SortBytes) : Option SortBytes :=
   341	  if index < 0 then none else
   342	  if dest.size < index + src.size then none else
   343	  let init := dest.data.extract 0 index.toNat
