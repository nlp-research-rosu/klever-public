   398	def integralFloatToInteger (value : SortFloat) : SortInt :=
   399	  let bits := value.toBits.toNat
   400	  let sign := bits / (2 ^ 63)
   401	  let exponentBits := (bits / (2 ^ 52)) % (2 ^ 11)
   402	  let fraction := bits % (2 ^ 52)
   403	  let magnitude :=
   404	    if exponentBits = 0 then
   405	      0
   406	    else if exponentBits = 2047 then
   407	      0
   408	    else
   409	      let significand := (2 ^ 52) + fraction
   410	      if exponentBits >= 1023 + 52 then
   411	        significand * (2 ^ (exponentBits - 1023 - 52))
   412	      else if exponentBits >= 1023 then
   413	        significand / (2 ^ (52 - (exponentBits - 1023)))
   414	      else
   415	        0
   416	  if sign = 0 then Int.ofNat magnitude else 0 - Int.ofNat magnitude
