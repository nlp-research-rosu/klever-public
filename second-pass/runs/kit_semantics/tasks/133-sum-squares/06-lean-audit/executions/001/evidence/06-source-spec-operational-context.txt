FILE /reference/k-proof/solution.py
     1	import math
     2	
     3	
     4	def sum_squares(lst):
     5	    total = 0
     6	    number = 0
     7	    rounded = 0
     8	    for number in lst:
     9	        rounded = math.ceil(number)
    10	        total += rounded * rounded
    11	    return total
