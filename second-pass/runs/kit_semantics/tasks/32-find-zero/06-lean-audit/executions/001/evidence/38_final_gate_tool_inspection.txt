def _run(
    command: list[str],
    *,
    cwd: Path,
    timeout: int,
) -> tuple[int, str]:
    try:
        result = subprocess.run(
            command,
            cwd=cwd,
            timeout=timeout,
            text=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
        )
        return result.returncode, result.stdout
    except subprocess.TimeoutExpired:
        return 124, f"TIMEOUT after {timeout}s"


def _regular_json(path: Path, label: str) -> dict[str, Any]:
    if not path.is_file() or path.is_symlink():
        raise KleanFinalGateError(f"{label} is not a regular file")
    try:
        document = json.loads(path.read_text())
    except (OSError, ValueError) as error:
        raise KleanFinalGateError(f"{label} is malformed") from error
    if not isinstance(document, dict):
        raise KleanFinalGateError(f"{label} must be a JSON object")
    return document


def _candidate_gate(candidate: Path, target: dict[str, Any]) -> None:
    entries = klean_export._tree_entries(candidate)
    proof = candidate / "Proof.lean"
    if not proof.is_file() or proof.is_symlink():

    for required in ("lakefile.lean", "lean-toolchain"):

        if not path.is_file() or path.is_symlink():

    candidate_sources: list[tuple[str, str]] = []

    parameters = target.get("parameters")

        raise KleanFinalGateError("generator target bindings are malformed")
    proof_text = proof.read_text()
    for parameter in parameters:
        if not isinstance(parameter, dict):
            raise KleanFinalGateError("generator target binding is malformed")
        name = parameter.get("name")

        if not isinstance(name, str) or not isinstance(lean_type, str):
            raise KleanFinalGateError("generator target binding is malformed")
        declarations = [
            (relative, match.start())
            for relative, text in candidate_sources

                rf"(?:noncomputable\s+)?def\s+{re.escape(name)}"
                rf"\s*(?::|\()",
                text,

        if len(declarations) != 1:
            raise KleanFinalGateError(

def _allowed_axioms(inventory: dict[str, Any]) -> set[str]:
    allowlist = inventory.get("allowlist")
    if not isinstance(allowlist, list):
        raise KleanFinalGateError("trust inventory allowlist is malformed")
    names = {"Classical.choice", "propext", "Quot.sound"}
    for entry in allowlist:
        if not isinstance(entry, dict):
            raise KleanFinalGateError("trust inventory entry is malformed")
        name = entry.get("name")
        if not isinstance(name, str) or not name:
            raise KleanFinalGateError("trust inventory name is malformed")
        names.add(name)
    return names

def run_gate(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    candidate: Path | None,
    *,
    toolchain_lock: Path,
    audit_input: Path,
    output: Path,
) -> dict[str, Any]:
    output = Path(output)
    if output.exists() or output.is_symlink():
        raise KleanFinalGateError(f"output already exists: {output}")
    try:
        document = check_final(
            frozen_input,
            discovery_manifest,
            generation,
            candidate,
            toolchain_lock=toolchain_lock,
            audit_input=audit_input,
        )
    except KleanFinalGateError as error:
        document = {
            "status": "FAIL",
            "error": str(error),
            **_result_binding(audit_input),
        }
    except BaseException as error:
        document = {
            "status": "AUDIT_ERROR",
            "error": f"{type(error).__name__}: {error}",
            **_result_binding(audit_input),
        }
    temporary = output.with_name(f".{output.name}.tmp")
    temporary.write_text(json.dumps(document, indent=2, sort_keys=True) + "\n")
    os.replace(temporary, output)
    return document

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--frozen-k", required=True, type=Path)
    parser.add_argument(
        "--discovery-manifest", required=True, type=Path
    )
    parser.add_argument("--generation", required=True, type=Path)
    parser.add_argument("--candidate", type=Path)
    parser.add_argument("--toolchain-lock", required=True, type=Path)
    parser.add_argument("--audit-input", required=True, type=Path)
    parser.add_argument("--output", type=Path)
    arguments = parser.parse_args(argv)
    try:
        if arguments.output is None:
            result = evaluate_gate(
                arguments.frozen_k,
                arguments.discovery_manifest,
                arguments.generation,
                arguments.candidate,
                toolchain_lock=arguments.toolchain_lock,
                audit_input=arguments.audit_input,
            )
        else:
            result = run_gate(
                arguments.frozen_k,
                arguments.discovery_manifest,
                arguments.generation,
                arguments.candidate,
                toolchain_lock=arguments.toolchain_lock,
                audit_input=arguments.audit_input,
                output=arguments.output,
            )
    except KleanFinalGateError as error:
