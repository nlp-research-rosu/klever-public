     1	def poly(xs: list, x: float):
     2	    value = 0.0
     3	    power = 1.0
     4	    coeff = 0.0
     5	    for coeff in xs:
     6	        value = value + coeff * power
     7	        power = power * x
     8	    return value
     9	
    10	
    11	def find_zero(xs: list):
    12	    begin = len(xs) / -len(xs)
    13	    end = len(xs) / len(xs)
    14	    center = 0.0
    15	    while poly(xs, begin) * poly(xs, end) > 0.0:
    16	        begin = begin * 2.0
    17	        end = end * 2.0
    18	    while end - begin > 0.0000000001:
    19	        center = (begin + end) / 2.0
    20	        if poly(xs, center) * poly(xs, begin) > 0.0:
    21	            begin = center
    22	        else:
    23	            end = center
    24	    return begin
