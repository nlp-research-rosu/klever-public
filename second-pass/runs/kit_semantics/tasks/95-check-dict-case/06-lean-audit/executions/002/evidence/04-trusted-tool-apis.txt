def _modules(masked_text: str) -> tuple[_Module, ...]:
    modules: list[_Module] = []
    open_module: tuple[
        str, int, list[str], list[_Sentence]
    ] | None = None
    open_rule_until = 0
    for line_start, line_end, keyword, rest in _line_heads(masked_text):
        if line_start < open_rule_until:
            continue
        open_rule_until = 0
        separated = bool(rest) and rest[0] in " \t"
        if keyword == "module":
            if not separated:
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            named = _MODULE_NAME_RE.fullmatch(rest)
            if named is None or open_module is not None:
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            open_module = (named.group("name"), line_end, [], [])
            continue
        if keyword == "endmodule":
            if rest.strip():
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            if open_module is None:
                raise KRuleInventoryError(
                    "verification.k has malformed module boundaries"
                )
            name, content_start, imports, sentences = open_module
            modules.append(
                _Module(
                    name=name,
                    content_start=content_start,
                    content_end=line_start,
                    imports=tuple(imports),
                    sentences=tuple(sentences),
                )
            )
            open_module = None
            continue
        if open_module is not None and keyword == "rule":
            open_module[3].append(
                _Sentence(keyword=keyword, start=line_start)
            )
            open_rule_until = _rule_bubble_end(masked_text, line_start)
            continue
        if open_module is not None and keyword == "imports" and separated:
            imported = rest.strip().split()
            if len(imported) != 1:
                raise KRuleInventoryError(
                    "verification.k has malformed imports"
                )
            open_module[2].append(imported[0])
        if (
            open_module is not None
            and keyword in _OUTER_SENTENCE_KEYWORDS
            and separated
        ):
            open_module[3].append(
                _Sentence(keyword=keyword, start=line_start)
            )
    if not modules or open_module is not None:
        raise KRuleInventoryError(
            "verification.k has malformed module boundaries"

def _verification_module(
    modules: tuple[_Module, ...], selected_name: str | None = None
) -> _Module:
    if selected_name is not None:
        selected = [module for module in modules if module.name == selected_name]
        if len(selected) != 1:
            raise KRuleInventoryError(
                "prove.sh selects an unknown verification main module: "
                + selected_name
            )
        return selected[0]
    candidates = [
        module
        for module in modules
        if module.name.endswith("VERIFICATION")
        and not module.name.endswith("-SYNTAX")
    ]
    if len(candidates) != 1:
        raise KRuleInventoryError(
            "verification.k must contain exactly one verification module"
        )
    return candidates[0]


def _last_verification_main_module(workspace: Path) -> str | None:
    script = workspace / "prove.sh"

    if not stat.S_ISREG(mode):
        raise KRuleInventoryError(f"prove.sh must be a regular file: {script}")
    try:
        text = script.read_text()
    except (OSError, UnicodeError) as error:
        raise KRuleInventoryError(f"cannot read prove.sh: {script}") from error
    logical = re.sub(r"\\\r?\n", " ", text)
    selected: str | None = None
    for line in logical.splitlines():
        code = line.split("#", 1)[0]
        if not re.search(r"(?:^|[;&|]\s*)kompile\s+", code):
            continue

def _verification_closure(
    modules: tuple[_Module, ...], verification: _Module
) -> tuple[_Module, ...]:
    by_name = {module.name: module for module in modules}
    reached: set[str] = set()
    pending = [verification.name]
    while pending:
        name = pending.pop()
        if name in reached:
            continue
        reached.add(name)
        module = by_name.get(name)
        if module is not None:
            pending.extend(module.imports)
    return tuple(module for module in modules if module.name in reached)


def _without_comments(text: str) -> str:
    """Remove K comments while preserving strings and their exact contents."""

    output: list[str] = []
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                output.append(character)
                state = "code"
            index += 1
            continue
        if state == "string":
            output.append(character)
            if character == "\\" and following:
                output.append(following)
                index += 2
                continue
            if character == '"':
                state = "code"
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character in "\r\n":
                output.append(character)
            index += 1
            continue
        if character == "/" and following == "/":
            output.append(" ")
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output.append(" ")
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        output.append(character)
        if character == '"':
            state = "string"
        index += 1
    return "".join(output)


def _attribute_tokens(text: str, masked_text: str) -> list[str]:
    tokens: list[str] = []
    start = 0
    depth = 0
    for index, character in enumerate(masked_text):
        if character in "([{":
            depth += 1
        elif character in ")]}":
            depth = max(depth - 1, 0)
        elif character == "," and depth == 0:
            token = _without_comments(text[start:index]).strip()
            if token:
                tokens.append(token)
            start = index + 1
    final = _without_comments(text[start:]).strip()
    if final:
        tokens.append(final)
    return tokens


def _rule_spans(text: str, module: _Module) -> list[_RuleSpan]:
    masked_text = _mask_non_code(text)
    sentences = module.sentences
    rules: list[_RuleSpan] = []
    seen_hashes: set[str] = set()
    for index, sentence in enumerate(sentences):
        if sentence.keyword != "rule":
            continue
        end_offset = (
            sentences[index + 1].start
            if index + 1 < len(sentences)
            else module.content_end
        )
        start_offset = sentence.start
        raw_segment = text[start_offset:end_offset]
        masked_segment = masked_text[start_offset:end_offset]
        logical_length = len(masked_segment.rstrip(" \t\r\n"))
        if logical_length == 0:
            continue
        rule_text = raw_segment[:logical_length].rstrip(" \t\r\n")
        masked_rule = _mask_non_code(rule_text)
        attribute_match = _TRAILING_ATTRIBUTES_RE.search(masked_rule)
        attribute_start = None
        attribute_end = None
        attributes: list[str] = []
        if attribute_match is not None:
            attribute_start = attribute_match.start("attributes")
            attribute_end = attribute_match.end("attributes")
            attributes = _attribute_tokens(
                rule_text[attribute_start:attribute_end],
                masked_rule[attribute_start:attribute_end],
            )
        normalized = " ".join(rule_text.split())
        digest = hashlib.sha256(normalized.encode()).hexdigest()
        if digest in seen_hashes:
            raise KRuleInventoryError(
                f"duplicate normalized simplification rule hash: {digest}"
            )
        seen_hashes.add(digest)
        start_line = text.count("\n", 0, start_offset) + 1
        end_line = start_line + rule_text.count("\n")
        rule = KRule(
            source_rule_id=f"rule-{digest}",
            module=module.name,
            start_line=start_line,
            end_line=end_line,
            normalized_sha256=digest,
            attributes=attributes,
            text=rule_text,
        )
        rules.append(
            _RuleSpan(
                rule=rule,
                start_offset=start_offset,
                end_offset=start_offset + len(rule_text),
                attribute_start=(
                    None
                    if attribute_match is None
                    else start_offset + attribute_match.start()
                ),
                attribute_end=(
                    None
                    if attribute_match is None
                    else start_offset + attribute_match.end()
                ),
            )
        )
    return rules


def _rule_documents(
    text: str, modules: tuple[_Module, ...]
) -> list[dict[str, Any]]:
    spans = [span for module in modules for span in _rule_spans(text, module)]
    spans.sort(key=lambda span: span.start_offset)
    return [dataclasses.asdict(span.rule) for span in spans]


def _attribute_token_spans(
    text: str,
    masked_text: str,
) -> list[tuple[int, int, str]]:
    spans: list[tuple[int, int, str]] = []
    start = 0
    depth = 0
    for index, character in enumerate(masked_text):
        if character in "([{":
            depth += 1
        elif character in ")]}":
            depth = max(depth - 1, 0)
        elif character == "," and depth == 0:
            spans.append(
                (
                    start,
                    index,
                    _without_comments(text[start:index]).strip(),
                )
            )
            start = index + 1
    spans.append(
        (
            start,
            len(text),
            _without_comments(text[start:]).strip(),
        )
    )
    return spans


def _blank_non_newlines(text: str) -> str:
    return "".join(
        character if character in "\r\n" else " "
        for character in text
    )


def desimplify_rule_ids(
    text: str,
    source_rule_ids: list[str],
) -> str:
    """Remove simplification from exactly the selected canonical rule IDs."""

    if len(source_rule_ids) != len(set(source_rule_ids)):
        raise KRuleInventoryError(
            "exact rule transformation contains a duplicate source rule ID"
        )
    selected = set(source_rule_ids)
    modules = _modules(_mask_non_code(text))
    spans = [
        span
        for local_module in modules
        for span in _rule_spans(text, local_module)
    ]
    by_id = {span.rule.source_rule_id: span for span in spans}
    unknown = sorted(selected - set(by_id))
    if unknown:
        raise KRuleInventoryError(
            "exact rule transformation names an unknown source rule ID: "
            + ", ".join(unknown)
        )

    edits: list[tuple[int, int]] = []
    for source_rule_id in source_rule_ids:
        span = by_id[source_rule_id]
        if span.attribute_start is None or span.attribute_end is None:
            raise KRuleInventoryError(
                "selected source rule has no simplification attribute: "
                f"{source_rule_id}"
            )
        attribute_text = text[
            span.attribute_start + 1 : span.attribute_end - 1
        ]
        masked_attributes = _mask_non_code(attribute_text)
        tokens = _attribute_token_spans(
            attribute_text,
            masked_attributes,
        )
        matches = [
            index
            for index, (_start, _end, token) in enumerate(tokens)
            if token == "simplification"
        ]
        if len(matches) != 1:
            raise KRuleInventoryError(
                "selected source rule has ambiguous simplification "
                f"attributes: {source_rule_id}"
            )
        token_index = matches[0]
        if len(tokens) == 1:
            edit_start = span.attribute_start
            edit_end = span.attribute_end
        elif token_index < len(tokens) - 1:
            edit_start = (
                span.attribute_start + 1 + tokens[token_index][0]
            )
            edit_end = (
                span.attribute_start + 1 + tokens[token_index + 1][0]
            )
        else:
            edit_start = (
                span.attribute_start + 1 + tokens[token_index - 1][1]
            )
            edit_end = (
                span.attribute_start + 1 + tokens[token_index][1]
            )
        edits.append((edit_start, edit_end))

    transformed = text
    for start, end in sorted(edits, reverse=True):
        transformed = (
            transformed[:start]
            + _blank_non_newlines(transformed[start:end])
            + transformed[end:]
        )
    return transformed

def validate_trust_parameter_links(
    expected_ids: list[str], parameters: list[object]
) -> set[str]:
    """Validate every opaque parameter without requiring one per theorem."""

    bound_rule_ids: set[str] = set()
    for parameter in parameters:
        if not isinstance(parameter, dict):
            raise KleanPreflightError("generated trust parameter is malformed")
        source_rule_ids = parameter.get("source_rule_ids")
        if (
            not isinstance(source_rule_ids, list)
            or not source_rule_ids
            or not all(
                isinstance(source_rule_id, str)
                and source_rule_id in expected_ids
                for source_rule_id in source_rule_ids
            )
        ):
            raise KleanPreflightError(
                "trust parameter is not bound to source obligations"
            )
        bound_rule_ids.update(source_rule_ids)
    return bound_rule_ids


def _check_imports(generated: Path, sources: list[Path]) -> None:
    for source in sources:
        for line in source.read_text().splitlines():
            stripped = line.strip()
            if not stripped.startswith("import "):
                continue
            imported = stripped.removeprefix("import ").strip()
            if (
                not imported
                or imported.startswith(("/", "."))
                or ".." in imported
                or '"' in imported
                or "\\" in imported
            ):
                raise KleanPreflightError(
                    f"untrusted Lean import in {source}: {imported!r}"
                )
    for name in ("lakefile.toml", "lakefile.lean", "lean-toolchain"):
        path = generated / name
        if path.exists() and (not path.is_file() or path.is_symlink()):
            raise KleanPreflightError(
                f"generated dependency metadata is not regular: {path}"
            )
        if path.is_file():
            text = path.read_text()
            if re.search(r"(?m)^\s*(?:path|dir)\s*=\s*\"/", text):
                raise KleanPreflightError(
                    f"absolute dependency path in generated {name}"
                )

def run_preflight(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    *,
    toolchain_lock: Path,
    run_command: Callable[..., tuple[int, str]] = _run,
) -> dict[str, Any]:
    frozen_input = Path(frozen_input)
    discovery_manifest = Path(discovery_manifest)
    generation = Path(generation)
    output = generation / "preflight.json"
    if output.exists() or output.is_symlink():
        raise KleanPreflightError(f"preflight result already exists: {output}")

    def error_document(error: KleanPreflightError) -> dict[str, Any]:
        try:
            stage1_hash = klean_export.tree_digest(frozen_input)
        except (OSError, klean_export.KleanExportError):
            stage1_hash = None
        try:
            stage3_hash = hashlib.sha256(
                discovery_manifest.read_bytes()
            ).hexdigest()
        except OSError:
            stage3_hash = None
        return {
            "schema_version": 3,
            "status": "KLEAN_PREFLIGHT_ERROR",
            "error": str(error),
            "stage1_workspace_sha256": stage1_hash,
            "stage3_discovery_manifest_sha256": stage3_hash,
        }

    publication_snapshot: PreflightSnapshot | None = None
    try:
        publication_snapshot = _preflight_snapshot(
            frozen_input, discovery_manifest, generation
        )
        document = check_generation(
            frozen_input,
            discovery_manifest,
            generation,
            toolchain_lock=toolchain_lock,
            run_command=run_command,
        )
        _require_unchanged_snapshot(
            publication_snapshot,
            frozen_input,
            discovery_manifest,
            generation,
        )
    except KleanPreflightError as error:
        document = error_document(error)
    temporary = output.with_name(f".{output.name}.tmp")
    temporary.write_text(json.dumps(document, indent=2, sort_keys=True) + "\n")
    if (
        document.get("status") in {"PASS", "KLEAN_NO_OBLIGATIONS"}
        and publication_snapshot is not None
    ):
        try:
            _require_unchanged_snapshot(
                publication_snapshot,
                frozen_input,
                discovery_manifest,
                generation,
            )
        except KleanPreflightError as error:
            document = error_document(error)
            temporary.write_text(
                json.dumps(document, indent=2, sort_keys=True) + "\n"
            )
    os.replace(temporary, output)
    return document


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", required=True, type=Path)
    parser.add_argument("--discovery-manifest", required=True, type=Path)
    parser.add_argument("--generation", required=True, type=Path)
    parser.add_argument(
        "--toolchain-lock",
        type=Path,
        default=REPO / "data/klean-toolchain.lock.json",
    )
    arguments = parser.parse_args(argv)
    try:
        result = run_preflight(
            arguments.input,
            arguments.discovery_manifest,
            arguments.generation,
            toolchain_lock=arguments.toolchain_lock,
        )
    except KleanPreflightError as error:
        print(f"Klean preflight failed: {error}", file=sys.stderr)
        return 2
    print(json.dumps(result, sort_keys=True))
    return (
        0
        if result["status"] in {"PASS", "KLEAN_NO_OBLIGATIONS"}
        else 1
    )
