  pid_t getpid(void) {

      if (status != NULL) {
          char line[256];
          while (fgets(line, sizeof(line), status) != NULL) {
              long host_pid;
              if (sscanf(line, "Pid:\t%ld", &host_pid) == 1) {
                  fclose(status);
                  return (pid_t)host_pid;
              }
          }
          fclose(status);
      }
      return (pid_t)syscall(SYS_getpid);
  }
