def findLeanInstall? : BaseIO (Option LeanInstall) := do
  if let some sysroot ← IO.getEnv "LEAN_SYSROOT" then
    return some <| ← LeanInstall.get sysroot
  let lean ← do
    if let some lean ← IO.getEnv "LEAN" then
      if lean.trim.isEmpty then
        return none
      else
        pure lean
    else
      pure "lean"

    return some <| ← LeanInstall.get sysroot
  return none
