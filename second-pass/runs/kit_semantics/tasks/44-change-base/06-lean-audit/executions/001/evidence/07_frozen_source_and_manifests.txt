     1	def change_base(x: int, base: int):
     2	    if x == 0:
     3	        return "0"
     4	    sign = ""
     5	    if x < 0:
     6	        sign = "-"
     7	        x = -x
     8	    result = ""
     9	    while x > 0:
    10	        result = chr(48 + x % base) + result
    11	        x = x // base
    12	    return sign + result
