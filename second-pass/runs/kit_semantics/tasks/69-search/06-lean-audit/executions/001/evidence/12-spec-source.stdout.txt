     1	requires "verification.k"
     2	
     3	module SPEC
     4	  imports VERIFICATION
     5	
     6	  // The inner loop adds exactly the number of occurrences of candidate in
     7	  // REM to the running frequency. It otherwise preserves the function state.
     8	  claim [inner-loop]:
     9	    <k>
    10	      #loop(
    11	        list(REM:ValSeq),
    12	        Name("item"),
    13	        If(
    14	          Compare(Name("item"), CmpOp("==", Name("candidate"))),
    15	          Assign(
    16	            Name("frequency"),
    17	            BinOp("+", Name("frequency"), Int(1))),
    18	          .Stmts))
    19	      => .K
    20	      ...
    21	    </k>
    22	    <env> L:Int </env>
    23	    <scopes>
    24	      ...
    25	      L |-> scope(
    26	        "lst"       |-> ref(H:Int)
    27	        "answer"    |-> A:Int
    28	        "candidate" |-> X:Val
    29	        "frequency" |-> (
    30	          C:Int => C +Int frequencyOf(projectIntTotal(X), REM))
    31	        "item"      |-> (OLDITEM:Val => ?NEWITEM:Val),
    32	        parent(0))
    33	      ...
    34	    </scopes>
    35	    <heap> ... H |-> list(FULL:ValSeq) ... </heap>
    36	    requires isIntVal(X) andBool allPositive(REM)
    37	
    38	  // The outer loop folds every remaining candidate using its exact frequency
    39	  // in FULL. The existential loop-local values are intentionally unobserved.
    40	  claim [outer-loop]:
    41	    <k>
    42	      #loop(
    43	        list(REM:ValSeq),
    44	        Name("candidate"),
    45	        Assign(Name("frequency"), Int(0))
    46	        For(
    47	          Name("item"),
    48	          Name("lst"),
    49	          If(
    50	            Compare(Name("item"), CmpOp("==", Name("candidate"))),
    51	            Assign(
    52	              Name("frequency"),
    53	              BinOp("+", Name("frequency"), Int(1))),
    54	            .Stmts))
    55	        If(
    56	          Compare(Name("frequency"), CmpOp(">=", Name("candidate"))),
    57	          If(
    58	            Compare(Name("candidate"), CmpOp(">", Name("answer"))),
    59	            Assign(
    60	              Name("answer"),
    61	              BinOp("+", Name("candidate"), Int(0))),
    62	            .Stmts),
    63	          .Stmts))
    64	      => .K
    65	      ...
    66	    </k>
    67	    <env> L:Int </env>
    68	    <scopes>
    69	      ...
    70	      L |-> scope(
    71	        "lst"       |-> ref(H:Int)
    72	        "answer"    |-> (A:Int => searchSummary(REM, FULL, A))
    73	        "candidate" |-> (OLDCAND:Val => ?NEWCAND:Val)
    74	        "frequency" |-> (OLDFREQ:Int => ?NEWFREQ:Int)
    75	        "item"      |-> (OLDITEM:Val => ?NEWITEM:Val),
    76	        parent(0))
    77	      ...
    78	    </scopes>
    79	    <heap> ... H |-> list(FULL:ValSeq) ... </heap>
    80	    requires allPositive(FULL) andBool allPositive(REM)
    81	
    82	  // Whole-program target: load the exact translated definition, resolve and
    83	  // call search, execute its body, and return the mathematical fold result.
    84	  claim [search-program]:
    85	    <k>
    86	      #loadAll(
    87	        Module(
    88	          FuncDef(
    89	            "search",
    90	            Params("lst"),
    91	            Assign(Name("answer"), UnaryOp("-", Int(1)))
    92	            Assign(Name("candidate"), Int(1))
    93	            Assign(Name("frequency"), Int(0))
    94	            Assign(Name("item"), Int(1))
    95	            For(
    96	              Name("candidate"),
    97	              Name("lst"),
    98	              Assign(Name("frequency"), Int(0))
    99	              For(
   100	                Name("item"),
   101	                Name("lst"),
   102	                If(
   103	                  Compare(
   104	                    Name("item"),
   105	                    CmpOp("==", Name("candidate"))),
   106	                  Assign(
   107	                    Name("frequency"),
   108	                    BinOp("+", Name("frequency"), Int(1))),
   109	                  .Stmts))
   110	              If(
   111	                Compare(
   112	                  Name("frequency"),
   113	                  CmpOp(">=", Name("candidate"))),
   114	                If(
   115	                  Compare(
   116	                    Name("candidate"),
   117	                    CmpOp(">", Name("answer"))),
   118	                  Assign(
   119	                    Name("answer"),
   120	                    BinOp("+", Name("candidate"), Int(0))),
   121	                  .Stmts),
   122	                .Stmts))
   123	            Return(Name("answer")))))
   124	      ~> Call(Name("search"), ref(0))
   125	      => searchSummary(INPUT, INPUT, -1) ~> .K
   126	    </k>
   127	    <env> 0 </env>
   128	    <scopes>
   129	      0  |-> scope(
   130	        (.Map => "search" |-> ?SEARCHCLOSURE:Val),
   131	        parent(-1))
   132	      -1 |-> builtinsScope
   133	    </scopes>
   134	    <scopeLoc> 1 </scopeLoc>
   135	    <heap> 0 |-> list(INPUT:ValSeq) </heap>
   136	    <heapLoc> 1 </heapLoc>
   137	    <stack> .List </stack>
   138	    <ret> noRet </ret>
   139	    <exc> NoExc </exc>
   140	    <exit-code> 0 </exit-code>
   141	    requires INPUT =/=K .ValSeq andBool allPositive(INPUT)
   142	endmodule
