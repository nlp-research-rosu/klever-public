     1	requires "reference-semantics/semantics.k"
     2	
     3	module VERIFICATION
     4	  imports MPY
     5	  imports INT
     6	  imports BOOL
     7	
     8	  // Guarded dynamic-to-static Int refinement. The projection has no evaluator
     9	  // outside its guard; on actual Int values it is the identity.
    10	  syntax Bool ::= isIntVal(Val) [function, total]
    11	  rule isIntVal(_:Int) => true
    12	  rule isIntVal(_:Val) => false [owise]
    13	
    14	  syntax Bool ::= definedProjectInt(Val) [function, total]
    15	  rule definedProjectInt(V:Val) => isIntVal(V)
    16	
    17	  syntax Int ::= projectIntTotal(Val)
    18	    [function, total, symbol(projectIntTotal), no-evaluators]
    19	
    20	  rule #Ceil({@V:Val}:>Int)
    21	    => ({ definedProjectInt(@V) #Equals true } #And #Ceil(@V))
    22	    [simplification]
    23	
    24	  rule projectIntTotal(V:Val) => {V}:>Int
    25	    requires definedProjectInt(V)
    26	    [concrete, simplification(10), preserves-definedness]
    27	
    28	  rule {V:Val}:>Int => projectIntTotal(V)
    29	    requires definedProjectInt(V)
    30	    [symbolic(V), simplification, preserves-definedness]
    31	
    32	  rule projectIntTotal(I:Int) => I [simplification]
    33	  rule projectIntTotal(projectIntTotal(V:Val)) => projectIntTotal(V)
    34	    [simplification]
    35	
    36	  // Guarded dispatch twins of the frozen MPY-INT equations. They do not add
    37	  // behavior: they expose the same integer operations when operands have the
    38	  // dynamic Val sort but the path condition establishes both are Int values.
    39	  rule applyCmp("==", V:Val, W:Val)
    40	    => projectIntTotal(V) ==Int projectIntTotal(W)
    41	    requires isIntVal(V) andBool isIntVal(W)
    42	    [simplification]
    43	
    44	  rule applyCmp(">=", V:Val, W:Val)
    45	    => projectIntTotal(V) >=Int projectIntTotal(W)
    46	    requires isIntVal(V) andBool isIntVal(W)
    47	    [simplification]
    48	
    49	  rule applyCmp(">", V:Val, W:Val)
    50	    => projectIntTotal(V) >Int projectIntTotal(W)
    51	    requires isIntVal(V) andBool isIntVal(W)
    52	    [simplification]
    53	
    54	  rule applyBin("+", V:Val, W:Val)
    55	    => projectIntTotal(V) +Int projectIntTotal(W)
    56	    requires isIntVal(V) andBool isIntVal(W)
    57	    [simplification]
    58	
    59	  // The HumanEval precondition: every input element is a positive integer.
    60	  syntax Bool ::= allPositive(ValSeq) [function, total]
    61	  rule allPositive(.ValSeq) => true
    62	  rule allPositive(vCons(V:Val, VS:ValSeq))
    63	    => isIntVal(V)
    64	       andBool projectIntTotal(V) >Int 0
    65	       andBool allPositive(VS)
    66	
    67	  // Exact frequency of X in a finite integer sequence.
    68	  syntax Int ::= frequencyOf(Int, ValSeq) [function, total]
    69	  rule frequencyOf(_:Int, .ValSeq) => 0
    70	  rule frequencyOf(X:Int, vCons(V:Val, VS:ValSeq))
    71	    => (#if X ==Int projectIntTotal(V) #then 1 #else 0 #fi)
    72	       +Int frequencyOf(X, VS)
    73	    requires isIntVal(V)
    74	  rule frequencyOf(X:Int, vCons(V:Val, VS:ValSeq))
    75	    => frequencyOf(X, VS)
    76	    requires notBool isIntVal(V)
    77	
    78	  // One outer-loop update.
    79	  syntax Int ::= updateAnswer(Int, Int, Int) [function, total]
    80	  rule updateAnswer(A:Int, X:Int, N:Int) => X
    81	    requires N >=Int X andBool X >Int A
    82	  rule updateAnswer(A:Int, X:Int, N:Int) => A
    83	    requires N <Int X
    84	  rule updateAnswer(A:Int, X:Int, N:Int) => A
    85	    requires N >=Int X andBool X <=Int A
    86	
    87	  // Exact fold performed by the outer loop over the remaining candidates.
    88	  syntax Int ::= searchSummary(ValSeq, ValSeq, Int) [function, total]
    89	  rule searchSummary(.ValSeq, _:ValSeq, A:Int) => A
    90	  rule searchSummary(
    91	         vCons(V:Val, REST:ValSeq),
    92	         FULL:ValSeq,
    93	         A:Int)
    94	    => searchSummary(
    95	         REST,
    96	         FULL,
    97	         updateAnswer(
    98	           A,
    99	           projectIntTotal(V),
   100	           frequencyOf(projectIntTotal(V), FULL)))
   101	    requires isIntVal(V)
   102	  rule searchSummary(
   103	         vCons(V:Val, REST:ValSeq),
   104	         FULL:ValSeq,
   105	         A:Int)
   106	    => searchSummary(REST, FULL, A)
   107	    requires notBool isIntVal(V)
   108	endmodule
