   297	def «Bytes2Int(_,_,_)_BYTES-HOOKED_Int_Bytes_Endianness_Signedness» (bytes : SortBytes) (endian : SortEndianness) (sign : SortSignedness) : Option SortInt :=
   298	  match sign with
   299	  | .unsignedBytes => Int.ofNat unsigned
   300	  | .signedBytes => if 2 * unsigned >= modulus then (Int.ofNat unsigned) - (Int.ofNat modulus)
   301	                    else Int.ofNat unsigned
   302	  where
   303	  modulus : Nat := res.1
   304	  unsigned : Nat := res.2
   305	  res : Nat×Nat :=
   306	    let littleEndian := match endian with
   307	                        | .littleEndianBytes => bytes.toList
   308	                            --match bytes with |⟨⟨l⟩⟩ => l
   309	                        | .bigEndianBytes => bytes.toList.reverse
   310	                            --match bytes with |⟨⟨l⟩⟩ => l.reverse
   311	    let go (res : Nat×Nat) (b : UInt8) : Nat×Nat :=
   312	      -- `place` is `res.1`
   313	      -- `acc` is `res.2`
   314	      let place := res.1 * 0x100
   315	      let acc := res.2 + res.1 * b.toNat
