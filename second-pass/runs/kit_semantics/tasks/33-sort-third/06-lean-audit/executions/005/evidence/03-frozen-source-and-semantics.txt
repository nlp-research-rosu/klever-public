     1	def sort_third(l: list):
     2	    thirds = sorted(l[::3])
     3	    result = []
     4	    i = 0
     5	    while i < len(l):
     6	        if i % 3 == 0:
     7	            result.append(thirds[i // 3])
     8	        else:
     9	            result.append(l[i])
    10	        i += 1
    11	    return result
