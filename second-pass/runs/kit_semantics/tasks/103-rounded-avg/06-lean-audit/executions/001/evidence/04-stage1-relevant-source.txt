FILE /reference/k-proof/solution.py
     1	def rounded_avg(n, m):
     2	    if n > m:
     3	        return -1
     4	    value = (n + m) // 2 + (n + m) % 2 * ((n + m) // 2 % 2)
     5	    digits = ""
     6	    while value > 1:
     7	        digits = chr(48 + value % 2) + digits
     8	        value = value // 2
     9	    digits = "1" + digits
    10	    return "0b" + digits
