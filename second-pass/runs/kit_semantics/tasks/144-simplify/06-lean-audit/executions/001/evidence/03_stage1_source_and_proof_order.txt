### solution.py
     1	def simplify(x, n):
     2	    part = 0
     3	    a = 0
     4	    b = 0
     5	    c = 0
     6	    d = 0
     7	    ch = ""
     8	    for ch in x + "/" + n:
     9	        if ch == "/":
    10	            part = part + 1
    11	        elif part == 0:
    12	            a = a * 10 + (ord(ch) - 48)
    13	        elif part == 1:
    14	            b = b * 10 + (ord(ch) - 48)
    15	        elif part == 2:
    16	            c = c * 10 + (ord(ch) - 48)
    17	        else:
    18	            d = d * 10 + (ord(ch) - 48)
    19	    return (a * c) % (b * d) == 0
