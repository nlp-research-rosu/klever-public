--- solution.py ---
     1	def decode_cyclic(s: str):
     2	    """
     3	    Takes a string encoded by rotating each complete three-character group
     4	    left once, and restores the original string.
     5	    """
     6	    result = ""
     7	    group = ""
     8	    char = ""
     9	    for char in s:
    10	        group = group + char
    11	        if len(group) == 3:
    12	            result = result + (group[2] + group[:2])
    13	            group = ""
    14	    return result + group
