     3	module VERIFICATION-SYNTAX
     4	  imports MPY
     5	  imports INT
     6	
     7	  syntax Stmts ::= "parseMusicBody" [function, total]
     8	                 | "parseMusicCharBody" [function, total]
     9	                 | "mutatedParseMusicBody" [function, total]
    10	                 | "mutatedParseMusicCharBody" [function, total]
    11	
    12	  // Exact mathematical state transition performed for one input character.
    13	  syntax Int ::= nextCurrent(Int, Int) [function, total]
    14	  syntax ValSeq ::= nextResult(ValSeq, Int, Int) [function, total]
    15	
    16	  // Exact fold of those transitions over the unprocessed character codes.
    17	  syntax Int ::= scanCurrent(IntSeq, Int) [function, total]
    18	  syntax ValSeq ::= scanResult(IntSeq, Int, ValSeq) [function, total]
    19	                  | musicResult(IntSeq) [function, total]
    20	endmodule
    21	
    22	module VERIFICATION
    23	  imports VERIFICATION-SYNTAX
    24	  imports MPY
    25	  imports INT
    26	  imports BOOL
    27	
    28	  rule parseMusicBody =>
    29	    Assign(Name("result"), ListExpr(.Exprs))
    30	    Assign(Name("current"), Int(0))
    31	    For(Name("char"), Name("music_string"), parseMusicCharBody)
    32	    If(Compare(Name("current"), CmpOp("==", Int(4))),
    33	       Expr(Call(Attribute(Name("result"), "append"), Name("current") , .Exprs)) .Stmts,
    34	       .Stmts)
    35	    Return(Name("result"))
    36	
    37	  rule parseMusicCharBody =>
    38	    If(Compare(Name("char"), CmpOp("==", Str("o"))),
    39	       Assign(Name("current"), Int(4)) .Stmts,
    40	       If(Compare(Name("char"), CmpOp("==", Str("."))),
    41	          Assign(Name("current"), Int(1)) .Stmts,
    42	          If(Compare(Name("char"), CmpOp("==", Str("|"))),
    43	             If(Compare(Name("current"), CmpOp("==", Int(4))),
    44	                Assign(Name("current"), Int(2)) .Stmts,
    45	                .Stmts)
    46	             Expr(Call(Attribute(Name("result"), "append"), Name("current") , .Exprs))
    47	             Assign(Name("current"), Int(0)),
    48	             If(Compare(Name("current"), CmpOp("==", Int(4))),
    49	                Expr(Call(Attribute(Name("result"), "append"), Name("current") , .Exprs)) .Stmts,
    50	                .Stmts)
    51	             Assign(Name("current"), Int(0)))))
    52	
    53	  // Validation-only body mutation; no positive proof claim references it.
    54	  rule mutatedParseMusicBody =>
    55	    Assign(Name("result"), ListExpr(.Exprs))
    56	    Assign(Name("current"), Int(0))
    57	    For(Name("char"), Name("music_string"), mutatedParseMusicCharBody)
    58	    If(Compare(Name("current"), CmpOp("==", Int(4))),
    59	       Expr(Call(Attribute(Name("result"), "append"), Name("current") , .Exprs)) .Stmts,
    60	       .Stmts)
    61	    Return(Name("result"))
    62	
    63	  rule mutatedParseMusicCharBody =>
    64	    If(Compare(Name("char"), CmpOp("==", Str("o"))),
    65	       Assign(Name("current"), Int(5)) .Stmts,
    66	       If(Compare(Name("char"), CmpOp("==", Str("."))),
    67	          Assign(Name("current"), Int(1)) .Stmts,
    68	          If(Compare(Name("char"), CmpOp("==", Str("|"))),
    69	             If(Compare(Name("current"), CmpOp("==", Int(4))),
    70	                Assign(Name("current"), Int(2)) .Stmts,
    71	                .Stmts)
    72	             Expr(Call(Attribute(Name("result"), "append"), Name("current") , .Exprs))
    73	             Assign(Name("current"), Int(0)),
    74	             If(Compare(Name("current"), CmpOp("==", Int(4))),
    75	                Expr(Call(Attribute(Name("result"), "append"), Name("current") , .Exprs)) .Stmts,
    76	                .Stmts)
    77	             Assign(Name("current"), Int(0)))))
    78	
    79	  rule nextCurrent(C:Int, _:Int) => 4
    80	    requires C ==Int 111
    81	  rule nextCurrent(C:Int, _:Int) => 1
    82	    requires C ==Int 46
    83	  rule nextCurrent(C:Int, _:Int) => 0
    84	    requires C =/=Int 111 andBool C =/=Int 46
    85	
    86	  rule nextResult(ACC:ValSeq, C:Int, CUR:Int)
    87	    => valSeqConcat(ACC, vCons(2, .ValSeq))
    88	    requires C ==Int 124 andBool CUR ==Int 4
    89	  rule nextResult(ACC:ValSeq, C:Int, CUR:Int)
    90	    => valSeqConcat(ACC, vCons(CUR, .ValSeq))
    91	    requires C ==Int 124 andBool CUR =/=Int 4
    92	  rule nextResult(ACC:ValSeq, C:Int, CUR:Int)
    93	    => valSeqConcat(ACC, vCons(4, .ValSeq))
    94	    requires C =/=Int 111 andBool C =/=Int 46 andBool C =/=Int 124
    95	     andBool CUR ==Int 4
    96	  rule nextResult(ACC:ValSeq, C:Int, CUR:Int) => ACC
    97	    requires (C ==Int 111 orBool C ==Int 46)
    98	      orBool (C =/=Int 124 andBool CUR =/=Int 4)
    99	
   100	  rule scanCurrent(.IntSeq, CUR:Int) => CUR
   101	  rule scanCurrent(iCons(C:Int, REST:IntSeq), CUR:Int)
   102	    => scanCurrent(REST, nextCurrent(C, CUR))
   103	
   104	  rule scanResult(.IntSeq, _:Int, ACC:ValSeq) => ACC
   105	  rule scanResult(iCons(C:Int, REST:IntSeq), CUR:Int, ACC:ValSeq)
   106	    => scanResult(REST, nextCurrent(C, CUR), nextResult(ACC, C, CUR))
   107	
   108	  rule musicResult(CS:IntSeq)
   109	    => valSeqConcat(scanResult(CS, 0, .ValSeq), vCons(4, .ValSeq))
   110	    requires scanCurrent(CS, 0) ==Int 4
   111	  rule musicResult(CS:IntSeq) => scanResult(CS, 0, .ValSeq)
   112	    requires scanCurrent(CS, 0) =/=Int 4
   113	endmodule
