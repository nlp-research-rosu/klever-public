     1	from typing import List
     2	
     3	
     4	def parse_nested_parens(paren_string: str) -> List[int]:
     5	    depths = []
     6	    depth = 0
     7	    deepest = 0
     8	    char = ""
     9	
    10	    for char in paren_string:
    11	        if char == "(":
    12	            depth += 1
    13	            if depth > deepest:
    14	                deepest = depth
    15	        elif char == ")":
    16	            depth -= 1
    17	        elif deepest > 0:
    18	            depths.append(deepest)
    19	            deepest = 0
    20	
    21	    if deepest > 0:
    22	        depths.append(deepest)
    23	
    24	    return depths
