     1	def prod_signs(arr):
     2	    """
     3	    Return the sum of the magnitudes in arr multiplied by the product of
     4	    their signs.  The empty input has no such product and returns None.
     5	    """
     6	    if arr == []:
     7	        return None
     8	
     9	    total = 0
    10	    sign = 1
    11	    for value in arr:
    12	        if value < 0:
    13	            total = total - value
    14	            sign = 0 - sign
    15	        else:
    16	            total = total + value
    17	            if value == 0:
    18	                sign = 0
    19	
    20	    return total * sign
