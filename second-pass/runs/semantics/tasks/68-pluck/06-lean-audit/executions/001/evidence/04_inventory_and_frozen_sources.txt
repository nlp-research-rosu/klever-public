     1	def pluck(arr):
     2	    smallest = -1
     3	    smallest_index = -1
     4	    index = 0
     5	    value = 0
     6	    for value in arr:
     7	        if value % 2 == 0:
     8	            if smallest == -1:
     9	                smallest = value
    10	                smallest_index = index
    11	            else:
    12	                if value < smallest:
    13	                    smallest = value
    14	                    smallest_index = index
    15	        index += 1
    16	    if smallest == -1:
    17	        return []
    18	    return [smallest, smallest_index]
