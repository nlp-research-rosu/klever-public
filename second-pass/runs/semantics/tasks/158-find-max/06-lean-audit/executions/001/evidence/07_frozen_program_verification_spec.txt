solution.py:
     1	def find_max(words):
     2	    best = ""
     3	    max_unique = 0
     4	    word = ""
     5	    unique = 0
     6	    for word in words:
     7	        unique = len(set(word))
     8	        if unique > max_unique:
     9	            best = word
    10	            max_unique = unique
    11	        elif unique == max_unique:
    12	            if word < best:
    13	                best = word
    14	    return best
