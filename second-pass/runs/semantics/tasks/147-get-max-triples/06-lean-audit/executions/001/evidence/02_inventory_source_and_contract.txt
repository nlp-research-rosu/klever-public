+ nl -ba /reference/k-proof/solution.py
     1	def get_max_triples(n):
     2	    zero_count = (n + 1) // 3
     3	    one_count = n - zero_count
     4	    zero_triples = zero_count * (zero_count - 1) * (zero_count - 2) // 6
     5	    one_triples = one_count * (one_count - 1) * (one_count - 2) // 6
     6	    return zero_triples + one_triples
