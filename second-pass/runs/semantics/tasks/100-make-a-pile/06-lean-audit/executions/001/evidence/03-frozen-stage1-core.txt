     1	requires "reference-semantics/semantics.k"
     2	
     3	module PILE-VERIFICATION
     4	  imports MPY
     5	
     6	  syntax Expr ::= "pileCondition" [macro]
     7	  rule pileCondition
     8	    => Compare(Name("i"), CmpOp("<", Name("n")))
     9	
    10	  syntax Stmts ::= "pileLoopBody" [macro]
    11	  rule pileLoopBody
    12	    => Expr(
    13	         Call(
    14	           Attribute(Name("stones"), "append"),
    15	           BinOp("+", Name("n"), BinOp("*", Int(2), Name("i")))))
    16	       AugAssign(Name("i"), "+", Int(1))
    17	
    18	  syntax Stmts ::= "pileBody" [macro]
    19	  rule pileBody
    20	    => Assign(Name("stones"), ListExpr(.Exprs))
    21	       Assign(Name("i"), Int(0))
    22	       While(pileCondition, pileLoopBody)
    23	       Return(Name("stones"))
    24	
    25	  syntax Val ::= "pileClosure" [macro]
    26	  rule pileClosure => closureVal(("n", .ParamNames), pileBody, 0)
    27	
    28	  syntax Module ::= "pileModule" [macro]
    29	  rule pileModule
    30	    => Module(FuncDef("make_a_pile", Params("n"), pileBody))
    31	
    32	  // pile(N, I) is the mathematical suffix at level index I:
    33	  // [N + 2*I, N + 2*(I+1), ..., N + 2*(N-1)].
    34	  syntax ValSeq ::= pile(Int, Int) [function, total]
    35	  rule pile(N:Int, I:Int) => .ValSeq
    36	    requires I >=Int N
    37	  rule pile(N:Int, I:Int)
    38	    => vCons(N +Int (2 *Int I), pile(N, I +Int 1))
    39	    requires I <Int N
    40	
    41	  // Two standard list-concatenation identities orient symbolic accumulators
    42	  // into the same form on successive induction steps.
    43	  rule valSeqConcat(VS:ValSeq, .ValSeq) => VS [simplification]
    44	  rule valSeqConcat(valSeqConcat(A:ValSeq, B:ValSeq), C:ValSeq)
    45	    => valSeqConcat(A, valSeqConcat(B, C))
    46	    [simplification]
    47	endmodule

$ nl -ba /reference/k-proof/solution.py
     1	def make_a_pile(n):
     2	    stones = []
     3	    i = 0
     4	    while i < n:
     5	        stones.append(n + 2 * i)
     6	        i += 1
     7	    return stones
