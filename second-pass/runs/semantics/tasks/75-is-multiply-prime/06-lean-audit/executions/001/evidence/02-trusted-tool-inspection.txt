_MODULE_NAME_RE = re.compile(
    r"^[ \t]+(?P<name>[A-Za-z][A-Za-z0-9_'-]*)"
    r"(?:[ \t]+\[[^\]\r\n]*\])?[ \t]*$"

    r"\[(?P<attributes>[^\[\]]*)\][ \t\r\n]*\Z", re.S

    r"<(?P<close>/)?(?P<name>[A-Za-z][A-Za-z0-9_'-]*)"

)

def canonical_json_sha256(document: Any) -> str:
    encoded = json.dumps(
        document,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
    ).encode()
    return hashlib.sha256(encoded).hexdigest()


def _mask_non_code(text: str) -> str:
    """Mask comments and string bodies without changing source offsets."""

    output = list(text)
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                state = "code"
            else:
                output[index] = " "
            index += 1
            continue
        if state == "string":
            if character == "\\" and following:
                output[index] = " "
                output[index + 1] = " "
                index += 2
                continue
            if character == '"':
                state = "code"
            elif character not in "\r\n":
                output[index] = " "
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                output[index] = output[index + 1] = " "
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                output[index] = output[index + 1] = " "
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character not in "\r\n":
                output[index] = " "
            index += 1
            continue
        if character == "/" and following == "/":
            output[index] = output[index + 1] = " "
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output[index] = output[index + 1] = " "
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        if character == '"':
            state = "string"
        index += 1
    return "".join(output)


def _line_ranges(text: str, start: int = 0) -> list[tuple[int, int, int]]:
    ranges: list[tuple[int, int, int]] = []
    line_start = start
    while line_start < len(text):
        newline = text.find("\n", line_start)
        next_start = len(text) if newline < 0 else newline + 1
        line_end = len(text) if newline < 0 else newline
        content_end = (
            line_end - 1
            if line_end > line_start and text[line_end - 1] == "\r"
            else line_end
        )
        ranges.append((line_start, content_end, next_start))
        if newline < 0:
            break
        line_start = next_start
    return ranges


def _line_heads(text: str) -> list[tuple[int, int, str, str]]:
    """Return physical-line word heads without assigning K grammar meaning."""

    heads: list[tuple[int, int, str, str]] = []
    for line_start, content_end, _next_start in _line_ranges(text):
        index = line_start
        while index < content_end and text[index] in " \t":
            index += 1
        word_end = index
        while word_end < content_end and (
            text[word_end].isalnum() or text[word_end] in "_'"
        ):
            word_end += 1
        if word_end > index:
            heads.append(
                (
                    line_start,
                    content_end,
                    text[index:word_end],
                    text[word_end:content_end],
                )
            )
    return heads


def _line_word(line: str) -> tuple[str, str, int]:
    stripped = line.lstrip()
    indent = len(line) - len(stripped)
    word_end = 0
    while word_end < len(stripped) and (
        stripped[word_end].isalnum() or stripped[word_end] in "_'"
    ):
        word_end += 1
    return stripped[:word_end], stripped[word_end:], indent + word_end


def _rule_continuation(line: str) -> tuple[str, int] | None:
    stripped = line.lstrip()
    if stripped.startswith("["):
        return "attributes", len(line) - len(stripped)
    word, rest, word_end = _line_word(line)
    for keyword in ("requires", "ensures"):
        if word == keyword and (
            not rest
            or not (
                rest[0].isalnum()
                or rest[0] in "_'"
            )
        ):
            return keyword, word_end
    return None


def _starts_outer_transition(line: str) -> bool:
    keyword, rest, _word_end = _line_word(line)
    separated = bool(rest) and rest[0] in " \t"
    if keyword == "module":
        return separated and _MODULE_NAME_RE.fullmatch(rest) is not None
    if keyword == "endmodule":
        return not rest.strip()
    if keyword == "rule":
        return not rest or separated
    return keyword in _OUTER_SENTENCE_KEYWORDS and separated


def _terminal_attribute_line(line: str) -> bool:
    match = _TRAILING_ATTRIBUTES_RE.search(line)
    if match is None:
        return False
    start = match.start()
    return start == 0 or line[start - 1].isspace()


def _scan_rule_delimiters(
    line: str,
    delimiters: list[str],
) -> None:
    closing = {")": "(", "]": "[", "}": "{"}
    index = 0
    while index < len(line):
        if line[index] == "<":
            cell = _CELL_TAG_RE.match(line, index)
            if cell is not None:
                marker = f"<{cell.group('name')}>"
                if cell.group("close"):
                    if delimiters and delimiters[-1] == marker:
                        delimiters.pop()
                elif not cell.group("self_close"):
                    delimiters.append(marker)
                index = cell.end()
                continue
        character = line[index]
        if character in "([{":
            delimiters.append(character)
        elif character in closing:
            if delimiters and delimiters[-1] == closing[character]:
                delimiters.pop()
        index += 1


def _rule_bubble_end(masked_text: str, start: int) -> int:
    """Find when an outer rule bubble returns control to outer parsing."""

    seen_rewrite = False
    clause_has_content = False
    candidate_end: int | None = None
    delimiters: list[str] = []
    ranges = _line_ranges(masked_text, start)
    for line_start, line_end, next_start in ranges:
        line = masked_text[line_start:line_end]
        if not line.strip():
            continue

        continuation = (
            _rule_continuation(line)
            if seen_rewrite and candidate_end is not None
            else None
        )
        if candidate_end is not None:
            if continuation is None and _starts_outer_transition(line):
                return candidate_end
            candidate_end = None

        content_start = 0
        if continuation is not None:
            kind, content_start = continuation
            if kind in {"requires", "ensures"}:
                clause_has_content = False

        if not seen_rewrite:
            rewrite = line.find("=>")
            if rewrite >= 0:
                seen_rewrite = True
                content_start = rewrite + 2
            else:
                content_start = len(line)

        _scan_rule_delimiters(line, delimiters)
        if seen_rewrite and line[content_start:].strip():
            clause_has_content = True

        if (
            seen_rewrite
            and clause_has_content
            and not delimiters
            and _terminal_attribute_line(line)
        ):
            return next_start
        if seen_rewrite and clause_has_content and not delimiters:
            candidate_end = next_start
    if candidate_end is not None:
        return candidate_end
    return len(masked_text) + 1

@dataclass(frozen=True)
class PreflightSnapshot:
    stage1_workspace_sha256: str
    stage3_discovery_manifest_sha256: str
    generated_tree_sha256: str
    sidecar_sha256: tuple[tuple[str, str], ...]


def _preflight_snapshot(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
) -> PreflightSnapshot:
    generated = generation / "generated"
    try:
        sidecars = tuple(
            (
                name,
                hashlib.sha256(
                    pipeline_path.read_bytes()
                ).hexdigest(),
            )
            for name in _SIDECAR_NAMES
            if (
                pipeline_path := generation / name
            ).is_file()
            and not pipeline_path.is_symlink()
        )
        if len(sidecars) != len(_SIDECAR_NAMES):
            raise KleanPreflightError(
                "preflight sidecar set is incomplete or unsafe"
            )
        return PreflightSnapshot(
            stage1_workspace_sha256=klean_export.tree_digest(
                frozen_input
            ),
            stage3_discovery_manifest_sha256=hashlib.sha256(
                discovery_manifest.read_bytes()
            ).hexdigest(),
            generated_tree_sha256=klean_export.tree_digest(generated),
            sidecar_sha256=sidecars,
        )
    except (OSError, klean_export.KleanExportError) as error:
        raise KleanPreflightError(
            "cannot snapshot immutable preflight inputs"
        ) from error


def _require_unchanged_snapshot(
    expected: PreflightSnapshot,
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
) -> None:
    current = _preflight_snapshot(
        frozen_input, discovery_manifest, generation
    )
    if current.stage1_workspace_sha256 != (
        expected.stage1_workspace_sha256
    ):
        raise KleanPreflightError(
            "frozen Stage 1 input changed during preflight"
        )
    if current.stage3_discovery_manifest_sha256 != (
        expected.stage3_discovery_manifest_sha256
    ):
        raise KleanPreflightError(
            "validated Stage 3 discovery manifest changed during preflight"
        )
    if current.generated_tree_sha256 != expected.generated_tree_sha256:
        raise KleanPreflightError(
            "generated project changed during preflight"
        )
    if current.sidecar_sha256 != expected.sidecar_sha256:
        raise KleanPreflightError(
            "generation sidecar changed during preflight"
        )


def _regular_json(path: Path, label: str) -> dict[str, Any]:
    if not path.is_file() or path.is_symlink():
        raise KleanPreflightError(f"{label} must be a regular file: {path}")
    try:
        document = json.loads(path.read_text())
    except (OSError, ValueError) as error:
        raise KleanPreflightError(f"{label} is malformed: {path}") from error
    if not isinstance(document, dict):
        raise KleanPreflightError(f"{label} must be a JSON object")
    return document

    declarations: dict[str, tuple[str, str]] = {}
    for source in sources:
        for declaration in klean_export.lean_trust_declarations(source):

            )

def _reject_proposition_trust(
    generated: Path,
    sources: list[Path],
    declarations: dict[str, tuple[str, str]],
) -> None:
    """Apply a policy that is independent of the generator's allowlist.

    Generated executable constants may remain opaque at the Klean boundary,
    but generated propositions/proofs may never be assumed.  In particular,
    the immutable target module must contain only a proposition definition,
    never an axiom or opaque proof.
    """

    for source in sources:
        relative = source.relative_to(generated).as_posix()
        if source.name == "Lemmas.lean" and re.search(
            r"(?m)^\s*(?:axiom|opaque)\s+", source.read_text()
        ):
            raise KleanPreflightError(
                f"proposition trust is forbidden in {relative}"
            )
    for name, (_kind, lean_type) in declarations.items():
        normalized = " ".join(lean_type.split())
        proposition_like = (
            normalized in {"Prop", "True", "False"}
            or " Prop" in f" {normalized}"
            or re.search(r":\s*Prop(?:\s|$)", normalized) is not None
            or normalized.startswith(("∀ ", "∃ "))
            or re.search(r"(^|[ (])(?:True|False)(?:$|[ )])", normalized)
            is not None
            or re.search(r"(?:^| )[^ ]+\s*=\s*[^ ]+", normalized) is not None
        )
        if proposition_like:
            raise KleanPreflightError(
                f"proposition trust is forbidden for {name}: {lean_type}"
            )

def validate_trust_parameter_links(
    expected_ids: list[str], parameters: list[object]
) -> set[str]:
    """Validate every opaque parameter without requiring one per theorem."""

    bound_rule_ids: set[str] = set()
    for parameter in parameters:
        if not isinstance(parameter, dict):
            raise KleanPreflightError("generated trust parameter is malformed")
        source_rule_ids = parameter.get("source_rule_ids")
        if (
            not isinstance(source_rule_ids, list)
            or not source_rule_ids
            or not all(
                isinstance(source_rule_id, str)
                and source_rule_id in expected_ids
                for source_rule_id in source_rule_ids
            )
        ):
            raise KleanPreflightError(
                "trust parameter is not bound to source obligations"
            )
        bound_rule_ids.update(source_rule_ids)
    return bound_rule_ids


def _check_imports(generated: Path, sources: list[Path]) -> None:
    for source in sources:
        for line in source.read_text().splitlines():
            stripped = line.strip()
            if not stripped.startswith("import "):
                continue
            imported = stripped.removeprefix("import ").strip()
            if (
                not imported
                or imported.startswith(("/", "."))
                or ".." in imported
                or '"' in imported
                or "\\" in imported
            ):
                raise KleanPreflightError(
                    f"untrusted Lean import in {source}: {imported!r}"
                )
    for name in ("lakefile.toml", "lakefile.lean", "lean-toolchain"):
        path = generated / name
        if path.exists() and (not path.is_file() or path.is_symlink()):
            raise KleanPreflightError(
                f"generated dependency metadata is not regular: {path}"
            )
        if path.is_file():
            text = path.read_text()
            if re.search(r"(?m)^\s*(?:path|dir)\s*=\s*\"/", text):
                raise KleanPreflightError(
                    f"absolute dependency path in generated {name}"
                )

def _verification_closure(
    modules: tuple[_Module, ...], verification: _Module
) -> tuple[_Module, ...]:

    reached: set[str] = set()
    pending = [verification.name]
    while pending:
        name = pending.pop()

def _without_comments(text: str) -> str:
    """Remove K comments while preserving strings and their exact contents."""

    output: list[str] = []
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                output.append(character)
                state = "code"
            index += 1
            continue
        if state == "string":
            output.append(character)
            if character == "\\" and following:
                output.append(following)
                index += 2
                continue
            if character == '"':
                state = "code"
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character in "\r\n":
                output.append(character)
            index += 1
            continue
        if character == "/" and following == "/":
            output.append(" ")
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output.append(" ")
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        output.append(character)
        if character == '"':
            state = "string"
        index += 1
    return "".join(output)


def _attribute_tokens(text: str, masked_text: str) -> list[str]:
    tokens: list[str] = []
    start = 0
    depth = 0
    for index, character in enumerate(masked_text):
        if character in "([{":
            depth += 1
        elif character in ")]}":
            depth = max(depth - 1, 0)
        elif character == "," and depth == 0:
            token = _without_comments(text[start:index]).strip()
            if token:
                tokens.append(token)
            start = index + 1
    final = _without_comments(text[start:]).strip()
    if final:
        tokens.append(final)
    return tokens


def _rule_spans(text: str, module: _Module) -> list[_RuleSpan]:
    masked_text = _mask_non_code(text)
    sentences = module.sentences
    rules: list[_RuleSpan] = []
    seen_hashes: set[str] = set()
    for index, sentence in enumerate(sentences):
        if sentence.keyword != "rule":
            continue
        end_offset = (
            sentences[index + 1].start
            if index + 1 < len(sentences)
            else module.content_end
        )
        start_offset = sentence.start
        raw_segment = text[start_offset:end_offset]
        masked_segment = masked_text[start_offset:end_offset]
        logical_length = len(masked_segment.rstrip(" \t\r\n"))
        if logical_length == 0:
            continue
        rule_text = raw_segment[:logical_length].rstrip(" \t\r\n")
        masked_rule = _mask_non_code(rule_text)
        attribute_match = _TRAILING_ATTRIBUTES_RE.search(masked_rule)
        attribute_start = None
        attribute_end = None
        attributes: list[str] = []
        if attribute_match is not None:
            attribute_start = attribute_match.start("attributes")
            attribute_end = attribute_match.end("attributes")
            attributes = _attribute_tokens(
                rule_text[attribute_start:attribute_end],
                masked_rule[attribute_start:attribute_end],
            )
        normalized = " ".join(rule_text.split())
        digest = hashlib.sha256(normalized.encode()).hexdigest()
        if digest in seen_hashes:
            raise KRuleInventoryError(
                f"duplicate normalized simplification rule hash: {digest}"
            )
        seen_hashes.add(digest)
        start_line = text.count("\n", 0, start_offset) + 1
        end_line = start_line + rule_text.count("\n")
        rule = KRule(
            source_rule_id=f"rule-{digest}",
            module=module.name,
            start_line=start_line,
            end_line=end_line,
            normalized_sha256=digest,
            attributes=attributes,
            text=rule_text,
        )
        rules.append(
            _RuleSpan(
                rule=rule,
                start_offset=start_offset,
                end_offset=start_offset + len(rule_text),
                attribute_start=(
                    None
                    if attribute_match is None
                    else start_offset + attribute_match.start()
                ),
                attribute_end=(
                    None
                    if attribute_match is None
                    else start_offset + attribute_match.end()
                ),
            )
        )
    return rules


def _rule_documents(
    text: str, modules: tuple[_Module, ...]
) -> list[dict[str, Any]]:
    spans = [span for module in modules for span in _rule_spans(text, module)]
    spans.sort(key=lambda span: span.start_offset)
    return [dataclasses.asdict(span.rule) for span in spans]


def _attribute_token_spans(
    text: str,
    masked_text: str,
) -> list[tuple[int, int, str]]:
    spans: list[tuple[int, int, str]] = []
    start = 0
    depth = 0
    for index, character in enumerate(masked_text):
        if character in "([{":
            depth += 1
        elif character in ")]}":
            depth = max(depth - 1, 0)
        elif character == "," and depth == 0:
            spans.append(
                (
                    start,
                    index,
                    _without_comments(text[start:index]).strip(),
                )
            )
            start = index + 1
    spans.append(
        (
            start,
            len(text),
            _without_comments(text[start:]).strip(),
        )
    )
    return spans


def _blank_non_newlines(text: str) -> str:
    return "".join(
        character if character in "\r\n" else " "
        for character in text
    )


def desimplify_rule_ids(
    text: str,
    source_rule_ids: list[str],
) -> str:
    """Remove simplification from exactly the selected canonical rule IDs."""

    if len(source_rule_ids) != len(set(source_rule_ids)):
        raise KRuleInventoryError(
            "exact rule transformation contains a duplicate source rule ID"
        )
    selected = set(source_rule_ids)
    modules = _modules(_mask_non_code(text))
    spans = [
        span
        for local_module in modules
        for span in _rule_spans(text, local_module)
    ]
    by_id = {span.rule.source_rule_id: span for span in spans}
    unknown = sorted(selected - set(by_id))
    if unknown:
        raise KRuleInventoryError(
            "exact rule transformation names an unknown source rule ID: "
            + ", ".join(unknown)
        )

    edits: list[tuple[int, int]] = []
    for source_rule_id in source_rule_ids:
        span = by_id[source_rule_id]
        if span.attribute_start is None or span.attribute_end is None:
            raise KRuleInventoryError(
                "selected source rule has no simplification attribute: "
                f"{source_rule_id}"
            )
        attribute_text = text[
            span.attribute_start + 1 : span.attribute_end - 1
        ]
        masked_attributes = _mask_non_code(attribute_text)
        tokens = _attribute_token_spans(
            attribute_text,
            masked_attributes,
        )
        matches = [
            index
            for index, (_start, _end, token) in enumerate(tokens)
            if token == "simplification"
        ]
        if len(matches) != 1:
            raise KRuleInventoryError(
                "selected source rule has ambiguous simplification "
                f"attributes: {source_rule_id}"
            )
        token_index = matches[0]
        if len(tokens) == 1:
            edit_start = span.attribute_start
            edit_end = span.attribute_end
        elif token_index < len(tokens) - 1:
            edit_start = (
                span.attribute_start + 1 + tokens[token_index][0]
            )
            edit_end = (
                span.attribute_start + 1 + tokens[token_index + 1][0]
            )
        else:
            edit_start = (
                span.attribute_start + 1 + tokens[token_index - 1][1]
            )
            edit_end = (
                span.attribute_start + 1 + tokens[token_index][1]
            )
        edits.append((edit_start, edit_end))

    transformed = text
    for start, end in sorted(edits, reverse=True):
        transformed = (
            transformed[:start]
            + _blank_non_newlines(transformed[start:end])
            + transformed[end:]
        )
    return transformed

        raise KleanPreflightError(f"preflight result already exists: {output}")

    def error_document(error: KleanPreflightError) -> dict[str, Any]:

            stage1_hash = klean_export.tree_digest(frozen_input)

            )
    os.replace(temporary, output)

def _framed_file_update(
    digest: Any, path: Path, expected_size: int
) -> None:

def require_real_directory(path: Path, label: str) -> Path:
    if not stat.S_ISDIR(_lstat(path, label).st_mode):
        raise PipelineContractError(f"{label} must be a real directory: {path}")
    try:
        return path.resolve(strict=True)
    except OSError as error:
        raise PipelineContractError(f"{label} cannot be resolved: {path}") from error


def require_regular_file(path: Path, label: str) -> Path:
    if not stat.S_ISREG(_lstat(path, label).st_mode):
        raise PipelineContractError(f"{label} must be a real regular file: {path}")
    try:

    except OSError as error:
        raise PipelineContractError(f"{label} cannot be resolved: {path}") from error

        for child in children:
            path = Path(child.path)
            try:
                mode = child.stat(follow_symlinks=False).st_mode
            except OSError as error:
