     1	def is_multiply_prime(a):
     2	    """Return whether a has exactly three prime factors, with multiplicity."""
     3	    factor_count = 0
     4	    factor = 2
     5	
     6	    while factor * factor <= a:
     7	        if a % factor == 0:
     8	            factor_count += 1
     9	            a //= factor
    10	        else:
    11	            factor += 1
    12	
    13	    if a > 1:
    14	        factor_count += 1
    15	
    16	    return factor_count == 3
