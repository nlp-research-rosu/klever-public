     1	requires "verification.k"

     3	module SPEC
     4	  imports VERIFICATION
     5	
     6	  // For an arbitrary floating-point input, executing the translated function
     7	  // returns exactly the semantics' Python floor-modulo primitive.  The
     8	  // supplied semantics evaluates that primitive concretely as
     9	  // n - floor(n / 1.0) * 1.0, i.e. the requested decimal part for n > 0.
    10	  claim
    11	    <k> #loadAll(solutionProgram)
    12	         ~> Call(Name("truncate_number"), (Float(N:Float), .Exprs))
    13	      => floatMod(N, 1.0) </k>
    14	    <env> 0 </env>
    15	    <scopes>
    16	      0 |-> scope(.Map, parent(-1))
    17	      -1 |-> builtinsScope
    18	      => 0 |-> scope(
    19	           "truncate_number" |-> closureVal(
    20	             ("number", .ParamNames),
    21	             Return(BinOp("%", Name("number"), Float(1.0))) .Stmts,
    22	             0),
    23	           parent(-1))
    24	         -1 |-> builtinsScope
    25	    </scopes>
    26	    <scopeLoc> 1 </scopeLoc>
    27	    <heap> .Map </heap>
    28	    <heapLoc> 0 </heapLoc>
    29	    <stack> .List </stack>
    30	    <ret> noRet </ret>
    31	    <exc> NoExc </exc>
    32	    <exit-code> 0 </exit-code>
    33	endmodule
