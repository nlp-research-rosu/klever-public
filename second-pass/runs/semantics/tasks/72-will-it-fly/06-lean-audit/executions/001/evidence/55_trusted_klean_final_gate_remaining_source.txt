    generation: Path,
    candidate: Path | None,
) -> dict[str, Any]:
    try:
        return {
            "k_workspace_sha256": pipeline_contract.sha256_tree(
                frozen_input
            ),
            "stage1_export_sha256": klean_export.tree_digest(
                frozen_input
            ),
            "discovery_manifest_sha256": (
                pipeline_contract.sha256_file(discovery_manifest)
            ),
            "klean_generation_sha256": (
                pipeline_contract.sha256_tree(generation)
            ),
            "lean_workspace_sha256": (
                pipeline_contract.sha256_tree(candidate)
                if candidate is not None
                else None
            ),
        }
    except (
        OSError,
        klean_export.KleanExportError,
        pipeline_contract.PipelineContractError,
    ) as error:
        raise KleanFinalGateError(
            "cannot hash Stage 6 mechanical inputs"
        ) from error


def _require_bound_inputs(
    audit_input: dict[str, Any],
    observed: dict[str, Any],
) -> None:
    expected = audit_input["resolution"]["hashes"]
    for key, digest in observed.items():
        if expected.get(key) != digest:
            raise KleanFinalGateError(
                f"Stage 6 {key} differs from audit input"
            )


def _result_binding(audit_input: Path) -> dict[str, Any]:
    try:
        document = _audit_input_binding(Path(audit_input))
    except KleanFinalGateError:
        return {}
    return {
        "mode": document["resolution"]["mode"],
        "resolved_input_sha256": document[
            "resolved_input_sha256"
        ],
        "input_hashes": document["resolution"]["hashes"],
    }


def check_final(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    candidate: Path | None,
    *,
    toolchain_lock: Path,
    audit_input: Path,
    run_command: Callable[..., tuple[int, str]] = _run,
) -> dict[str, Any]:
    frozen_input = Path(frozen_input)
    discovery_manifest = Path(discovery_manifest)
    generation = Path(generation)
    candidate = Path(candidate) if candidate is not None else None
    audit_binding = _audit_input_binding(Path(audit_input))
    signed_resolution = audit_binding["resolution"]
    if (
        signed_resolution["mode"] == "CLASSIFICATION_AND_PROOF"
        and candidate is None
    ):
        raise KleanFinalGateError(
            "proof-bearing audit requires a Stage 5 Lean candidate"
        )
    if (
        signed_resolution["mode"] == "CLASSIFICATION_ONLY"
        and candidate is not None
    ):
        raise KleanFinalGateError(
            "classification-only audit must not use a Stage 5 Lean "
            "candidate"
        )
    expected_mode = signed_resolution["mode"]
    observed_mode = (
        "CLASSIFICATION_AND_PROOF"
        if candidate is not None
        else "CLASSIFICATION_ONLY"
    )
    if observed_mode != expected_mode:
        raise KleanFinalGateError(
            "Stage 6 mechanical mode differs from audit input"
        )
    before_hashes = _input_hashes(
        frozen_input,
        discovery_manifest,
        generation,
        candidate,
    )
    _require_bound_inputs(audit_binding, before_hashes)
    try:
        preflight = klean_preflight.check_generation(
            frozen_input,
            discovery_manifest,
            generation,
            toolchain_lock=toolchain_lock,
            run_command=run_command,
        )
    except klean_preflight.KleanPreflightError as error:
        raise KleanFinalGateError(
            f"deterministic Klean preflight failed: {error}"
        ) from error
    preflight_status = preflight.get("status")
    if preflight_status not in {"PASS", "KLEAN_NO_OBLIGATIONS"}:
        raise KleanFinalGateError(
            "Stage 4 Klean preflight is not mechanically auditable"
        )
    manifest = _regular_json(
        generation / "generator-manifest.json", "generator manifest"
    )
    target = manifest.get("target")
    if preflight_status == "KLEAN_NO_OBLIGATIONS":
        if candidate is not None:
            raise KleanFinalGateError(
                "classification-only audit must not use a Stage 5 Lean "
                "candidate"
            )
        if target is not None or preflight.get("obligation_count") != 0:
            raise KleanFinalGateError(
                "KLEAN_NO_OBLIGATIONS has a target or nonempty domain set"
            )
        after_hashes = _input_hashes(
            frozen_input,
            discovery_manifest,
            generation,
            candidate,
        )
        if after_hashes != before_hashes:
            raise KleanFinalGateError(
                "resolved inputs changed during mechanical audit"
            )
        _require_bound_inputs(audit_binding, after_hashes)
        return {
            "status": "PASS",
            "mode": "CLASSIFICATION_ONLY",
            "resolved_input_sha256": audit_binding[
                "resolved_input_sha256"
            ],
            "input_hashes": signed_resolution["hashes"],
            "semantic_classification": "NOT_EVALUATED",
            "target": None,
            "klean_preflight": preflight,
            "generation_sha256": klean_export.tree_digest(generation),
            "candidate_sha256": None,
            "used_axioms": [],
            "diagnostics": [],
        }
    if candidate is None:
        raise KleanFinalGateError(
            "proof-bearing audit requires a Stage 5 Lean candidate"
        )
    proof_check = check_proof_candidate(
        generation, candidate, run_command=run_command
    )
    after_hashes = _input_hashes(
        frozen_input,
        discovery_manifest,
        generation,
        candidate,
    )
    if after_hashes != before_hashes:
        if (
            after_hashes.get("lean_workspace_sha256")
            != before_hashes.get("lean_workspace_sha256")
        ):
            raise KleanFinalGateError(
                "candidate changed during mechanical audit"
            )
        raise KleanFinalGateError(
            "resolved inputs changed during mechanical audit"
        )
    _require_bound_inputs(audit_binding, after_hashes)
    return {
        "status": "PASS",
        "mode": "CLASSIFICATION_AND_PROOF",
        "resolved_input_sha256": audit_binding[
            "resolved_input_sha256"
        ],
        "input_hashes": signed_resolution["hashes"],
        "semantic_classification": "NOT_EVALUATED",
        "target": proof_check["target"],
        "klean_preflight": preflight,
        "generation_sha256": proof_check["generation_sha256"],
        "candidate_sha256": proof_check["candidate_sha256"],
        "used_axioms": proof_check["used_axioms"],
        "diagnostics": proof_check["diagnostics"],
    }


def run_gate(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    candidate: Path | None,
    *,
    toolchain_lock: Path,
    audit_input: Path,
    output: Path,
) -> dict[str, Any]:
    output = Path(output)
    if output.exists() or output.is_symlink():
        raise KleanFinalGateError(f"output already exists: {output}")
    try:
        document = check_final(
            frozen_input,
            discovery_manifest,
            generation,
            candidate,
            toolchain_lock=toolchain_lock,
            audit_input=audit_input,
        )
    except KleanFinalGateError as error:
        document = {
            "status": "FAIL",
            "error": str(error),
            **_result_binding(audit_input),
        }
    except BaseException as error:
        document = {
            "status": "AUDIT_ERROR",
            "error": f"{type(error).__name__}: {error}",
            **_result_binding(audit_input),
        }
    temporary = output.with_name(f".{output.name}.tmp")
    temporary.write_text(json.dumps(document, indent=2, sort_keys=True) + "\n")
    os.replace(temporary, output)
    return document


def evaluate_gate(
    frozen_input: Path,
    discovery_manifest: Path,
    generation: Path,
    candidate: Path | None,
    *,
    toolchain_lock: Path,
    audit_input: Path,
) -> dict[str, Any]:
    try:
        return check_final(
            frozen_input,
            discovery_manifest,
            generation,
            candidate,
            toolchain_lock=toolchain_lock,
            audit_input=audit_input,
        )
    except KleanFinalGateError as error:
        return {
            "status": "FAIL",
            "error": str(error),
            **_result_binding(audit_input),
        }
    except BaseException as error:
        return {
            "status": "AUDIT_ERROR",
            "error": f"{type(error).__name__}: {error}",
            **_result_binding(audit_input),
        }


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--frozen-k", required=True, type=Path)
    parser.add_argument(
        "--discovery-manifest", required=True, type=Path
    )
    parser.add_argument("--generation", required=True, type=Path)
    parser.add_argument("--candidate", type=Path)
    parser.add_argument("--toolchain-lock", required=True, type=Path)
    parser.add_argument("--audit-input", required=True, type=Path)
    parser.add_argument("--output", type=Path)
    arguments = parser.parse_args(argv)
    try:
        if arguments.output is None:
            result = evaluate_gate(
                arguments.frozen_k,
                arguments.discovery_manifest,
                arguments.generation,
                arguments.candidate,
                toolchain_lock=arguments.toolchain_lock,
                audit_input=arguments.audit_input,
            )
        else:
            result = run_gate(
                arguments.frozen_k,
                arguments.discovery_manifest,
                arguments.generation,
                arguments.candidate,
                toolchain_lock=arguments.toolchain_lock,
                audit_input=arguments.audit_input,
                output=arguments.output,
            )
    except KleanFinalGateError as error:
        print(f"Klean final gate failed: {error}", file=sys.stderr)
        return 2
    print(json.dumps(result, sort_keys=True))
    return 0 if result["status"] == "PASS" else 1


if __name__ == "__main__":
    raise SystemExit(main())
