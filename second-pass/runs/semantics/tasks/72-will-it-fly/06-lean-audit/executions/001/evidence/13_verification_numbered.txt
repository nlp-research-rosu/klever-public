     1	requires "reference-semantics/semantics.k"
     2	
     3	module VERIFICATION
     4	  imports MPY
     5	
     6	  // Mathematical model used by the contract.
     7	  syntax Bool ::= allInts(ValSeq) [function, total]
     8	  rule allInts(.ValSeq) => true
     9	  rule allInts(vCons(_:Int, REST:ValSeq)) => allInts(REST)
    10	  rule allInts(vCons(V:Val, _:ValSeq)) => false
    11	    requires notBool isInt(V)
    12	
    13	  syntax Int ::= sumIntVS(ValSeq) [function, total]
    14	  rule sumIntVS(.ValSeq) => 0
    15	  rule sumIntVS(vCons(I:Int, REST:ValSeq)) => I +Int sumIntVS(REST)
    16	
    17	  syntax ValSeq ::= snocVS(ValSeq, Val) [function, total]
    18	  rule snocVS(.ValSeq, V:Val) => vCons(V, .ValSeq)
    19	  rule snocVS(vCons(H:Val, T:ValSeq), V:Val)
    20	    => vCons(H, snocVS(T, V))
    21	
    22	  syntax ValSeq ::= reverseVS(ValSeq) [function, total]
    23	  rule reverseVS(.ValSeq) => .ValSeq
    24	  rule reverseVS(vCons(H:Val, T:ValSeq))
    25	    => snocVS(reverseVS(T), H)
    26	
    27	  // Symbolic summaries of two terminating folds in the supplied semantics.
    28	  // They agree with subscript.k's q[::-1] and builtins.k's sum(q), while
    29	  // allowing a proof over an arbitrary (rather than fixed-length) ValSeq.
    30	  rule doSlice(list(VS:ValSeq), noB, noB, someB(-1))
    31	    => list(reverseVS(VS))
    32	    [priority(40)]
    33	
    34	  rule <k> #sumAcc(list(VS:ValSeq), 0) => sumIntVS(VS) ... </k>
    35	    requires allInts(VS)
    36	    [priority(40)]
    37	
    38	  syntax Expr ::= "willItFlyResult"
    39	  rule willItFlyResult
    40	    => BoolOp(
    41	         "and",
    42	         Compare(
    43	           Name("q"),
    44	           CmpOp(
    45	             "==",
    46	             Subscript(
    47	               Name("q"),
    48	               Slice(NoBound, NoBound, UnaryOp("-", Int(1)))))),
    49	         Compare(Call(Name("sum"), Name("q")), CmpOp("<=", Name("w"))))
    50	
    51	  // The translated entry point, factored into named terms so the claims can
    52	  // execute definition, lookup, argument binding, return, and frame cleanup.
    53	  syntax Module ::= "willItFlyModule" [function]
    54	  rule willItFlyModule
    55	    => Module(
    56	         FuncDef(
    57	           "will_it_fly",
    58	           Params("q", "w"),
    59	           Return(willItFlyResult)))
    60	
    61	  syntax Val ::= "willItFlyClosure" [function]
    62	  rule willItFlyClosure
    63	    => closureVal(("q", "w"), Return(willItFlyResult), 0)
    64	endmodule
