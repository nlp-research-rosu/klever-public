     1	def bf(planet1, planet2):
     2	    planets = (
     3	        "Mercury",
     4	        "Venus",
     5	        "Earth",
     6	        "Mars",
     7	        "Jupiter",
     8	        "Saturn",
     9	        "Uranus",
    10	        "Neptune",
    11	    )
    12	    if planet1 not in planets or planet2 not in planets:
    13	        return ()
    14	    first = planets.index(planet1)
    15	    second = planets.index(planet2)
    16	    if first < second:
    17	        return planets[first + 1:second]
    18	    return planets[second + 1:first]
