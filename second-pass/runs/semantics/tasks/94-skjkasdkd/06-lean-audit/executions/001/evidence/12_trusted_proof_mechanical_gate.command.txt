import json
import os
import subprocess
from pathlib import Path
from tools.klean_final_gate import check_proof_candidate

lean = (
    "/opt/elan/toolchains/"
    "leanprover--lean4---v4.22.0/bin/lean"
)

def run(command, *, cwd, timeout):
    environment = dict(os.environ)
    environment.update({
        "LD_PRELOAD": "/tmp/audit-work/libproc_exe_shim.so",
        "LAKE_HOME": (
            "/opt/elan/toolchains/leanprover--lean4---v4.22.0"
        ),
        "LEAN_SYSROOT": (
            "/opt/elan/toolchains/leanprover--lean4---v4.22.0"
        ),
    })
    actual = command
    if command[:3] == ["lake", "env", "lean"]:
        actual = [lean, *command[3:]]
        environment["LEAN_PATH"] = (
            f"{Path(cwd) / '.lake/build/lib/lean'}:"
            "/tmp/klean-generated-build/lib/lean"
        )
    completed = subprocess.run(
        actual,
        cwd=cwd,
        timeout=timeout,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        env=environment,
    )
    return completed.returncode, completed.stdout

result = check_proof_candidate(
    Path("/reference/klean-generation"),
    Path("/candidate"),
    run_command=run,
)
print(json.dumps(result, indent=2, sort_keys=True))
