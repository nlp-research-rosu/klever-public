from tools.k_rule_inventory import inventory_verification

manifest_class = {
    entry["source_rule_id"]: entry["classification"]
    for entry in manifest["rules"]
}

independent = []

        classification = "OPERATIONAL_RULE"
        rationale = (
            "A configuration or <k>-cell execution/observation step: local "
            "lookup, assignment/binding, strictness-normalized comparison/"
            "control/call dispatch, or structural iterator execution."
        )
    elif 195 <= line <= 248:
        classification = "DEFINITION"
        rationale = (
            "A guarded base/recursive equation defining a named mathematical "
            "summary: trialPrime, trialDivisor, isPrime, largestPrime, "
            "digitAcc, or digitSum."
        )
    elif 252 <= line <= 302:
        classification = "DEFINITION"
        rationale = (
            "A macro equation defining a named AST fragment from the exact "
            "translated source function."
