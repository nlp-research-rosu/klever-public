$ nl -ba /reference/k-proof/solution.py
     1	def encode(message):
     2	    result = ""
     3	    char = ""
     4	    code = 0
     5	    for char in message:
     6	        char = char.swapcase()
     7	        code = ord(char)
     8	        if code == 97 or code == 101 or code == 105 or code == 111 or code == 117 or code == 65 or code == 69 or code == 73 or code == 79 or code == 85:
     9	            result += chr(code + 2)
    10	        else:
    11	            result += char
    12	    return result
