     1	def minSubArraySum(nums):
     2	    """
     3	    Given an array of integers nums, find the minimum sum of any non-empty
     4	    sub-array of nums.
     5	    """
     6	    smallest = nums[0]
     7	    current = 0
     8	    value = 0
     9	    for value in nums:
    10	        current = current + value
    11	        if value < current:
    12	            current = value
    13	        if current < smallest:
    14	            smallest = current
    15	    return smallest
