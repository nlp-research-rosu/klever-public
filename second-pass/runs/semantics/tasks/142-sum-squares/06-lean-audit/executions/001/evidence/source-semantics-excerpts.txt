$ nl -ba /reference/k-proof/solution.py | sed -n "1,80p"
     1	def sum_squares(lst):
     2	    total = 0
     3	    index = 0
     4	    for value in lst:
     5	        if index % 3 == 0:
     6	            total += value * value
     7	        else:
     8	            if index % 4 == 0:
     9	                total += value * value * value
    10	            else:
    11	                total += value
    12	        index += 1
    13	    return total
