def _without_comments(text: str) -> str:
    """Remove K comments while preserving strings and their exact contents."""

    output: list[str] = []
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                output.append(character)
                state = "code"
            index += 1
            continue
        if state == "string":
            output.append(character)
            if character == "\\" and following:
                output.append(following)
                index += 2
                continue
            if character == '"':
                state = "code"
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character in "\r\n":
                output.append(character)
            index += 1
            continue
        if character == "/" and following == "/":
            output.append(" ")
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output.append(" ")
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        output.append(character)
        if character == '"':
            state = "string"
        index += 1
    return "".join(output)


def _attribute_tokens(text: str, masked_text: str) -> list[str]:
    tokens: list[str] = []
    start = 0
    depth = 0
    for index, character in enumerate(masked_text):
        if character in "([{":
            depth += 1
        elif character in ")]}":
            depth = max(depth - 1, 0)
        elif character == "," and depth == 0:
            token = _without_comments(text[start:index]).strip()
            if token:
                tokens.append(token)
            start = index + 1
    final = _without_comments(text[start:]).strip()
    if final:
        tokens.append(final)
    return tokens
