   290	      | .succ l, 0 => pad :: bytes l 0
   291	      | .succ l, n => ⟨(n % 256).toNat⟩ :: bytes l (n / 256)
   292	    bytes x0.toNat x1

   294	-- Adapted from
   295	-- https://github.com/runtimeverification/haskell-backend/blob/362dab30d6435ec117862fea722be67373572034/kore/src/Kore/Builtin/InternalBytes.hs#L527-L543
   296	-- Note that we use `List.foldl` and not `ByteArray.foldl` for ease of reasoning
   297	def «Bytes2Int(_,_,_)_BYTES-HOOKED_Int_Bytes_Endianness_Signedness» (bytes : SortBytes) (endian : SortEndianness) (sign : SortSignedness) : Option SortInt :=
   298	  match sign with
   299	  | .unsignedBytes => Int.ofNat unsigned
   300	  | .signedBytes => if 2 * unsigned >= modulus then (Int.ofNat unsigned) - (Int.ofNat modulus)
   301	                    else Int.ofNat unsigned
   302	  where
   303	  modulus : Nat := res.1
   304	  unsigned : Nat := res.2
   305	  res : Nat×Nat :=
   306	    let littleEndian := match endian with
   307	                        | .littleEndianBytes => bytes.toList
   308	                            --match bytes with |⟨⟨l⟩⟩ => l
   309	                        | .bigEndianBytes => bytes.toList.reverse
   310	                            --match bytes with |⟨⟨l⟩⟩ => l.reverse
   311	    let go (res : Nat×Nat) (b : UInt8) : Nat×Nat :=
   312	      -- `place` is `res.1`
   313	      -- `acc` is `res.2`
   314	      let place := res.1 * 0x100
   315	      let acc := res.2 + res.1 * b.toNat
   316	      ⟨place, acc⟩
   317	    List.foldl go (1, 0) littleEndian
   318	
   319	/--
   320	Pads to the right `len - b.length` bytes with specified `val` value
   321	-/
   322	def «padRightBytes(_,_,_)_BYTES-HOOKED_Bytes_Bytes_Int_Int» (b : SortBytes) (len val : SortInt) : Option SortBytes :=
   323	  if val < 0 ∨ 255 < val then none else
   324	  some { data := (Array.rightpad len.toNat (⟨val.toNat⟩ : UInt8) b.data)}
   325	
   326	/--
   327	Pads to the left `len - b.length` bytes with specified `val` value
   328	-/
   329	def «padLeftBytes(_,_,_)_BYTES-HOOKED_Bytes_Bytes_Int_Int» (b : SortBytes) (len val : SortInt) : Option SortBytes :=
   330	  if val < 0 ∨ 255 < val then none else
   331	  some { data := (Array.leftpad len.toNat (⟨val.toNat⟩ : UInt8) b.data)}
   332	
   333	def «lengthBytes(_)_BYTES-HOOKED_Int_Bytes» (x0 : SortBytes) : Option SortInt :=
   334	  Int.ofNat x0.size
   335	
   336	/--
   337	Replaces the contents of `dest` from index `index` with the contents of `src`
   338	If `dest.size < index + src.size` the result is `none`
   339	-/
   340	def «replaceAtBytes(_,_,_)_BYTES-HOOKED_Bytes_Bytes_Int_Bytes» (dest : SortBytes) (index : SortInt) (src : SortBytes) : Option SortBytes :=
   341	  if index < 0 then none else
   342	  if dest.size < index + src.size then none else
   343	  let init := dest.data.extract 0 index.toNat
   344	  let rem := dest.data.extract (index.toNat + src.size) dest.size
   345	  some { data := init ++ src.data ++ rem }
   346	
   347	/--
   348	Get a new `Bytes` object containing a range of bytes from the input `Bytes`
   349	-/
   350	def «substrBytes(_,_,_)_BYTES-HOOKED_Bytes_Bytes_Int_Int» (b : SortBytes) (startIndex : SortInt) (endIndex : SortInt) : Option SortBytes :=
   351	  if startIndex < 0 then none else
   352	  if endIndex < startIndex then none else
   353	  if b.size < endIndex then none else
   354	  b.extract startIndex.toNat endIndex.toNat
