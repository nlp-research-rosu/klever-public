     1	def largest_smallest_integers(lst):
     2	    largest_negative = 0
     3	    smallest_positive = 0
     4	    value = 0
     5	
     6	    for value in lst:
     7	        if value < 0:
     8	            if largest_negative == 0 or value > largest_negative:
     9	                largest_negative = value
    10	        if value > 0:
    11	            if smallest_positive == 0 or value < smallest_positive:
    12	                smallest_positive = value
    13	
    14	    if largest_negative == 0:
    15	        largest_negative = None
    16	    if smallest_positive == 0:
    17	        smallest_positive = None
    18	
    19	    return (largest_negative, smallest_positive)
