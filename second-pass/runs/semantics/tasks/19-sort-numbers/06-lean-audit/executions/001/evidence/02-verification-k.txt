     1	requires "reference-semantics/semantics.k"
     2	
     3	module SORT-NUMBERS-VERIFICATION
     4	  imports MPY
     5	
     6	  // The exact statement bodies emitted in solution.mpy.  Naming them keeps
     7	  // the proof claims readable without changing the translated program.
     8	  syntax Stmts ::= "numberBody" [macro]
     9	  rule numberBody =>
    10	       If(Compare(Name("word"), CmpOp("==", Str("zero"))),
    11	          Return(Int(0)), .Stmts)
    12	       If(Compare(Name("word"), CmpOp("==", Str("one"))),
    13	          Return(Int(1)), .Stmts)
    14	       If(Compare(Name("word"), CmpOp("==", Str("two"))),
    15	          Return(Int(2)), .Stmts)
    16	       If(Compare(Name("word"), CmpOp("==", Str("three"))),
    17	          Return(Int(3)), .Stmts)
    18	       If(Compare(Name("word"), CmpOp("==", Str("four"))),
    19	          Return(Int(4)), .Stmts)
    20	       If(Compare(Name("word"), CmpOp("==", Str("five"))),
    21	          Return(Int(5)), .Stmts)
    22	       If(Compare(Name("word"), CmpOp("==", Str("six"))),
    23	          Return(Int(6)), .Stmts)
    24	       If(Compare(Name("word"), CmpOp("==", Str("seven"))),
    25	          Return(Int(7)), .Stmts)
    26	       If(Compare(Name("word"), CmpOp("==", Str("eight"))),
    27	          Return(Int(8)), .Stmts)
    28	       Return(Int(9))
    29	
    30	  syntax Stmts ::= "sortBody" [macro]
    31	  rule sortBody =>
    32	       Return(
    33	         Call(
    34	           Attribute(Str(" "), "join"),
    35	           Call(
    36	             Name("sorted"),
    37	             Call(Attribute(Name("numbers"), "split"), .Exprs),
    38	             KwArg("key", Name("number_value")))))
    39	
    40	  syntax Module ::= "solutionModule" [macro]
    41	  rule solutionModule =>
    42	       Module(
    43	         FuncDef("number_value", Params("word"), numberBody)
    44	         FuncDef("sort_numbers", Params("numbers"), sortBody))
    45	
    46	  // These are the actual module-level closures produced by loading the two
    47	  // translated definitions.
    48	  syntax Val ::= "numberKey" [macro]
    49	  rule numberKey => closureVal("word", numberBody, 0)
    50	
    51	  // An arbitrary valid input is represented by a sequence whose element sort
    52	  // admits only the ten numeral words.
    53	  syntax NumWord ::= "zeroW" | "oneW" | "twoW" | "threeW" | "fourW"
    54	                   | "fiveW" | "sixW" | "sevenW" | "eightW" | "nineW"
    55	  syntax NumWords ::= ".NumWords" | nw(NumWord, NumWords)
    56	
    57	  syntax Val ::= wordVal(NumWord) [function, total]
    58	  rule wordVal(zeroW)  => str(strToCodes("zero"))
    59	  rule wordVal(oneW)   => str(strToCodes("one"))
    60	  rule wordVal(twoW)   => str(strToCodes("two"))
    61	  rule wordVal(threeW) => str(strToCodes("three"))
    62	  rule wordVal(fourW)  => str(strToCodes("four"))
    63	  rule wordVal(fiveW)  => str(strToCodes("five"))
    64	  rule wordVal(sixW)   => str(strToCodes("six"))
    65	  rule wordVal(sevenW) => str(strToCodes("seven"))
    66	  rule wordVal(eightW) => str(strToCodes("eight"))
    67	  rule wordVal(nineW)  => str(strToCodes("nine"))
    68	
    69	  syntax ValSeq ::= wordsVS(NumWords) [function, total]
    70	  rule wordsVS(.NumWords)    => .ValSeq
    71	  rule wordsVS(nw(W, WORDS)) => vCons(wordVal(W), wordsVS(WORDS))
    72	
    73	  syntax IntSeq ::= wordCodes(NumWord) [function, total]
    74	  rule wordCodes(zeroW)  => strToCodes("zero")
    75	  rule wordCodes(oneW)   => strToCodes("one")
    76	  rule wordCodes(twoW)   => strToCodes("two")
    77	  rule wordCodes(threeW) => strToCodes("three")
    78	  rule wordCodes(fourW)  => strToCodes("four")
    79	  rule wordCodes(fiveW)  => strToCodes("five")
    80	  rule wordCodes(sixW)   => strToCodes("six")
    81	  rule wordCodes(sevenW) => strToCodes("seven")
    82	  rule wordCodes(eightW) => strToCodes("eight")
    83	  rule wordCodes(nineW)  => strToCodes("nine")
    84	
    85	  // The equations give encodedWords its concrete meaning. It remains an
    86	  // ordinary constructor on an abstract WORDS variable, which is the
    87	  // documented problem-local abstraction boundary for symbolic split.
    88	  syntax IntSeq ::= encodedWords(NumWords)
    89	  rule encodedWords(.NumWords) => .IntSeq
    90	  rule encodedWords(nw(W, .NumWords)) => wordCodes(W)
    91	  rule encodedWords(nw(W, nw(W2, WORDS)))
    92	    => seqConcat(wordCodes(W),
    93	                 iCons(32, encodedWords(nw(W2, WORDS))))
    94	
    95	  rule <k> #applyK(
    96	             toCall(boundMethodV(str(encodedWords(WORDS)), "split")),
    97	             .Vals)
    98	        => #alloc(list(wordsVS(WORDS))) ... </k>
    99	       [priority(30)]
   100	
   101	  // MPY-SORT specifies keyed sorting symbolically with the trusted primitive
   102	  // sortKeyVS.  Joining that sequence is precisely the contract result.
   103	  syntax Val ::= numericOutput(NumWords) [function, total]
   104	  rule numericOutput(WORDS)
   105	    => str(joinCodes(iCons(32, .IntSeq),
   106	                     sortKeyVS(wordsVS(WORDS), numberKey)))
   107	endmodule
