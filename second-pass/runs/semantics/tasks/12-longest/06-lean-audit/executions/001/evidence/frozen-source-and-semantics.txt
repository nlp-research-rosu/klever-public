     3	
     4	def longest(strings: List[str]) -> Optional[str]:
     5	    result = None
     6	    string = None
     7	    for string in strings:
     8	        if result is None:
     9	            result = string
    10	        elif len(string) <= len(result):
    11	            result = result
    12	        elif len(string) > len(result):
    13	            result = string
