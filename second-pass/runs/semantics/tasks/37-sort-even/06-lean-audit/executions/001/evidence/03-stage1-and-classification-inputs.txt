     1	def sort_even(l: list):
     2	    evens = sorted(l[::2])
     3	    odds = l[1::2]
     4	    result = []
     5	    i = 0
     6	    odd = 0
     7	    for odd in odds:
     8	        result.append(evens[i])
     9	        result.append(odd)
    10	        i += 1
    11	    return result + evens[len(odds):]
