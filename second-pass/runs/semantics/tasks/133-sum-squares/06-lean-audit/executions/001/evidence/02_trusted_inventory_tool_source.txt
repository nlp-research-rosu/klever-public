"""Canonical lexical inventory of simplification rules in verification.k."""

import dataclasses

import json

_MODULE_NAME_RE = re.compile(
    r"^[ \t]+(?P<name>[A-Za-z][A-Za-z0-9_'-]*)"
    r"(?:[ \t]+\[[^\]\r\n]*\])?[ \t]*$"
)
_TRAILING_ATTRIBUTES_RE = re.compile(
    r"\[(?P<attributes>[^\[\]]*)\][ \t\r\n]*\Z", re.S
)
_CELL_TAG_RE = re.compile(
    r"<(?P<close>/)?(?P<name>[A-Za-z][A-Za-z0-9_'-]*)"
    r"[ \t]*(?P<self_close>/)?>"
)
_OUTER_SENTENCE_KEYWORDS = {
    "imports",
    "syntax",
    "rule",
    "configuration",
    "context",
    "claim",
    "alias",
}

def canonical_json_sha256(document: Any) -> str:
    encoded = json.dumps(
        document,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
    ).encode()
    return hashlib.sha256(encoded).hexdigest()


def _mask_non_code(text: str) -> str:
    """Mask comments and string bodies without changing source offsets."""

    output = list(text)
    index = 0
    block_depth = 0
    state = "code"
    while index < len(text):
        character = text[index]
        following = text[index + 1] if index + 1 < len(text) else ""
        if state == "line-comment":
            if character in "\r\n":
                state = "code"
            else:
                output[index] = " "
            index += 1
            continue
        if state == "string":
            if character == "\\" and following:
                output[index] = " "
                output[index + 1] = " "
                index += 2
                continue
            if character == '"':
                state = "code"
            elif character not in "\r\n":
                output[index] = " "
            index += 1
            continue
        if state == "block-comment":
            if character == "/" and following == "*":
                output[index] = output[index + 1] = " "
                block_depth += 1
                index += 2
                continue
            if character == "*" and following == "/":
                output[index] = output[index + 1] = " "
                block_depth -= 1
                index += 2
                if block_depth == 0:
                    state = "code"
                continue
            if character not in "\r\n":
                output[index] = " "
            index += 1
            continue
        if character == "/" and following == "/":
            output[index] = output[index + 1] = " "
            state = "line-comment"
            index += 2
            continue
        if character == "/" and following == "*":
            output[index] = output[index + 1] = " "
            state = "block-comment"
            block_depth = 1
            index += 2
            continue
        if character == '"':
            state = "string"
        index += 1
    return "".join(output)


def _line_ranges(text: str, start: int = 0) -> list[tuple[int, int, int]]:
    ranges: list[tuple[int, int, int]] = []
    line_start = start
    while line_start < len(text):
        newline = text.find("\n", line_start)
        next_start = len(text) if newline < 0 else newline + 1
        line_end = len(text) if newline < 0 else newline
        content_end = (
            line_end - 1
            if line_end > line_start and text[line_end - 1] == "\r"
            else line_end
        )
        ranges.append((line_start, content_end, next_start))
        if newline < 0:
            break
        line_start = next_start
    return ranges


def _line_heads(text: str) -> list[tuple[int, int, str, str]]:
    """Return physical-line word heads without assigning K grammar meaning."""

    heads: list[tuple[int, int, str, str]] = []
    for line_start, content_end, _next_start in _line_ranges(text):
        index = line_start
        while index < content_end and text[index] in " \t":
            index += 1
        word_end = index
        while word_end < content_end and (
            text[word_end].isalnum() or text[word_end] in "_'"
        ):
            word_end += 1
        if word_end > index:
            heads.append(
                (
                    line_start,
                    content_end,
                    text[index:word_end],
                    text[word_end:content_end],
                )
            )
    return heads


def _line_word(line: str) -> tuple[str, str, int]:
    stripped = line.lstrip()
    indent = len(line) - len(stripped)
    word_end = 0
    while word_end < len(stripped) and (
        stripped[word_end].isalnum() or stripped[word_end] in "_'"
    ):
        word_end += 1
    return stripped[:word_end], stripped[word_end:], indent + word_end


def _rule_continuation(line: str) -> tuple[str, int] | None:
    stripped = line.lstrip()
    if stripped.startswith("["):
        return "attributes", len(line) - len(stripped)
    word, rest, word_end = _line_word(line)
    for keyword in ("requires", "ensures"):
        if word == keyword and (
            not rest
            or not (
                rest[0].isalnum()
                or rest[0] in "_'"
            )
        ):
            return keyword, word_end
    return None


def _starts_outer_transition(line: str) -> bool:
    keyword, rest, _word_end = _line_word(line)
    separated = bool(rest) and rest[0] in " \t"
    if keyword == "module":
        return separated and _MODULE_NAME_RE.fullmatch(rest) is not None
    if keyword == "endmodule":
        return not rest.strip()
    if keyword == "rule":
        return not rest or separated
    return keyword in _OUTER_SENTENCE_KEYWORDS and separated


def _terminal_attribute_line(line: str) -> bool:
    match = _TRAILING_ATTRIBUTES_RE.search(line)
    if match is None:
        return False
    start = match.start()
    return start == 0 or line[start - 1].isspace()


def _scan_rule_delimiters(
    line: str,
    delimiters: list[str],
) -> None:
    closing = {")": "(", "]": "[", "}": "{"}
    index = 0
    while index < len(line):
        if line[index] == "<":
            cell = _CELL_TAG_RE.match(line, index)
            if cell is not None:
                marker = f"<{cell.group('name')}>"
                if cell.group("close"):
                    if delimiters and delimiters[-1] == marker:
                        delimiters.pop()
                elif not cell.group("self_close"):
                    delimiters.append(marker)
                index = cell.end()
                continue
        character = line[index]
        if character in "([{":
            delimiters.append(character)
        elif character in closing:
            if delimiters and delimiters[-1] == closing[character]:
                delimiters.pop()
        index += 1


def _rule_bubble_end(masked_text: str, start: int) -> int:
    """Find when an outer rule bubble returns control to outer parsing."""

    seen_rewrite = False
    clause_has_content = False
    candidate_end: int | None = None
    delimiters: list[str] = []
    ranges = _line_ranges(masked_text, start)
    for line_start, line_end, next_start in ranges:
        line = masked_text[line_start:line_end]
        if not line.strip():
            continue

        continuation = (
            _rule_continuation(line)
            if seen_rewrite and candidate_end is not None
            else None
        )
        if candidate_end is not None:
            if continuation is None and _starts_outer_transition(line):
                return candidate_end
            candidate_end = None

        content_start = 0
        if continuation is not None:
            kind, content_start = continuation
            if kind in {"requires", "ensures"}:
                clause_has_content = False

        if not seen_rewrite:
            rewrite = line.find("=>")
            if rewrite >= 0:
                seen_rewrite = True
                content_start = rewrite + 2
            else:
                content_start = len(line)

        _scan_rule_delimiters(line, delimiters)
        if seen_rewrite and line[content_start:].strip():
            clause_has_content = True

        if (
            seen_rewrite
            and clause_has_content
            and not delimiters
            and _terminal_attribute_line(line)
        ):
            return next_start
        if seen_rewrite and clause_has_content and not delimiters:
            candidate_end = next_start
    if candidate_end is not None:
        return candidate_end
    return len(masked_text) + 1

#!/usr/bin/env python3
"""Validate Stage 3 classifications against the canonical K rule inventory."""

from __future__ import annotations

import json
import stat
from pathlib import Path
from typing import Any

from tools.k_rule_inventory import KRuleInventoryError, inventory_verification


_MANIFEST_KEYS = {"schema_version", "inventory_sha256", "rules"}
_RULE_KEYS = {"source_rule_id", "classification", "rationale"}
_CLASSIFICATIONS = {
    "DEFINITION",
    "OPERATIONAL_RULE",
    "PROVED_DERIVED_LEMMA",
    "DOMAIN_LEMMA",
}


class LemmaDiscoveryContractError(RuntimeError):
    """Raised when a Stage 3 manifest crosses its trust boundary."""


def _regular_json(path: Path) -> dict[str, Any]:
    path = Path(path)
    try:
        mode = path.stat(follow_symlinks=False).st_mode
    except OSError as error:
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest must be a regular file: {path}"
        ) from error
    if not stat.S_ISREG(mode):
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest must be a regular file: {path}"
        )
    try:
        document = json.loads(path.read_bytes())
    except (OSError, UnicodeError, ValueError) as error:
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest is malformed: {path}"
        ) from error
    if not isinstance(document, dict):
        raise LemmaDiscoveryContractError(
            "lemma discovery manifest must be a JSON object"
        )
    return document


def _exact_keys(
    document: dict[str, Any], expected: set[str], label: str
) -> None:
    unexpected = sorted(set(document) - expected)
    if unexpected:
        raise LemmaDiscoveryContractError(
            f"{label} has unexpected key: {unexpected[0]}"
        )
    missing = sorted(expected - set(document))
    if missing:
        raise LemmaDiscoveryContractError(
            f"{label} is missing key: {missing[0]}"
        )


def validate_trust_boundary(
    workspace: Path, manifest: Path
) -> dict[str, Any]:
    try:
        inventory = inventory_verification(Path(workspace))
    except KRuleInventoryError as error:
        raise LemmaDiscoveryContractError(str(error)) from error
    document = _regular_json(Path(manifest))
    _exact_keys(document, _MANIFEST_KEYS, "lemma discovery manifest")
    schema_version = document["schema_version"]
    if (
        not isinstance(schema_version, int)
        or isinstance(schema_version, bool)
        or schema_version != 2
    ):
        raise LemmaDiscoveryContractError(
            "lemma discovery schema_version must be 2"
        )
    if document["inventory_sha256"] != inventory["inventory_sha256"]:
        raise LemmaDiscoveryContractError(
            "lemma discovery inventory_sha256 does not match canonical inventory"
        )
    entries = document["rules"]
    if not isinstance(entries, list):
        raise LemmaDiscoveryContractError(
            "lemma discovery rules must be a JSON list"
        )
    canonical_rules = inventory["rules"]
    canonical_by_id = {
        rule["source_rule_id"]: rule for rule in canonical_rules
    }
    classified_by_id: dict[str, dict[str, str]] = {}
    for index, entry in enumerate(entries):
        label = f"lemma discovery rule {index}"
        if not isinstance(entry, dict):
            raise LemmaDiscoveryContractError(f"{label} must be an object")
        _exact_keys(entry, _RULE_KEYS, label)
        source_rule_id = entry["source_rule_id"]
        if not isinstance(source_rule_id, str) or not source_rule_id:
            raise LemmaDiscoveryContractError(
                f"{label} source_rule_id must be a nonempty string"
            )
        if source_rule_id in classified_by_id:
            raise LemmaDiscoveryContractError(
                f"duplicate lemma discovery source_rule_id: {source_rule_id}"
            )
        if source_rule_id not in canonical_by_id:
            raise LemmaDiscoveryContractError(
                f"unknown lemma discovery source_rule_id: {source_rule_id}"
            )
        classification = entry["classification"]
        if (
            not isinstance(classification, str)
            or classification not in _CLASSIFICATIONS
        ):
            raise LemmaDiscoveryContractError(
                f"{label} classification must be DEFINITION, "
                "OPERATIONAL_RULE, PROVED_DERIVED_LEMMA, or DOMAIN_LEMMA"
            )
        canonical_rule = canonical_by_id[source_rule_id]
        if (
            "simplification" in canonical_rule["attributes"]
            and classification
            in {"OPERATIONAL_RULE", "PROVED_DERIVED_LEMMA"}
        ):
            raise LemmaDiscoveryContractError(
                f"{label} simplification rule must be classified as "
                "DEFINITION or DOMAIN_LEMMA"
            )
        rationale = entry["rationale"]
        if not isinstance(rationale, str) or not rationale.strip():
            raise LemmaDiscoveryContractError(
                f"{label} rationale must be a nonempty string"
            )
        classified_by_id[source_rule_id] = {
            "classification": classification,
            "rationale": rationale,
        }
    missing = [
        rule["source_rule_id"]
        for rule in canonical_rules
        if rule["source_rule_id"] not in classified_by_id
    ]
    if missing:
        raise LemmaDiscoveryContractError(
            f"lemma discovery manifest is missing source_rule_id: {missing[0]}"
        )
    definitions: list[dict[str, Any]] = []
    operational_rules: list[dict[str, Any]] = []
    proved_derived_lemmas: list[dict[str, Any]] = []
    domain_lemmas: list[dict[str, Any]] = []
    for rule in canonical_rules:
        classification = classified_by_id[rule["source_rule_id"]]
        classified_rule = {**rule, **classification}
        role = classification["classification"]
        if role == "DEFINITION":
            definitions.append(classified_rule)
        elif role == "OPERATIONAL_RULE":
            operational_rules.append(classified_rule)
        elif role == "PROVED_DERIVED_LEMMA":
            proved_derived_lemmas.append(classified_rule)
        else:
            domain_lemmas.append(classified_rule)
    return {
        **inventory,
        "definitions": definitions,
        "operational_rules": operational_rules,
        "proved_derived_lemmas": proved_derived_lemmas,
        "domain_lemmas": domain_lemmas,
    }
