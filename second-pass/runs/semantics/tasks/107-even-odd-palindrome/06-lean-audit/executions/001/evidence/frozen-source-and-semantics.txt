     1	def even_odd_palindrome(n):
     2	    if n < 10:
     3	        return (n // 2, (n + 1) // 2)
     4	
     5	    if n < 100:
     6	        two_digit = n // 11
     7	        return (4 + two_digit // 2, 5 + (two_digit + 1) // 2)
     8	
     9	    if n == 1000:
    10	        return (48, 60)
    11	
    12	    hundreds = n // 100
    13	    current_block = (n % 100 - hundreds) // 10 + 1
    14	    even = 8 + 10 * ((hundreds - 1) // 2)
    15	    odd = 10 + 10 * (hundreds // 2)
    16	
    17	    if hundreds % 2 == 0:
    18	        even += current_block
    19	    else:
    20	        odd += current_block
    21	
    22	    return (even, odd)

     3	// the cooled node hands to applyUn/applyBin/applyCmp (declared in core.k; each domain file
     4	// owns its cases — bool.k, int.k, float.k, str.k, list.k, tuple.k, set.k, dict.k).
     5	
     6	module MPY-OPERATORS
     7	  imports MPY-CORE
     8	  imports MPY-ITER
     9	
    10	  rule <k> UnaryOp(OP:String, V:Val) => applyUn(OP, V) ... </k>
    11	
    12	  rule <k> BinOp(OP:String, L:Val, R:Val) => applyBin(OP, L, R) ... </k>
    13	
    14	  // Compare's right operand sits under the CmpOp wrapper — contexts instead of attributes
    15	  context Compare(HOLE, _)
    16	  context Compare(_:Val, CmpOp(_, HOLE))
    17	  rule <k> Compare(LV:Val, CmpOp(OP:String, RV:Val)) => applyCmp(OP, LV, RV) ... </k> [owise]
    18	
    19	  rule applyCmp("is",     V:Val, noneV) => V ==K noneV
    20	  rule applyCmp("is not", V:Val, noneV) => notBool (V ==K noneV)
     9	  // ==== iteration (the iterator protocol's tuple case) ======================
    10	  rule <k> #iterNext(tuple(.ValSeq))                => #iterDone ... </k>
    11	  rule <k> #iterNext(tuple(vCons(V:Val, R:ValSeq))) => #iterYield(V, tuple(R)) ... </k>
    12	
    13	  // ==== TupleExpr: tuple(ValSeq) literal + == != ===========================
    14	  syntax ApplyK ::= "toTuple"
    15	  rule <k> TupleExpr(ES:Exprs) => #evalArgs(ES, .Vals, toTuple) ... </k>
    16	  rule <k> #applyK(toTuple, ACC:Vals) => tuple(vals2valSeq(ACC)) ... </k>
    17	
    18	  rule applyCmp("==", tuple(A:ValSeq), tuple(B:ValSeq)) => A ==K B
